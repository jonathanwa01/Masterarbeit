\chapter{Algebraic Spaces}

We have seen in ... that the representable presheaf of a scheme is indeeed a sheaf in $\SchOver{Et}$. Via the Yoneda embedding we can identify a scheme $U$ with its correspoding sheaf and denote it also by $U$.

\begin{definition}[Representable Morphism]
	(\cite[][(Definition 4.1.1), p.~105]{alper2026stacks})
	A morphism $X\to Y$ of sheaves over $\Sch$ is called representable by schemes, if for every scheme $T$ and every morphism $T\to Y$, the fibre product $X\times_Y T$ is a scheme.
\end{definition}

\begin{definition}[Algebraic Space]
	(\cite[][(Definition 4.1.2), p.~105]{alper2026stacks})
	An algebraic space $X$ is a sheaf $X:\SchOver{Et} \to \Set$, such that there exists a scheme $U$ and an \etale\ surjection $U\to X$ representable by schemes.
\end{definition}

\begin{remark}
	In analogy with schemes, which are Zariski-locally given by affine schemes, an algebraic space is a sheaf admitting an \etale\ cover by schemes, and hence, after refinement (\ref{prop:affine-etale-refinement}), by affine schemes. Thus an algebraic space is a sheaf that is \etale-locally a scheme.
	
	Indeed, let $U \to X$ be an \etale\ surjection, where $U$ is a scheme, and let $\{U_i\}$ be an open affine cover of $U$. Then the induced morphism
	\[
	\coprod_i U_i \to X
	\]
	is again an \etale\ surjection. It follows that every algebraic space admits an \etale\ cover by affine schemes.
\end{remark}





\begin{theorem}[Representability of the Diagonal]
	(\cite[][(Theorem 4.2.1), p.~118]{alper2026stacks})
	The diagonal $\Delta_X: X\to X\times X$ of an algebraic space $X$ is represantable by schemes.
\end{theorem}

\begin{proposition}
	Let $U$ be a scheme and let

	\[\begin{tikzcd}
		R & {U\times_S U}
		\arrow["t"', shift right, from=1-1, to=1-2]
		\arrow["s", shift left, from=1-1, to=1-2]
	\end{tikzcd}\]
	be an etale equivalence relation. Denote the quotient of presheaves by $F:= U/R$ and let $X = F^\sharp$ be its sheafification. Then there is an isomorphism of sheaves
	\[
	U\times_X U \cong R.
	\]
	
\end{proposition}

\begin{proof}
	For every sscheme $T$ we get an isomorphism $(U\times_X U)(T) \cong U(T) \times_{X(T))} U(T)$. Via Yoneda, elements are given by morphisms $f,g: U(T)$, such that $\pi(f) = \pi(g)\in X(T)$, where $\pi:U\to X$ denotes the canonical morphism $U\to F \to F^\sharp$ of the sheafification. By \cite[][Tag 00WK]{stacks-project} for the section $\sigma:= \pi(f) = \pi (g) \in F^\sharp(T)$, we get an \etale\ cover $\{T_i\to T\}_i$ and elements $\sigma_i\in F(T_i) = U(T_i) / R(T_i)$, such that 
	\begin{enumerate}
		\item $\sigma \mid_{T_i} = \eta^2(\sigma_i)$, where $\eta^2$ denotes the counit at $F$, see \ref{lem:sheafification}.
		\item For every $i,j$ there exists an \etale\ covering $\{T_{i,j,k}\to T_i\times_T T_j\}_k$, such that $(\sigma_i)\mid_{T_{i,j,k}} = (\sigma_j)\mid_{T_{i,j,k}}$.
	\end{enumerate}
	By definition of $F(T_i) = U(T_i)/R(T_i)$, each $\sigma_i$ is represented by a pair or morphisms $f_i, g_i:T_i \to U$, which are in the same equivalence class under $R$. Hence, there is an $h_i\in R(T_i)$, such that $s\circ h_i = f_i$ and $t\circ h_i = g_i$.  
	
	By 1), we can choose representatives $f, g$, such that $f\mid_{T_i} = f_i$ and $g\mid_{T_i} = g_i$. 
	So \etale\ locally on $T_i$, $(f_i,g_i)$ comes from a morphism $h_i$ to $R$.
	These local $h_i:T_i\to R$ agree on the overlaps $T_i\times_T T_j$, because they were defined by the same section $\sigma_i$ of the sheaf $F^\sharp$. Since $R$ is representable, hence a sheaf in the \etale\ site, these $h_i$ glue to a unique morphism $h:T\to R$. Thus, from $(f,g)\in U\times_X U)(T)$ we get an $h\in R(T)$ with
	\[
	(f,g) = (s\circ h, t\circ h).
	\]
	Conversely, starting with a map $h:T\to R$ composing with $s,t$ yields
	\[
	f:= s\circ h:T\to U, g:= t\circ h:T\to U.
	\]
	These represent by definition the same class in $F(T)$, hence are equal in $X(T)$.
\end{proof}
