\chapter{Algebraic Spaces}

\section{Algebraic Spaces}

While schemes are Zariski local given by affine schemes, we coarse the topology towards the \etale\ topology. Algebraic spaces are therefore thought of as "spaces", given etale locally by schemes.

\begin{definition}[Representable Morphism]
	(\cite[][(Definition 4.1.1), p.~105]{alper2026stacks})
	A morphism $X\to Y$ of sheaves over $\Sch$ is called representable by schemes, if for every scheme $T$ and every morphism $T\to Y$, the fibre product $X\times_Y T$ is a scheme.
\end{definition}

We have seen in Theorem \ref{thm:representable-fppf-sheaf} that the representable presheaf of a scheme is indeeed a sheaf in $\SchOver{fppf}$. Via the Yoneda embedding we identify a scheme $U$ with its correspoding sheaf and denote it also by $U$. 

\begin{definition}[\Etale\ morphisms of sheaves]
	\label{def:etale-morphisms-of-spaces}
	Given a morphism $f:X\to Y$ of sheaves over the big \etale\ site. We call $f$ \etale, if $f$ is representable and for every scheme $T\to Y$, the projection $X\times_YT\to T$ is \etale.
\end{definition}

\begin{remark}
	Note, that via Yoneda the morphism $X\times_Y T \to T$ corresponds to a morphism of schemes, since both are representable by assumption.
\end{remark}

\begin{definition}[Algebraic Space]
	(\cite[][(Definition 4.1.2), p.~105]{alper2026stacks})
	An algebraic space $X$ is a sheaf $X:\SchOver{S} \to \Set$ in the fppf-topology, such that there exists a scheme $U$ and an \etale\ surjection $U\to X$ representable by schemes.
\end{definition}

\begin{remark}
	In analogy with schemes, which are Zariski-locally given by affine schemes, an algebraic space is a sheaf admitting an \etale\ cover by schemes, and hence, after refinement, by affine schemes. Thus an algebraic space is a sheaf that is \etale-locally a scheme.
	
	Indeed, let $U \to X$ be an \etale\ surjection, where $U$ is a scheme, and let $\{U_i\}$ be an open affine cover of $U$. Then the induced morphism
	\[
	\coprod_i U_i \to X
	\]
	is again an \etale\ surjection. It follows that every algebraic space admits an \etale\ cover by affine schemes.
\end{remark}

\begin{theorem}[Representability of the Diagonal]
	(\cite[][(Theorem 4.2.1), p.~118]{alper2026stacks})
	The diagonal $\Delta_X: X\to X\times X$ of an algebraic space $X$ is represantable by schemes.
\end{theorem}

\begin{remark}
	A groupoid is a category in which every morphism is invertible.
	To express this notion geometrically, we regard the objects and
	morphisms as spaces $U$ and $R$, respectively, equipped with source
	and target maps
	\[
	s,t\colon R\rightrightarrows U,
	\]
	together with identity, inversion, and composition maps satisfying
	the usual groupoid axioms.
	This formulation allows us to consider groupoids internal to
	categories such as schemes or algebraic spaces, and to impose
	geometric conditions on their structure maps.
	In particular, we call such a groupoid \etale\ if both $s$ and $t$
	are \etale.
\end{remark}

\begin{example}
	(\cite[][Example 4.4.3]{alper2026stacks})
	Let $G$ be group scheme acting on a scheme $U$ via multiplication $\sigma: G\times U\to U$. Then
	\[
		\mathrm{pr_2}, \sigma: G\times U \rightrightarrows U
	\]
	is n \etale groupoid of schemes. The inverse is given by 
	\begin{align*}
		i:G\times U &\to G\times U \\
		(g, u) \mapsto (g^{-1}, gu),
	\end{align*}
	and the composition is given by
	\begin{align*}
		c:(G\times U) \times_{\mathrm{pr}_2, U, \sigma} (G\times U) &\to G\times U \\
		((g, u), (g', u') | u' = gu) &\mapsto (g'g, u).
	\end{align*}

\end{example}

\begin{definition}[\Etale\ equivalence relation]
	An \etale\ equivalence relation on a scheme $U$ is an etale groupoid $R\rightrightarrows U$, such that
	\[
		R\rightarrow U\times U,
	\]
	is a monomorphism, i.e. for every scheme $T$, $R(T) \to U(T)\times U(T)$ is a injective.
\end{definition}

\begin{remark}
	Identity morphisms give reflexivity, inverses give symmetry and composition gives transitivity.
\end{remark}


\begin{proposition}\label{prop:etale-quotient-isom}
	Let $U$ be a scheme and let
	% https://q.uiver.app/#q=WzAsMixbMCwwLCJSIl0sWzEsMCwiVVxcdGltZXNfUyBVIl0sWzAsMSwidCIsMix7Im9mZnNldCI6MX1dLFswLDEsInMiLDAseyJvZmZzZXQiOi0xfV1d
	\[\begin{tikzcd}
		R & {U\times_S U}
		\arrow["t"', shift right, from=1-1, to=1-2]
		\arrow["s", shift left, from=1-1, to=1-2]
	\end{tikzcd}\]
	be an \etale\ equivalence relation. Denote by $F:=U/R$ the quotient presheaf and let
	\[
	X:=F^\sharp
	\]
	be its sheafification. Then the canonical morphism
	\[
	R \to U\times_X U,\qquad h\mapsto (s\circ h,t\circ h),
	\]
	is an isomorphism of sheaves.
\end{proposition}

\begin{proof}
	Let $p:U\to X$ be the canonical map. For every scheme $T$, the set
	\[
	(U\times_XU)(T)
	\]
	consists of pairs of morphisms $f,g:T\to U$ such that $p\circ f=p\circ g$ in $X(T)$.
	
	Define a morphism of sheaves
	\[
	\varphi:R\to U\times_XU,\qquad h\mapsto (s\circ h,t\circ h).
	\]
	It is well-defined because $s(h)$ and $t(h)$ define the same class in $F(T)$, hence have the same image in $X(T)$.
	
	To prove surjectivity, let $(f,g)\in (U\times_XU)(T)$. Since $p\circ f=p\circ g$ in $X(T)$, the two sections of $F(T)$ defined by $f$ and $g$ become equal in the sheafification. Hence, by \cite[][Tag 00WK]{stacks-project} there exists a covering $\{T_i\to T\}$ such that the restrictions of $f$ and $g$ have the same image in
	\[
	F(T_i)=U(T_i)/R(T_i).
	\]
	Therefore, for each $i$, there exists $h_i\in R(T_i)$ with
	\[
	s\circ h_i=f|_{T_i},\qquad t\circ h_i=g|_{T_i}.
	\]
	On overlaps, the restrictions of $h_i$ and $h_j$ have the same images under $(s,t):R\to U\times_SU$. Since $(s,t)$ is a monomorphism, they are equal. Because $R$ is representable, it is a sheaf, so the $h_i$ glue uniquely to a morphism $h:T\to R$ with
	\[
	s\circ h=f,\qquad t\circ h=g.
	\]
	Thus $\varphi(T)$ is surjective.
	
	For injectivity, suppose $h,h'\in R(T)$ satisfy
	\[
	s\circ h=s\circ h',\qquad t\circ h=t\circ h'.
	\]
	Since $(s,t):R\to U\times_SU$ is a monomorphism, it follows that $h=h'$. Thus $\varphi(T)$ is injective.
	
	Hence $\varphi$ is an isomorphism of sheaves.
\end{proof}





\begin{lemma}[Quotients by \etale\ equivalence relations]\label{lem:quotient-by-etale-eq}
	(\cite[Tag 0265]{stacks-project})
	Let $U$ be an affine scheme and let
	\[
	R \rightrightarrows U
	\]
	be an \etale\ equivalence relation. Then the sheaf
	\[
	X:=(U/R)^\sharp
	\]
	is an algebraic space, and the canonical morphism
	\[
	U\to X
	\]
	is surjective and \etale.
	We will denote the algebraic space $X$ by $U/R$.
\end{lemma}


\begin{proof}
	Let $X$ be the fppf sheafification of the presheaf quotient
	$U/R$, and write $p:U\to X$ for the canonical morphism.
	We first show that the canonical morphism $p:U\to X$ is surjective. Let $T$ be a scheme and let
	\[
	\sigma\in X(T).
	\]
	By the local description of sheafification (\cite[][Tag 00WK]{stacks-project}), there exists an \etale\ covering $\{T_i\to T\}_i$ and sections
	\[
	\sigma_i\in (U/R)(T_i)=U(T_i)/R(T_i)
	\]
	whose images in $X(T_i)$ are equal to $\sigma|_{T_i}$. Choosing representatives of the classes $\sigma_i$, we obtain morphisms
	\[
	f_i:T_i\to U
	\]
	such that
	\[
	p\circ f_i=\sigma|_{T_i}.
	\]
	Hence $p:U\to X$ is surjective as a morphism of sheaves.
	
	We first see that $t:R\to U$ is separated. Since $U$ is affine, $U$ is separated. The canonical morphism
	\[
	R \longrightarrow U\times U
	\longrightarrow
	U\times U
	\]
	is a monomorphism, hence separated. Consequently, $R$
	is separated, and therefore
	$t:R\to U$ is separated. Since $t$ is also étale, it is
	locally quasi-finite.
	
	Let $a:T\to X$ be a morphism from a scheme, and set
	\[
	Y:=T\times_X U.
	\]
	Choose an fppf covering $\{T_i\to T\}_{i\in I}$ and
	morphisms $f_i:T_i\to U$ such that
	\[
	p\circ f_i=a|_{T_i},
	\]
	i.e. 
	% https://q.uiver.app/#q=WzAsNCxbMSwwLCJVIl0sWzEsMSwiWCJdLFswLDEsIlQiXSxbMCwwLCJUX2kiXSxbMiwxLCJhIiwyXSxbMCwxLCJwIl0sWzMsMl0sWzMsMCwiZl9pIl1d
	\[\begin{tikzcd}
		{T_i} & U \\
		T & X
		\arrow["{f_i}", from=1-1, to=1-2]
		\arrow[from=1-1, to=2-1]
		\arrow["p", from=1-2, to=2-2]
		\arrow["a"', from=2-1, to=2-2]
	\end{tikzcd}\]
	commutes. By the Proposition \ref{prop:etale-quotient-isom}, $U\times_X U\cong R$,
	with projections corresponding to $t$ and $s$.
	Associativity of fibre products therefore gives
	\[
	Y\times_T T_i
	\cong T_i\times_X U
	\cong T_i\times_{f_i,U,t}R
	=:Y_i.
	\]
	In particular, $Y_i$ is a scheme, since $R$ and $U$ are schemes, and $Y_i\to T_i$
	is separated and étale, being a base change of $t$.
	
	The identifications $Y_i\cong Y\times_T T_i$ endow the
	schemes $Y_i$ with a canonical descent datum relative
	to $\{T_i\to T\}$. Since each $Y_i\to T_i$ is separated
	and locally quasi-finite, this descent datum is effective
	by \cite[Lemma 37.57.1]{stacks-project}.
	Thus there exists a scheme $V\to T$ whose associated
	fppf sheaf is isomorphic to $Y$.
	Hence $Y$ is representable by a scheme.
	Hene, $p$ is representable by schemes.
	
	Moreover, the base change of $Y\to T$ to each $T_i$
	is the étale morphism $Y_i\to T_i$.
	Since being étale is fppf local on the target,
	$Y\to T$ is étale. Thus $p$ is étale.
	
	Finally, $t:R\to U$ has a section given by the identity
	of the equivalence relation, and hence is surjective.
	Therefore each $Y_i\to T_i$ is surjective.
	Surjectivity descends along the covering
	$\{T_i\to T\}$, so $Y\to T$ is surjective.
	Consequently, $p:U\to X$ is representable, surjective,
	and étale. Since $X$ is an fppf sheaf, it is an
	algebraic space.
\end{proof}

	Suppose $X$ is a scheme, the diagonal $\Delta_{X/k}$ is an immersion and via base change, $j$ is an immersion.  $\Delta$ is open in $R$ and $\Delta$ and $\Gamma$ are disjoint, but since every open neighborhood of $(0,0)$ in intersects $\Gamma$, $j(\Delta)$ is not open in $j(R)$ and $j$ is not a homeomorphism, and hence, not an immersion. Contradiction.
\end{example}

\begin{remark}
	This example illustrates perfectly, why algebriac spaces naturally expands the category of schemes. Although $R\rightrightarrows U$ are \etale\ morphisms of schemes, the quotient need not be a scheme again.
\end{remark}
