\begin{proposition}[Grothendieck construction]\label{prop:grothendieck-construction}
	Let $\mathcal C$ be a category. There is an equivalence between
	\begin{enumerate}
		\item categories fibred in groupoids over $\mathcal C$, and
		\item pseudofunctors
		\[
		F : \mathcal C^{op} \to \Grpd.
		\]
	\end{enumerate}
	
	Under this equivalence, a fibred category $\mathcal F \to \mathcal C$
	is sent to the pseudofunctor
	\[
	B \longmapsto \mathcal F(B),
	\]
	which assigns to each object its fibre and to each morphism
	$f : B' \to B$ the pullback functor
	\[
	f^* : \mathcal F(B) \to \mathcal F(B').
	\]
	
	Conversely, a pseudofunctor determines a category fibred in groupoids
	via the Grothendieck construction.
	
	See \cite[][Section~3.1.2]{vistoli2007notes} and
	\cite[][Tags 02XJ, 003S]{stacks-project}.
\end{proposition}

\begin{remark}
	This construction originates in SGA~1, Exp.~VI, due to
	Alexander Grothendieck, where categories fibred in groupoids are introduced as a framework for descent.
	
	The appearance of pseudofunctors rather than ordinary functors reflects
	the fact that pullbacks in a fibred category are only defined up to
	canonical isomorphism. In particular, for composable morphisms
	\[
	B'' \xrightarrow{f} B' \xrightarrow{g} B
	\]
	one has a natural isomorphism
	\[
	f^* g^* \Rightarrow (g \circ f)^*,
	\]
	and similarly for identities. These isomorphisms satisfy coherence
	conditions, giving rise to a pseudofunctor rather than a strict functor.
	
	From a modern higher-categorical point of view, this is not a defect
	but the correct notion of functoriality. The failure of strict
	functoriality is resolved by passing to homotopy-coherent functors.
	Accordingly, the Grothendieck construction may be viewed as a special
	case of a more general principle, namely that Grothendieck fibrations
	classify functors valued in categories. In higher category theory, this
	extends to the straightening-unstraightening equivalence between
	functors into $\infty$-categories and cartesian fibrations, see
	\cite[Chapter~2.4 and Section~3.2]{lurie2008higher}.
\end{remark}

\section{Stacks}
Conceptually, stacks generalize sheaves by allowing values in groupoids
instead of sets. Equivalently, stacks are $2$-sheaves on a site,
or sheaves valued in $\infty$-groupoids.

\begin{definition}[Stack]\label{def:stack}
	(\cite[][Tag 02ZH]{stacks-project})
	Let $(\CatC,J)$ be a site, and let
	\[
	p_{\mathcal{F}}:\mathcal{F}\to\CatC
	\]
	be a category fibred in groupoids. We call $\mathcal{F}$ a stack over $(\CatC,J)$ if for every object $B\in\CatC$ and every covering $\{f_i:U_i\to U\}_{i\in I}$ in $J$, the following conditions hold:
	\begin{enumerate}
		\item \textbf{Sheaf condition for isomorphisms.}
		For every pair of objects $X,Y\in\mathcal{F}(B)$, the presheaf
		\begin{align*}
			Isom_U(X,Y):(\CatC/U)^{op} &\to\Set \\
			(g:U'\to U) &\mapsto
			Isom_{\mathcal{F}(U')}(g^*X,g^*Y)
		\end{align*}
	
		is a sheaf on the slice site $(\CatC/U,J_U)$.
		
		\item \textbf{Effective descent for objects.}
		Let $\{f_i:U_i\to U\}_{i\in I}$ be a covering, and for each $i$ let $X_i\in\mathcal{F}(U_i)$ be an object. Suppose moreover that for every pair $i,j$ there is an isomorphism
		\[
		\varphi_{ij}:\pi_1^*X_i\xrightarrow{\sim}\pi_2^*X_j
		\qquad\text{in }\mathcal{F}(U_i\times_U U_j),
		\]
		where $\pi_1:U_i\times_U U_j\to U_i$ and $\pi_2:U_i\times_U U_j\to U_j$ are the projections, and that these isomorphisms satisfy the cocycle condition on triple fibre products
		\[
		U_i\times_U U_j\times_U U_k.
		\]
		Then there exists an object $X\in\mathcal{F}(U)$ together with isomorphisms
		\[
		\alpha_i:f_i^*X\xrightarrow{\sim}X_i
		\qquad\text{in }\mathcal{F}(U_i)
		\]
		such that for every pair $i,j$ the diagram
		\[
		\begin{tikzcd}
			\pi_1^*f_i^*X \arrow[r,"\pi_1^*\alpha_i"] \arrow[d,"\cong"']
			& \pi_1^*X_i \arrow[d,"\varphi_{ij}"] \\
			\pi_2^*f_j^*X \arrow[r,"\pi_2^*\alpha_j"']
			& \pi_2^*X_j
		\end{tikzcd}
		\]
		commutes in $\mathcal{F}(U_i\times_U U_j)$.
	\end{enumerate}
\end{definition}

\begin{remark}
	Following Definition 2.6 of \cite[][p. 13]{khan2022stacks} we get a nice $\infty$-categorical picture of a stack, namely as a $2$-sheaf, satisfying descent.
	
	The passage from algebraic spaces to stacks may be viewed as a categorification
	of the sheaf condition. Recall that an algebraic space may be described by its
	functor of points as a sheaf of sets on the big \'etale site. Following remark \ref{rem:sheaf-ses-cech}, for a covering
	$U \to X$, the sheaf condition may be expressed by the equalizer diagram
	\[
	F(X) \to F(U) \rightrightarrows F(U \times_X U).
	\]
	Thus, sections over $X$ are precisely those sections over $U$ whose pullbacks
	to $U \times_X U$ agree.
	
	For moduli problems, however, local objects usually do not agree literally on
	overlaps, but only up to isomorphism. This leads to replacing sets by
	groupoids. One therefore considers an assignment
	\[
	U \longmapsto \mathcal X(U),
	\]
	where $\mathcal X(U)$ is a groupoid, together with pullback functors along
	morphisms. 
	In this language, the sheaf condition is replaced by descent. For a covering
	$U \to X$, a descent datum consists of an object $x \in \mathcal X(U)$
	together with an isomorphism
	\[
	\pi_2^*x \xrightarrow{\cong} \pi_1^*x
	\]
	in $\mathcal X(U \times_X U)$ satisfying the cocycle condition on
	$U \times_X U \times_X U$. The stack condition asserts that every such
	descent datum is effective. If all fibres are discrete groupoids, this reduces
	to the usual sheaf condition for set-valued functors.
\end{remark}

\begin{example}[A quotient stack of pencils of quadrics up to orthogonal action]
	Let \(k\) be a field with \(\operatorname{char}(k)\neq 2\), and let \(V\) be an
	\(n\)-dimensional \(k\)-vector space. Let
	\[
	q \in \operatorname{Sym}^2(V^\vee)
	\]
	be a nondegenerate quadratic form. We denote by \(O(q)\) the corresponding
	orthogonal group scheme, defined by
	\[
	O(q)
	=
	\{g\in \operatorname{GL}(V) \mid q(gv,gw)=q(v,w)
	\text{ for all } v,w\in V\}.
	\]
	After choosing a basis of \(V\), this can be written as the closed subgroup
	scheme
	\[
	O(q)
	=
	\{g\in \operatorname{GL}_n \mid g^T Q g = Q\},
	\]
	where \(Q\) is the symmetric matrix representing \(q\).
	
	Consider the Grassmannian
	\[
	X
	=
	\operatorname{Gr}\bigl(2,\operatorname{Sym}^2(V^\vee)\bigr).
	\]
	A \(k\)-point of \(X\) is a two-dimensional linear subspace
	\[
	P \subset \operatorname{Sym}^2(V^\vee),
	\]
	which may be interpreted as a pencil of quadratic forms on \(V\).
	
	The group \(O(q)\) acts on \(\operatorname{Sym}^2(V^\vee)\) by change of
	coordinates. Explicitly, for \(g\in O(q)\) and
	\(A\in \operatorname{Sym}^2(V^\vee)\), define
	\[
	(g\cdot A)(v,w)
	=
	A(g^{-1}v,g^{-1}w).
	\]
	In matrix notation, this action is given by
	\[
	g\cdot A
	=
	(g^{-1})^T A g^{-1}.
	\]
	This induces an action of \(O(q)\) on \(X\) by
	\[
	g\cdot P
	=
	\{g\cdot A \mid A\in P\}.
	\]
	
	We define the quotient stack of pencils of quadratic forms modulo orthogonal
	change of coordinates to be
	\[
	\mathcal X
	=
	\left[
	\operatorname{Gr}\bigl(2,\operatorname{Sym}^2(V^\vee)\bigr) / O(q)
	\right].
	\]
	
	For a \(k\)-scheme \(S\), an object of \(\mathcal X(S)\) consists of an
	\(O(q)\)-torsor
	\[
	\pi\colon E\to S
	\]
	together with an \(O(q)\)-equivariant morphism
	\[
	E \longrightarrow
	\operatorname{Gr}\bigl(2,\operatorname{Sym}^2(V^\vee)\bigr).
	\]
	Equivalently, it is a rank-two subbundle
	\[
	\mathcal P
	\subset
	E\times^{O(q)} \operatorname{Sym}^2(V^\vee).
	\]
	Thus the stack \(\mathcal X\) classifies families of pencils of quadratic forms
	on vector bundles locally isomorphic to \(V\), equipped with an orthogonal
	structure modeled on \(q\), modulo orthogonal changes of coordinates.
	
	This quotient stack is typically not an algebraic space. Indeed, the stabilizer
	of a pencil
	\[
	P\in \operatorname{Gr}\bigl(2,\operatorname{Sym}^2(V^\vee)\bigr)
	\]
	is the group scheme
	\[
	\operatorname{Stab}_{O(q)}(P)
	=
	\{g\in O(q) \mid g\cdot P=P\}.
	\]
	The central element \(-I\in O(q)\) acts trivially on
	\(\operatorname{Sym}^2(V^\vee)\), since
	\[
	(-I)\cdot A
	=
	((-I)^{-1})^T A (-I)^{-1}
	=
	(-I)^T A (-I)
	=
	A.
	\]
	Therefore \(-I\) fixes every pencil \(P\). Hence every stabilizer contains the
	subgroup
	\[
	\{\pm I\}\cong \mu_2.
	\]
	Consequently, the quotient stack
	\[
	\left[
	\operatorname{Gr}\bigl(2,\operatorname{Sym}^2(V^\vee)\bigr) / O(q)
	\right]
	\]
	has nontrivial stabilizers at every point. In particular, it is a genuinely
	stacky quotient rather than an ordinary quotient variety or algebraic space.
\end{example}

\begin{lemma}[Products and pullbacks of stacks]
	\cite[][04TD]{stacks-project}
	The category of stacks over a base scheme $S$ is closed under products and pullbacks.
\end{lemma}

\section{Algebraic Stacks}

\begin{definition}[Representable morphism of stacks]
	(\cite[][Tag 02YQ]{stacks-project})
	Let $\mathcal{X}, \mathcal{Y}$ be stacks over $(\SchOver{S})_{fppf}$ and let $f:\mathcal{X} \to \mathcal{Y}$ be a morphism of stacks. We call $f$ representable by algebraic spaces, if for all schemes $U$ over $S$ and $U\to \mathcal{Y}$, $U\times_{\mathcal{Y}} \mathcal{X}$ is an algebraic space.
\end{definition}



\begin{definition}[Algebraic Stack]
	(\cite[][Tag 026O]{stacks-project})
	Let $S$ be a scheme. An algebraic stack over $S$ is a category fibred in groupoids
	\[
	\mathcal{X} \to (\SchOver{S})_{\mathrm{fppf}}
	\]
	such that:
	\begin{enumerate}
		\item $\mathcal{X}$ is a stack in groupoids over $(\SchOver{S})_{\mathrm{fppf}}$;
		\item the diagonal
		\[
		\Delta_{\mathcal{X}} \colon \mathcal{X} \to \mathcal{X} \times \mathcal{X}
		\]
		is representable by algebraic spaces;
		\item there exists a scheme $U$ and a smooth surjective morphism
		\[
		U \to \mathcal{X}.
		\]
	\end{enumerate}
\end{definition}

\begin{proposition}
	\label{prop:algspaces-are-algstacks}
	There is a fully faithful embedding of algebraic spaces into algebraic stacks $\AlgSp \subset \AlgStk$.
\end{proposition}

\begin{proof}
	This follows from the Grothendieck construction, proposition \ref{prop:grothendieck-construction}, and the fact that smooth morphisms are \etale.
\end{proof}

\begin{definition}[Points of algebraic stacks]
	(\cite[][Tag 04XE]{stacks-project})
	Let $\mathcal{X}$ be an algebraic stack, $K, L$ two fields and let $p:\Specof{K}\to \mathcal{X}$, $q: \Specof{L}\to \mathcal{X}$ be a morphisms of stacks. We call $p$ and $q$ equivalent, if there exists a field $\Omega$ such that the following diagram is $2$-commutative:
	% https://q.uiver.app/#q=WzAsNCxbMCwwLCJcXFNwZWNvZntcXE9tZWdhfSJdLFsxLDAsIlxcU3BlY29me0t9Il0sWzAsMSwiXFxTcGVjb2Z7TH0iXSxbMSwxLCJcXG1hdGhjYWx7WH0iXSxbMSwzLCJwIl0sWzIsMywicSIsMl0sWzAsMl0sWzAsMV1d
	\[\begin{tikzcd}
		{\Specof{\Omega}} & {\Specof{K}} \\
		{\Specof{L}} & {\mathcal{X}}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-1, to=2-1]
		\arrow["p", from=1-2, to=2-2]
		\arrow["q"', from=2-1, to=2-2]
	\end{tikzcd}\]
	A point of $\mathcal{X}$ is an equivalence class of morphisms $\Specof{K}\to \mathcal{X}$ for a field $K$. We write $|\mathcal{X}I$ for the set of points of $\mathcal{X}$.
\end{definition}

\begin{remark}
	The fact that $|\mathcal{X}|$ is a set rather than a proper class relies crucially on $\mathcal{X}$ being an algebraic stack. For a general stack, the collection of all isomorphism classes of morphisms $\Specof{K}\to \mathcal{X}$, as $K$ ranges over all fields, forms a priori only a proper class in ZFC set theory, since there is no bound on the cardinality of $K$.
	
	For an algebraic stack however, this issue is resolved by the existence of a smooth surjective presentation $U\to \mathcal{X}$ from a scheme $U$. Since $|U|$ is a genuine set and the map $U\to \mathcal{X}$ is surjective (\cite[][Tag 04XI]{stacks-project}), every point of $|\mathcal{X}|$ is represented by a point of $U$. One then sees, that $|\mathcal{X}|$ is the set-theoretic quotient $|U|/|R|$, where $R=U\times_X U$ is the algebraic space encoding the equivalence relation. This is indeed a set, and one verifies that it agrees with the description via field-valued points.
\end{remark}

\begin{definition}[\Etale\ morphism of algebraic stacks]
	Let $f:\mathcal{X} \to \mathcal{Y}$ be a morphism of algebraic stacks. We call $f$ \etale, if it is representable and for all schemes $U\to \mathcal{Y}$, the morphism $U\times_\mathcal{Y}\mathcal{X}\to U$ is \etale.
\end{definition}

\begin{definition}[Inertia stack]
	(\cite[][Tag 036X]{stacks-project})
	Let \(\mathcal{X}\) be an algebraic stack over \(S\). The inertia stack of \(\mathcal{X}\), denoted
	\[
	\mathcal{I}_{\mathcal X},
	\]
	is the stack over \((\SchOver{S})_{\mathrm{fppf}}\) defined as follows:
	
	For every scheme $U$ over $S$, the groupoid \(I_{\mathcal X}(U)\) has
	\begin{enumerate}
		\item objects given by pairs \((x,\alpha)\), where
		\[
		x \in \mathcal X(U)
		\quad\text{and}\quad
		\alpha \in \operatorname{Aut}_{\mathcal X(U)}(x),
		\]
		\item morphisms
		\[
		(x,\alpha) \to (y,\beta)
		\]
		given by morphisms \(f : x \to y\) in \(\mathcal X(U)\) such that
		\[
		f \circ \alpha = \beta \circ f.
		\]
	\end{enumerate}
\end{definition}

\begin{remark}
	The inertia stack should be thought of as the stack-theoretic analogue of the stabilizer of a group action. Since stacks take values in groupoids, points may have non-trivial automorphism groups, and the inertia stack records these automorphisms globally.
	
	Let $G$ be an algebraic group acting on a scheme $X$. For a point $x\in X$, the stabilizer is
	\[
	G_x = \{g\in G \mid g\cdot x = x\}.
	\]
	In the quotient stack $[X/G]$, this stabilizer appears as the automorphism group of the corresponding object:
	\[
	\Aut_{[X/G]}(x) \cong G_x.
	\]
	The inertia stack collects these automorphism groups into a single stack. 
	In the quotient case, this gives 
	\[
	\mathcal{I}_{[X/G]} \cong [\{(x,g)\in X\times G \mid g\cdot x = x\}/G]
	\]
	From a higher categorical perspective, when we view stacks as $2$-truncated $\infty$-stacks, we can view the inertia stack essentially as the loop stack 
	\[
	\mathcal{I}_\mathcal{X} \simeq \text{Map}(S^1, \mathcal{X}).
	\]
	Thus, any non trivial, non free action of an algebraic group $G$ on a scheme $X$ induces a proper quotient stack.
\end{remark}

\begin{lemma}
	Let $\mathcal X$ be an algebraic stack over $S$. Then there is an isomorphism of stacks
	\[
	\mathcal{I}_{\mathcal X}
	\cong
	\mathcal X \times_{\mathcal X \times \mathcal X} \mathcal X,
	\]
	where both morphisms $\mathcal X \to \mathcal X \times \mathcal X$ are equal to the diagonal
	\[
	\Delta_{\mathcal X} : \mathcal X \to \mathcal X \times \mathcal X,
	\]
	i.e. the following diagram is a 2-pullback:
	% https://q.uiver.app/#q=WzAsNCxbMCwxLCJcXG1hdGhjYWx7WH0iXSxbMSwxLCJcXG1hdGhjYWx7WH1cXHRpbWVzIFxcbWF0aGNhbHtYfSJdLFsxLDAsIlxcbWF0aGNhbHtYfSJdLFswLDAsIlxcbWF0aGNhbHtJfV9cXG1hdGhjYWx7WH0iXSxbMCwxLCJcXERlbHRhIiwyXSxbMiwxLCJcXERlbHRhIl0sWzMsMF0sWzMsMl0sWzMsMSwiIiwxLHsic3R5bGUiOnsibmFtZSI6ImNvcm5lciJ9fV1d
	\[\begin{tikzcd}
		{\mathcal{I}_\mathcal{X}} & {\mathcal{X}} \\
		{\mathcal{X}} & {\mathcal{X}\times \mathcal{X}}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-1, to=2-1]
		\arrow["\lrcorner"{anchor=center, pos=0.125}, draw=none, from=1-1, to=2-2]
		\arrow["\Delta", from=1-2, to=2-2]
		\arrow["\Delta"', from=2-1, to=2-2]
	\end{tikzcd}\]
\end{lemma}

\begin{proof}
	Let $U$ be a scheme over $S$. By definition of the fibre product of stacks, an object of
	\[
	\left(\mathcal X \times_{\mathcal X \times \mathcal X} \mathcal X\right)(U)
	\]
	consists of objects $x,y \in \mathcal X(U)$ together with an isomorphism
	\[
	\varphi : \Delta_{\mathcal X}(x) \xrightarrow{\sim} \Delta_{\mathcal X}(y)
	\]
	in $(\mathcal X \times \mathcal X)(U)$.
	
	Now
	\[
	\Delta_{\mathcal X}(x) = (x,x)
	\quad\text{and}\quad
	\Delta_{\mathcal X}(y) = (y,y).
	\]
	Hence an isomorphism
	\[
	\varphi : (x,x) \xrightarrow{\sim} (y,y)
	\]
	is given by a pair of isomorphisms
	\[
	(f,g)
	\]
	with
	\[
	f : x \xrightarrow{\sim} y
	\quad\text{and}\quad
	g : x \xrightarrow{\sim} y
	\]
	in $\mathcal X(U)$.
	
	From such a tuple $(x,y,f,g)$ we obtain a pair
	\[
	(x,\alpha),
	\qquad
	\alpha = g^{-1}\circ f \in \operatorname{Aut}_{\mathcal X(U)}(x).
	\]
	Conversely, given a pair $(x,\alpha)$ with
	\[
	\alpha \in \operatorname{Aut}_{\mathcal X(U)}(x),
	\]
	we obtain an object of the fibre product by taking
	\[
	x=y
	\]
	and the isomorphism
	\[
	(\alpha,id_x) : (x,x) \xrightarrow{\sim} (x,x).
	\]
	These two constructions are inverse to each other up to canonical isomorphism. Thus the objects of
	\[
	\left(\mathcal X \times_{\mathcal X \times \mathcal X} \mathcal X\right)(U)
	\]
	identify with pairs $(x,\alpha)$, where $x \in \mathcal X(U)$ and
	\[
	\alpha \in \operatorname{Aut}_{\mathcal X(U)}(x).
	\]
	It remains to identify the morphisms. A morphism in the fibre product from
	\[
	(x,y,f,g)
	\quad\text{to}\quad
	(x',y',f',g')
	\]
	is given by morphisms
	\[
	a : x \to x',
	\qquad
	b : y \to y'
	\]
	in $\mathcal X(U)$ such that the square
	% https://q.uiver.app/#q=WzAsNCxbMCwwLCIoeCx4KSJdLFsxLDAsIih5LHkpIl0sWzAsMSwiKHgnLHgnKSJdLFsxLDEsIih5Jyx5JykiXSxbMCwyLCIoYSxhKSIsMl0sWzAsMSwiKGYsZykiXSxbMSwzLCIoYixiKSJdLFsyLDMsIihmJyxnJykiXV0=
	\[\begin{tikzcd}
		{(x,x)} & {(y,y)} \\
		{(x',x')} & {(y',y')}
		\arrow["{(f,g)}", from=1-1, to=1-2]
		\arrow["{(a,a)}"', from=1-1, to=2-1]
		\arrow["{(b,b)}", from=1-2, to=2-2]
		\arrow["{(f',g')}", from=2-1, to=2-2]
	\end{tikzcd}\]
	commutes in $(\mathcal X \times \mathcal X)(U)$. This means precisely that
	\[
	b\circ f = f'\circ a
	\qquad\text{and}\qquad
	b\circ g = g'\circ a.
	\]
	Passing to the corresponding automorphisms
	\[
	\alpha = g^{-1}\circ f
	\qquad\text{and}\qquad
	\beta = (g')^{-1}\circ f',
	\]
	one checks that this is equivalent to
	\[
	a \circ \alpha = \beta \circ a.
	\]
	Hence morphisms correspond exactly to morphisms
	\[
	a : x \to x'
	\]
	such that
	\[
	a \circ \alpha = \beta \circ a.
	\]
	
	Therefore the stack
	\[
	\mathcal X \times_{\mathcal X \times \mathcal X} \mathcal X
	\]
	is identified with the inertia stack $I_{\mathcal X}$.
\end{proof}

\begin{corollary}
	There is a natural morphism
	\[
	I_{\mathcal X} \to \mathcal X
	\]
	given by either projection from the fibre product. On objects it sends
	\[
	(x,\alpha) \mapsto x.
	\]
\end{corollary}
