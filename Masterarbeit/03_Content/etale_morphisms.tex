\chapter{\Etale\ Morphisms}

\Etale\ morphisms are the central class of morphisms in this thesis, since they form the basic building blocks of the étale Grothendieck topology on schemes. Before we dive deeper into general schemes, we first recall the relevant notions for rings and affine schemes.


\section{Étale Morphisms}

\begin{definition}[Étale morphism]\label{def:etale-morphism-foundational}
	Let $f:X\to S$ be a morphism of schemes. We call $f$ \emph{étale} if it is locally of finite presentation and formally étale, i.e. if for every first order thickening $T_0\hookrightarrow T$ with $T$ affine, every commutative diagram
	\[
	\begin{tikzcd}
		T_0 \arrow[r] \arrow[d, hook] & X \arrow[d] \\
		T \arrow[r] & S
	\end{tikzcd}
	\]
	admits a unique lift $T\to X$.
\end{definition}

\begin{lemma}[Basic properties of étale morphisms]\label{lem:etale-basic-properties}
	\begin{enumerate}
		\item Étale morphisms are stable under composition.
		\item Étale morphisms are stable under base change.
		\item Every open immersion is étale.
		\item Étale morphisms are smooth.
		\item Étale morphisms are unramified.
		\item Étale morphisms are locally of finite presentation.
		\item The diagonal of an étale morphism is an open immersion.
	\end{enumerate}
\end{lemma}

\begin{lemma}[Étale topology as a refinement of the Zariski topology]\label{lem:zariski-etale}
	Every Zariski open immersion is an étale morphism. Hence every Zariski cover is an étale cover, and the Zariski topology is coarser than the étale topology.
\end{lemma}

\begin{lemma}[Étale morphisms are smooth]\label{lem:etale-smooth}
	Every étale morphism is smooth of relative dimension $0$.
\end{lemma}

\begin{lemma}[Smooth morphisms are fppf locally presentable]\label{lem:smooth-fppf}
	Every smooth morphism is faithfully flat and locally of finite presentation. In particular, every smooth cover is an fppf cover.
\end{lemma}

\begin{lemma}[Fppf morphisms are fpqc morphisms]\label{lem:fppf-fpqc}
	Every faithfully flat morphism that is locally of finite presentation is fpqc. Hence every fppf cover is an fpqc cover.
\end{lemma}

\begin{remark}
	This definition of étale morphism is intended to be the foundational notion used throughout the thesis. It is strong enough to support the discussion of étale covers, étale locality, algebraic spaces, and the étale site, while also fitting naturally into the chain of Grothendieck topologies above.
\end{remark}
