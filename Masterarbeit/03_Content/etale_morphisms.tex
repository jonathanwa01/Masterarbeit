\chapter{\Etale\ Morphisms}\label{ch:etale-morphisms}

\Etale\ morphisms are the central class of morphisms in this thesis: they define the \etale\ topology, algebraic spaces and Deligne Mumford stacks are \etale\ locally schemes, and the Keel Mori theorem is proven by gluing local coarse moduli spaces along \etale\ morphisms.
In this chapter we collect the definitions and properties of \etale\ morphisms of schemes that are used later.

We define \etale\ morphisms using flatness and vanishing
of the K\"ahler differentials, which is the standards approach. This avoids the theory of smooth morphisms and infinitesimal lifting properties, although this seems homotopically to be the more natural definition. We then collect
the geometric properties needed later, especially the
description of diagonals, stability under base change, and
the lifting of geometric points along \etale\ coverings.
Foundational results are stated with references to the
Stacks Project.

Throughout this chapter, all morphisms are morphisms of
schemes unless explicitly stated otherwise.

\section{Definitions}
%==========================================================

\begin{definition}[Finiteness conditions]\label{def:lft-lfp}
	(\cite[][Tags 01T2, 01TQ]{stacks-project})
	A morphism of schemes $f:X\to Y$ is called
	\emph{locally of finite type}, respectively
	\emph{locally of finite presentation}, if for every
	pair of affine open subschemes
	\[
	U=\Specof{B}\subseteq X,
	\qquad
	V=\Specof{A}\subseteq Y
	\]
	with $f(U)\subseteq V$, the induced ring map $A\to B$
	makes $B$ a finitely generated, respectively finitely
	presented, $A$-algebra.
\end{definition}

\begin{remark}
	Every morphism locally of finite presentation is locally
	of finite type. Both conditions are local on the source
	and target in the Zariski topology.
\end{remark}

\subsection{Relative differentials}

We briefly recall the notation for K\"ahler differentials.
Their role in this chapter is to express unramifiedness
algebraically.

\begin{definition}
	Let $A \to B$ be a ring homomorphism and let $M$ be a
	$B$-module. An \emph{$A$-derivation} from $B$ to $M$ is an
	additive map $d:B\to M$ such that
	\[
	d(bb')=bd(b')+b'd(b)
	\qquad\text{and}\qquad
	d(a)=0
	\]
	for all $b,b'\in B$ and all $a\in A$, where $a$ is identified
	with its image in $B$.
	
	The set of $A$-derivations from $B$ to $M$ is denoted by
	$\operatorname{Der}_A(B,M)$. It is a $B$-module under pointwise
	addition and scalar multiplication:
	\[
	(d_1+d_2)(b):=d_1(b)+d_2(b),
	\qquad
	(cd)(b):=cd(b)
	\]
	for $d_1,d_2,D\in\operatorname{Der}_A(B,M)$ and $b,c\in B$.
\end{definition}

\begin{definition}[K\"ahler differentials]
	Let $A\to B$ be a ring map. The $B$-module $\Omega_{B/A}$ of K\"ahler differentials together with the universal $A$-derivation
	\[
	d:B\longrightarrow\Omega_{B/A}
	\]
	is given by the following universal property: For every $B$-module $M$ and any derivation $d': B \to M$ there exists a unique morphism of $B$ modules $\Omega_{B/A} \to M$, such that the following diagram commutes:
	% https://q.uiver.app/#q=WzAsMyxbMCwwLCJCIl0sWzIsMCwiXFxPbWVnYV97Qi9BfSJdLFsxLDEsIk0iXSxbMCwyLCJkJyIsMl0sWzAsMSwiZCJdLFsxLDIsIlxcZXhpc3RzISIsMCx7InN0eWxlIjp7ImJvZHkiOnsibmFtZSI6ImRhc2hlZCJ9fX1dXQ==
	\[\begin{tikzcd}
		B && {\Omega_{B/A}} \\
		& M
		\arrow["d", from=1-1, to=1-3]
		\arrow["{d'}"', from=1-1, to=2-2]
		\arrow["{\exists!}", dashed, from=1-3, to=2-2]
	\end{tikzcd}\]
\end{definition}

\begin{remark}
	In particular, $\Omega_{B/A}$ represents the functor $\operatorname{Der}_A(B,-)$. 
	Equivalently, $\Omega_{B/A}$ is generated as a
	$B$-module by symbols $db$, for $b\in B$, subject
	to the relations
	\[
	d(b+b')=db+db',
	\qquad
	d(bb')=b\,db'+b'\,db,
	\qquad
	da=0.
	\]
	In particular, $\Omega_{B/A}=0$ if and only if every
	$A$-derivation from $B$ into a $B$-module vanishes.
	
\end{remark}

\begin{definition}
	(\cite[][Tag 01UM]{stacks-project})
	Let $f:X\to Y$ be a morphism. The
	sheaf of  differentials
	$\Omega_{X/Y}$ is the quasi-coherent
	$\mathcal{O}_X$-module obtained by gluing the modules
	of K\"ahler differentials on affine charts.
	More precisely, if
	\[
	U=\Specof{B}\subseteq X,
	\qquad
	V=\Specof{A}\subseteq Y,
	\qquad
	f(U)\subseteq V,
	\]
	then
	\[
	\Gamma(U, \Omega_{X/Y}) \cong \Omega_{B/A},
	\]
\end{definition}

\begin{remark}
	Consequently, $\Omega_{X/Y}=0$ can be checked on
	affine open charts. We use relative differentials
	only through this local description and the
	characterisation of unramified morphisms below.
\end{remark}


\subsection{Unramified and \etale\ morphisms}

\begin{definition}[Unramified morphism]
	\label{def:unramified-morphism}
	(\cite[][Section 02G3]{stacks-project})
	A morphism $f:X\to Y$ is \emph{unramified} if it
	is locally of finite type and
	\[
	\Omega_{X/Y}=0.
	\]
\end{definition}

\begin{definition}[\Etale\ morphism]
	\label{def:etale-morphism}
	(\cite[][Section 02GH]{stacks-project})
	A morphism $f:X\to Y$ is \emph{\etale} if it is
	flat, locally of finite presentation, and unramified.
	
	We say that $f$ is \etale\ \emph{at} $x\in X$
	if there exists an open neighbourhood $U\subseteq X$
	of $x$ such that $f|_U:U\to Y$ is \etale.
\end{definition}

\begin{remark}\label{rem:etale-definitions-agree}
	Since local finite presentation implies local finite
	type, the definition can equivalently be written as
	\[
	f\text{ is \etale}
	\quad\Longleftrightarrow\quad
	\begin{cases}
		f\text{ is flat},\\
		f\text{ is locally of finite presentation},\\
		\Omega_{X/Y}=0.
	\end{cases}
	\]
	This agrees with the usual definition by
	\cite[][Tag 02GV]{stacks-project}: a morphism is
	\etale\ if and only if it is flat and G-unramified,
	where G-unramified means unramified and locally of
	finite presentation. We do not otherwise use the
	latter terminology.
\end{remark}

\begin{remark}
	The finiteness conditions are part of the definitions.
	In particular, we do not use the term unramified for
	an arbitrary morphism satisfying
	$\Omega_{X/Y}=0$, nor the term \etale\ for an
	arbitrary flat morphism satisfying this equality.
\end{remark}

\begin{example}[Standard \Etale\ morphisms]
	\label{ex:standard-etale}
	Let $A$ be a ring and $g,h \in A[X]$, such that $g$ is monic and the formal derivative $g'$ is invertible in $B:=A[X]_h/(g)$. Let $f:A \to B$ be the morphism given by $A \to A[X] \to A[X]_h \to B$, then the induced morphism $f^*:\Specof{B} \to \Specof{A}$ is \etale. We call this morphism the standard \etale\ morphism.
\end{example}

\begin{proof}
	Since $g$ is monic, lets say of degree $n$, we have an isomorphism of $A$ modules $C:= A[X]/(g) \cong A \oplus A\bar{X} \oplus \cdots \oplus A\bar{X}^{n-1}$. Thus, $C$ is free, hence flat over $A$. Now the following diagram commutes:
	% https://q.uiver.app/#q=WzAsMyxbMCwwLCJBIl0sWzIsMCwiQyJdLFsxLDEsIkJcXGNvbmcgQ19oIl0sWzAsMl0sWzAsMV0sWzEsMl1d
	\[\begin{tikzcd}
		A && C \\
		& {B\cong C_h}
		\arrow[from=1-1, to=1-3]
		\arrow[from=1-1, to=2-2]
		\arrow[from=1-3, to=2-2]
	\end{tikzcd}\]
	Since localizing is exact, $B$ is flat over $C$ and hence over $A$.
	Next we prove, that $f^*$ is unramified. Since $B =A[X]_h/(g) \cong A[X, T]/(g(X), Th(X)-1)$, $B$ is of finite type over $A$. 
	We are left to show, that 
	\[
	\Omega_{B/A} = 0.
	\]
	Let $x$ be the image of $X$ under the morphism $A[X] \to B$ and let $d: B\to \Omega_{B/A}$ be the universal derivation. We show, that $d = 0$.
	Since $g(x) = 0$ in $B$, we have 
	\[
	0 = d(g(x)) = g'(x) dx.
	\]
	But since $g'(x)$ is a unit, $dx = 0$.  Since $h$ is a unit in $B$, we get 
	\begin{align*}
		0 &= d(1) = d(h(x) h(x)^{-1}) = h(x) d(h(x)^{-1})  + h^{-1}(x)d(h(x)) \\
		&= h(x) d(h(x)^{-1}) + h^{-1}(x) h'(x) dx = h(x) d(h(x)^{-1}) 
	\end{align*}
	and hence, $d(h(x)^{-1}) = 0$. But since $B$ is generated by $x$ and $h(x)^{-1}$, we conclude that $d = 0$, i.e. $\Omega_{B/A} = 0$.
\end{proof}

\begin{remark}
	The condition that $g'$ is invertible is analogous to the
	nonvanishing derivative condition in the implicit function theorem.
	Over $\mathbb{C}$, it ensures that the equation $g(t,X)=0$
	can locally be solved for $X$ as a holomorphic function of $t$.
	In fact, all \etale\ morphisms are Zariski locally of this form, see \cite[Tag 02GT]{stacks-project}.
\end{remark}

\begin{corollary}\label{cor:etale-polynomial-local-ring}
	Let $(A, \mathfrak{m})$ be a local ring with residue
	field $\kappa(\mathfrak{m})$, and let
	$g\in A[X]$ be a monic polynomial of degree $\geq 1$.
	Suppose that its reduction $\bar{g}\in\kappa(\mathfrak{m}[X]$
	is separable. Then
	\[
	\Specof{A[X]/(g)}\longrightarrow\Specof{A}
	\]
	is finite and \etale.
\end{corollary}

\begin{proof}
	Set $B=A[X]/(g)$ and write $x$ for the image of $X$.
	Since $g$ is monic, $B$ is finite free over $A$.
	Moreover,
	\[
	B\otimes_A\kappa(\mathfrak{m}\cong\kappa(\mathfrak{m}[X]/(\bar g).
	\]
	Since $\gcd(\bar g,\bar g')=1$, the image of $g'(x)$
	in $B\otimes_A\kappa(\mathfrak{m}$ is a unit. Thus, the
	multiplication map
	\[
	\mu:B\longrightarrow B,\qquad
	b\longmapsto g'(x)b
	\]
	becomes an isomorphism after tensoring with $\kappa(\mathfrak{m}$.
	
	Let $M=\operatorname{coker}(\mu)$.
	This is a finite $A$-module, since it is a quotient
	of $B$. By right exactness of tensor products,
	\[
	M\otimes_A\kappa(\mathfrak{m}
	\cong\operatorname{coker}(\mu\otimes_A\kappa(\mathfrak{m})
	=0.
	\]
	Nakayama's lemma therefore gives $M=0$.
	Hence $\mu$ is surjective, so there exists $b\in B$
	with $g'(x)b=1$.
	
	Thus $g'(x)$ is a unit, and
	Example~\ref{ex:standard-etale}, with $h=1$,
	shows that the morphism is \etale.
	It is finite because $B$ is finite over $A$.
\end{proof}

\begin{example}[Finite separable field extensions]
	\label{ex:finite-separable-etale}
	Let $L/k$ be a finite separable field extension.
	By the primitive element theorem, there exists
	$\alpha\in L$ such that $L=k(\alpha)$.
	If $g\in k[X]$ denotes the minimal polynomial of
	$\alpha$, then
	\[
	L\cong k[X]/(g).
	\]
	Since $L/k$ is separable, $g$ is separable.
	Applying Corollary~\ref{cor:etale-polynomial-local-ring}
	to the local ring $(k, (0))$ shows that
	\[
	\Specof{L}\longrightarrow\Specof{k}
	\]
	is finite and \etale.
\end{example}


\section{Diagonals and open immersions}


The following geometric characterisation of unramifiedness
is especially important in the construction of coarse
moduli spaces.

\begin{lemma}[Diagonal criterion for unramifiedness]
	\label{lem:unramified-diagonal}
	(\cite[][Tag 02GE]{stacks-project})
	Let $f:X\to Y$ be a morphism.
	\begin{enumerate}
		\item If $f$ is unramified, its diagonal
		\[
		\Delta_{X/Y}:X\longrightarrow X\times_Y X
		\]
		is an open immersion.
		\item Conversely, if $f$ is locally of finite
		type and $\Delta_{X/Y}$ is an open immersion,
		then $f$ is unramified.
	\end{enumerate}
\end{lemma}

\begin{corollary}[Diagonal of an \etale\ morphism]
	\label{cor:etale-separated-diagonal}
	Let $f:X\to Y$ be \etale. Then
	$\Delta_{X/Y}$ is an open immersion.
	If $f$ is also separated, its diagonal is an open
	and closed immersion. In that case there is a
	decomposition
	\[
	X\times_Y X
	=
	\Delta_{X/Y}(X)
	\sqcup
	\bigl((X\times_Y X)\setminus\Delta_{X/Y}(X)\bigr)
	\]
	into open and closed subschemes.
\end{corollary}

\begin{proof}
	An \etale\ morphism is unramified by definition,
	so its diagonal is an open immersion by
	Lemma~\ref{lem:unramified-diagonal}.
	If $f$ is separated, its diagonal is also a closed
	immersion by definition. Its image and the
	complementary open subscheme are therefore both
	open and closed.
\end{proof}

\begin{remark}
	The corresponding statement for algebraic spaces
	is \cite[][Tag 05W1]{stacks-project}.
	After base change, it also applies to representable
	\etale\ separated morphisms of Deligne--Mumford
	stacks. This is the input used to prove that the
	stabilizer-preserving locus is open in the
	Keel--Mori construction.
\end{remark}

\begin{lemma}[Unramified monomorphisms]
	\label{lem:unramified-monomorphism}
	(\cite[][Tag 05VH]{stacks-project})
	Let $f:X\to Y$ be unramified.
	Then $f$ is a monomorphism if and only if it is
	universally injective.
\end{lemma}

\begin{proposition}[Open immersion criterion]
	\label{prop:etale-open-immersion}
	(\cite[][Tag 02LC]{stacks-project})
	A morphism $f:X\to Y$ is an open immersion if and
	only if it is \etale\ and universally injective.
	In particular, an \etale\ monomorphism is an
	open immersion.
\end{proposition}


\section{Locality}


\begin{lemma}
	\label{lem:etale-permanence}
	The following statements hold.
	\begin{enumerate}
		\item Compositions of \etale\ morphisms are
		\etale\
		(\cite[][Tag 02GN]{stacks-project}).
		\item Base changes of \etale\ morphisms are
		\etale. Thus, if $f:X\to Y$ is \etale\ and
		$Y'\to Y$ is any morphism, then
		\[
		X\times_Y Y'\longrightarrow Y'
		\]
		is \etale\
		(\cite[][Tag 02GO]{stacks-project}).
		\item Every open immersion is \etale\
		(\cite[][Tag 02GP]{stacks-project}).
		\item If $X\to S$ and $Y\to S$ are \etale,
		then every $S$-morphism $X\to Y$ is \etale\
		(\cite[][Tag 02GW]{stacks-project}).
	\end{enumerate}
\end{lemma}

\begin{remark}
	The first three properties allow us to form
	\etale\ coverings, pull them back, refine them
	by open coverings, and compose them.
	The fourth property applies to comparison maps
	between schemes \etale\ over a common base.
\end{remark}

\begin{lemma}[\Etale\ is local on the source]
	\label{lem:etale-local-on-source}
	(\cite[][Tag 02GJ]{stacks-project})
	Let $f:X\to Y$ be a morphism and let
	$X=\bigcup_i U_i$ be an open covering.
	Then $f$ is \etale\ if and only if each
	restriction $f|_{U_i}:U_i\to Y$ is \etale.
	
	Moreover, if $f$ is \etale\ and $U\subseteq X$,
	$V\subseteq Y$ are open subschemes with
	$f(U)\subseteq V$, then the induced morphism
	$U\to V$ is \etale.
\end{lemma}

\begin{corollary}[Disjoint unions]
	\label{cor:etale-disjoint-union}
	Let $\{X_i\to Y\}_{i\in I}$ be a family of
	morphisms. Then
	\[
	\coprod_{i\in I}X_i\longrightarrow Y
	\]
	is \etale\ if and only if every $X_i\to Y$
	is \etale.
\end{corollary}

\begin{proof}
	The summands $X_i$ form an open covering of
	$\coprod_i X_i$, so the assertion follows from
	Lemma~\ref{lem:etale-local-on-source}.
\end{proof}

\begin{lemma}[\Etale\ is local on the target]
	\label{lem:etale-local-on-target}
	(\cite[][Tag 02VN]{stacks-project})
	Let $f:X\to Y$ be a morphism and let
	$\{Y_i\to Y\}_{i\in I}$ be an fpqc covering.
	Then $f$ is \etale\ if and only if every
	base change
	\[
	X\times_Y Y_i\longrightarrow Y_i
	\]
	is \etale.
\end{lemma}

\begin{remark}
	In particular, being \etale\ may be checked after
	a Zariski, \etale, or fppf covering of the target.
	Together with base-change stability, this is what
	makes the definition of a representable \etale\
	morphism of sheaves independent of the chosen
	covering of the target.
\end{remark}

\begin{lemma}[Further consequences]
	\label{lem:etale-consequences}
	Let $f:X\to Y$ be \etale. Then:
	\begin{enumerate}
		\item $f$ is locally of finite presentation;
		\item $f$ is flat;
		\item $f$ is unramified;
		\item $f$ is locally quasi-finite;
		\item $f$ is universally open.
	\end{enumerate}
\end{lemma}

\begin{proof}
	The first three assertions are part of the
	definition. An \etale\ morphism is locally
	quasi-finite by
	\cite[][Tag 03WS]{stacks-project} and open by
	\cite[][Tag 03WT]{stacks-project}.
	Since every base change of $f$ is again \etale,
	every base change is open.
\end{proof}

\begin{remark}
	Universal openness is useful when refining
	coverings. In particular, the images of affine
	open subschemes in an \etale\ covering form
	an open covering of the target.
\end{remark}

\section{\Etale\ morphisms over fields}
%==========================================================

Over a field, \etale\ schemes have a particularly simple
description.

\begin{lemma}[\Etale\ schemes over a field]
	\label{lem:etale-over-field}
	(\cite[][Tag 02GL]{stacks-project})
	Let $k$ be a field and let $X$ be a $k$-scheme.
	Then $X\to\Specof{k}$ is \etale\ if and only if
	\[
	X\cong\coprod_{i\in I}\Specof{k_i},
	\]
	where each $k_i/k$ is a finite separable field
	extension.
\end{lemma}

\begin{remark}
	The index set $I$ need not be finite.
	Thus an \etale\ scheme over a field is a disjoint
	union of finite separable points, but need not
	be a finite scheme over that field.
\end{remark}

\begin{corollary}[Geometric fibres]
	\label{cor:etale-geometric-fibres}
	Let $f:X\to Y$ be \etale.
	For every $y\in Y$, the fibre
	\[
	X_y=X\times_Y\Specof{\kappa(y)}
	\]
	is a disjoint union of spectra of finite
	separable extensions of $\kappa(y)$.
	For every geometric point
	$\bar{y}:\Specof{k}\to Y$, the geometric fibre
	is a disjoint union of copies of $\Specof{k}$.
\end{corollary}

\begin{proof}
	Both fibre morphisms are \etale\ by base-change
	stability. Apply Lemma~\ref{lem:etale-over-field}.
	If $k$ is algebraically closed, every finite
	separable extension of $k$ equals $k$.
\end{proof}

\begin{corollary}\label{cor:etale-over-alg-closed}
	Let $k$ be algebraically closed and let
	$X\to\Specof{k}$ be \etale.
	Then
	\[
	X\cong\coprod_{i\in I}\Specof{k}.
	\]
	If $X$ is nonempty, the morphism
	$X\to\Specof{k}$ admits a section.
\end{corollary}

\begin{proof}
	The decomposition follows from
	Lemma~\ref{lem:etale-over-field}.
	If $X$ is nonempty, choose one component.
	Its inclusion into $X$ is a section.
\end{proof}

\begin{proposition}[Fibrewise characterisation]
	\label{prop:etale-characterisations}
	(\cite[][Tag 02GM]{stacks-project})
	Let $f:X\to Y$ be locally of finite presentation.
	The following are equivalent:
	\begin{enumerate}
		\item $f$ is \etale;
		\item $f$ is flat and unramified;
		\item $f$ is flat and, for every $y\in Y$,
		the fibre $X_y$ is a disjoint union of
		spectra of finite separable extensions of
		$\kappa(y)$.
	\end{enumerate}
\end{proposition}

\begin{proof}
	The equivalence of (1) and (2) is our definition
	under the stated finite-presentation hypothesis.
	The implication (1)$\Rightarrow$(3) follows from
	Corollary~\ref{cor:etale-geometric-fibres}.
	The converse is
	\cite[][Tag 02GM]{stacks-project}.
\end{proof}

\begin{remark}
	This characterisation gives a geometric picture:
	an \etale\ morphism is a flat, locally finitely
	presented family whose geometric fibres are
	discrete reduced schemes.
	It need not be finite or surjective.
\end{remark}
