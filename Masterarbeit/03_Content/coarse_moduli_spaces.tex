\chapter{Coarse Moduli Spaces}

In this chapter we introduce coarse moduli spaces, which provide a way to pass
from DM stacks to algebraic spaces by identifying objects up to
isomorphism. This construction plays a central role in the Keel-Mori theorem.

\section{Motivation}

Let $\mathcal{X}$ be a DM stack. For a scheme $T$, the groupoid
$\mathcal{X}(T)$ consists of objects over $T$ together with their isomorphisms.
A natural goal is to construct a geometric object whose points correspond to isomorphism classes of objects of $\mathcal{X}$.

However, in general $\mathcal{X}$ is not representable by a scheme or algebraic space due to the presence of automorphisms. The notion of a coarse moduli space provides a best possible approximation.

\section{Definition of Coarse Moduli Space}

\begin{definition}[Coarse moduli space]
	\label{def:coarse-moduli-space}
	Let $\mathcal{X}$ be a DM stack. A morphism
	\[
	\pi : \mathcal{X} \to X
	\]
	to an algebraic space $X$ is called a coarse moduli space if:
	
	\begin{enumerate}
		\item \textbf{Universality:}  
		For every algebraic space $Y$ and every morphism
		\[
		f : \mathcal{X} \to Y,
		\]
		there exists a unique morphism
		\[
		g : X \to Y
		\]
		such that $f = g \circ \pi$.
		
		\item \textbf{Geometric points:}  
		For every algebraically closed field $k$, the induced map
		\[
		|\mathcal{X}(k)| \to X(k)
		\]
		identifies $X(k)$ with the set of isomorphism classes of objects of
		$\mathcal{X}(k)$.
	\end{enumerate}
\end{definition}


\section{Basic Properties}

\begin{lemma}[Affine quotient by a finite group]\label{lem:affine-quotient-finite-group}
	(\cite[Theorem 4.3.6. (1)]{alper2026stacks})
	Let $G$ be a finite group acting on an affine scheme $\Spec(A)$. Then the canonical morphism
	\[
	\Spec(A) \longrightarrow \Spec(A^G)
	\]
	induces a morphism of stacks
	\[
	[\Spec(A)/G] \longrightarrow \Spec(A^G),
	\]
	and this morphism is a coarse moduli space.
\end{lemma}

\begin{proof}
	We check the two defining properties of a coarse moduli space.
	
	\emph{(i) Geometric points.}
	The induced map on geometric points identifies the set of $G$-orbits in $\Spec(A)(\bar{k})$ with the set of points of $\Spec(A^G)(\bar{k})$.
	The morphism $\Spec(A) \to \Spec(A^G)$ is finite and surjective, since $A$ is integral over $A^G$ for a finite group action. In particular, two
	geometric points have the same image in $\Spec(A^G)$ if and only if they lie in the same $G$-orbit.
	
	\emph{(2) Universality.}
	Let $Y$ be an algebraic space and let
	\[
	f : [\Spec(A)/G] \longrightarrow Y
	\]
	be a morphism. Since $f$ is $G$-invariant, its restriction to $\Spec(A)$
	corresponds locally on an étale cover of $Y$ to a $G$-invariant morphism of
	affine schemes.
	Since $Y$ is a sheaf on the \etale\ site, these local factorizations glue
	to a unique morphism $\Spec(A^G)\to Y$ making the following diagram commute:
	% https://q.uiver.app/#q=WzAsMyxbMCwwLCJbXFxTcGVjb2Z7QX0vR10iXSxbMSwwLCJZIl0sWzEsMSwiXFxTcGVjb2Z7QV5HfSJdLFswLDEsImYiXSxbMSwyXSxbMCwyXV0=
	\[\begin{tikzcd}
		{[\Specof{A}/G]} & Y \\
		& {\Specof{A^G}}
		\arrow["f", from=1-1, to=1-2]
		\arrow[from=1-1, to=2-2]
		\arrow[from=1-2, to=2-2]
	\end{tikzcd}\]
	
	Thus $\pi : [\Spec(A)/G]\to \Spec(A^G)$ satisfies the universal property of a
	coarse moduli space.
\end{proof}


\begin{lemma}\label{lem:coarse-moduli-etale-local}\label{lem:coarse-moduli-space-local-on-target}
	(\cite[Lemma 5.2.10]{alper2026stacks})
	Let $\mathcal{X}$ be a stack and $\pi : \mathcal{X} \to X$ be a morphism to an algebraic space. Suppose
	that there is an étale covering $\{f_i : X_i \to X\}$ such that
	\[
	\pi_i : \mathcal{X} \times_X X_i \longrightarrow X_i
	\]
	is a coarse moduli space for each $i$. Then $\pi : \mathcal{X} \to X$ is a
	coarse moduli space.
\end{lemma}

\begin{proof}
	We verify the two axioms of Definition~\ref{def:coarse-moduli-space}.
	Let $\mathcal{X}_i := \mathcal{X} \times_X X_i$.
	
	\textbf{Geometric points}
	Let $k$ be an algebraically closed field and let
	$x : \Spec(k) \to X$ be a geometric point. Since $\{f_i : X_i \to X\}$
	is an étale covering, there exists an  $i$ and a lift
	$\tilde{x} : \Spec(k) \to X_i$ with $f_i \circ \tilde{x} = x$, by lemma \ref{lem:geometric-points-etale-cover}.
	
	Since $\pi_i$ is a coarse moduli space, the induced map
	\[
	|(\mathcal{X}_i)(k)| \longrightarrow X_i(k)
	\]
	is a bijection onto isomorphism classes. The fiber of $\pi$ over $x$
	is canonically identified with the fiber of $\pi_i$ over $\tilde{x}$,
	via the base-change isomorphism
	\[
	\mathcal{X} \times_X \Spec(k)
	\;\cong\;
	(\mathcal{X} \times_X X_i) \times_{X_i} \Spec(k) = \mathcal{X}_i\times_{X_i}\Specof{k}.
	\]
	Hence the bijection for $\pi_i$ implies the corresponding bijection for
	$\pi$.
	
	\medskip
	\textbf{Universality}
	Let $Y$ be an algebraic space and $f : \mathcal{X} \to Y$ a morphism.
	For each $i$, consider the composite
	\[
	f_i: \mathcal{X}_i \xrightarrow{\;p_i\;} \mathcal{X}
	\xrightarrow{\;f\;} Y.
	\]
	Since $\pi_i$ is a coarse moduli space, there exists a unique morphism
	$g_i : X_i \to Y$ such that the following commutes:

	g_i|_{X_{ij}} = g_j|_{X_{ij}}.

	coincide.

\begin{lemma}\label{lem:invariants-flat-base-change}
	(\cite[Lemma 5.2.7]{alper2026stacks})
	Let $G$ be a finite abstract group acting on an affine scheme $\Specof{A}$ and let $A^G \to B$ be a flat ring morphism. Then the action of $G$ on $\Specof{A}$ induces an action on $\Spec(B\times_{A^G} A)$ and $(B\times_{A^G} A)^G \cong B$.
\end{lemma}

\begin{proof}
	By definition, $A^G$ is given by the following equalizer
	% https://q.uiver.app/#q=WzAsNCxbMCwwLCIwIl0sWzEsMCwiQV5HIl0sWzIsMCwiQSJdLFszLDAsIlxccHJvZF97Z1xcaW4gR31BIl0sWzIsMywicF8xIiwwLHsib2Zmc2V0IjotMX1dLFsyLDMsInBfMiIsMix7Im9mZnNldCI6MX1dLFsxLDJdLFswLDFdXQ==
	\[\begin{tikzcd}
		0 & {A^G} & A & {\prod_{g\in G}A,}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-2, to=1-3]
		\arrow["{p_1}", shift left, from=1-3, to=1-4]
		\arrow["{p_2}"', shift right, from=1-3, to=1-4]
	\end{tikzcd}\]
	where $p_1(a) = (a)_{g\in G}$ and $p_2(a) = (ga)_{g\in G}$.
	Since $A^G \to B$ is flat,
	% https://q.uiver.app/#q=WzAsNCxbMCwwLCIwIl0sWzEsMCwiQiJdLFsyLDAsIkFcXG90aW1lc197QV5HfUIiXSxbMywwLCJcXHByb2Rfe2dcXGluIEd9QVxcb3RpbWVzX3tBXkd9QiJdLFsyLDMsInBfMSIsMCx7Im9mZnNldCI6LTF9XSxbMiwzLCJwXzIiLDIseyJvZmZzZXQiOjF9XSxbMSwyXSxbMCwxXV0=

		0 & B & {A\otimes_{A^G}B} & {\prod_{g\in G}A\otimes_{A^G}B}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-2, to=1-3]
		\arrow["{p_1}", shift left, from=1-3, to=1-4]

\begin{lemma}[Pullback of a quotient presentation]
	\label{lem:pullback-quotient-presentation}
	Let $G$ be a finite abstract group acting on an algebraic
	space $U$, and let
	\[
	f:\mathcal{Y}\longrightarrow[U/G]
	\]
	be a morphism of DM stacks, which is representable by algebraic soaces. Set
	\[
	V:=\mathcal{Y}\times_{[U/G]}U.
	\]
	Then $V\to\mathcal{Y}$ is a $G$-torsor. Its $G$-action
	makes the projection $V\to U$ equivariant, and the induced
	morphism
	\[
	[V/G]\longrightarrow\mathcal{Y}
	\]
	is an equivalence.
	
	If $f$ is affine and $U$ is an affine scheme, then $V$ is an affine scheme.
\end{lemma}

\begin{proof}
	By definition, the following is $2$-cartesian:
	% https://q.uiver.app/#q=WzAsNCxbMCwwLCJWIl0sWzAsMSwiXFxtYXRoY2Fse1l9Il0sWzEsMSwiW1UvR10iXSxbMSwwLCJVIl0sWzMsMl0sWzEsMiwiZiIsMl0sWzAsMV0sWzAsM11d
	\[\begin{tikzcd}
		V & U \\
		{\mathcal{Y}} & {[U/G]}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-1, to=2-1]
		\arrow[from=1-2, to=2-2]
		\arrow["f"', from=2-1, to=2-2]
	\end{tikzcd}\]
	The standard morphism $U\to[U/G]$ is a $G$-torsor.
	Its pullback $V\to\mathcal{Y}$ is therefore a $G$-torsor,
	with an equivariant projection $V\to U$.
	
	We show that the induced morphism
	$[V/G]\to\mathcal{Y}$ is an equivalence.
	This may be checked after an fppf covering of
	$\mathcal{Y}$, which trivializes the torsor $V\to\mathcal{Y}$.
	Over such a covering $T\to\mathcal{Y}$, being trivial yields an identification
	\[
	[(G\times T)/G]\longrightarrow T,
	\]
	which is an equivalence. Hence $[V/G]\simeq\mathcal{Y}$.
	
	Finally, $V\to U$ is the base change of $f$. If $f$ is affine and $U$ is affine,
	then $V$ is an affine scheme.
\end{proof}









\begin{lemma}\label{lem:coarse-moduli-flat-base-change}
	(\cite[Theorem 4.3.6. (1)]{alper2023stacks})
	Let $G$ be a finite abstract group acting on an affine scheme $\Specof{A}$ and let $\pi: [\Specof{A} / G] \to \Specof{A^G}$ be the coarse moduli space by Lemma \ref{lem:coarse-moduli-etale-local}. 
	Then the base change of $\pi$ along any flat morphism $X' \to \Specof{A^G}$ of algebraic spaces is a corase moduli space:
	% https://q.uiver.app/#q=WzAsNCxbMSwwLCJbXFxTcGVjb2Z7QX0gL0ddIl0sWzEsMSwiXFxTcGVjb2Z7QV5HfSJdLFswLDEsIlgnIl0sWzAsMCwiWCdcXHRpbWVzX3tcXFNwZWNvZntBXkd9fVtcXFNwZWNvZntBfSAvR10iXSxbMiwxXSxbMCwxLCJcXHBpIl0sWzMsMF0sWzMsMl1d
	\[\begin{tikzcd}
		{X'\times_{\Specof{A^G}}[\Specof{A} /G]} & {[\Specof{A} /G]} \\
		{X'} & {\Specof{A^G}}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-1, to=2-1]
		\arrow["\pi", from=1-2, to=2-2]
		\arrow[from=2-1, to=2-2]
	\end{tikzcd}\]
\end{lemma}

\begin{proof}
	By Lemma \ref{lem:coarse-moduli-etale-local} being a coarse moduli space is \etale\ local on the target. If we cover $X'$ by affine schemes, we may, without loss of generality, assume $X' = \Specof{B}$ to be affine. The flat morphism $X' \to \Specof{A^G}$ is induced by the flat morphism $A^G \to B$. 
	Set
	\[
	C := A\otimes_{A^G}B.
	\]
	Since $G$ acts trivial on $A^G$, the $G$ action on $A$ induces a $G$ action on $C$ by 
	\[
	g(a\otimes b) := g(a) \otimes b.
	\]
	
	Let $\mathcal{Y} = \Specof{B} \times_{\Specof{A^G}}[\Specof{A} / G]$ and consider the following diagram:
	
	% https://q.uiver.app/#q=WzAsNixbMCwyLCJcXFNwZWNvZntCfSJdLFsxLDIsIlxcU3BlY29me0FeR30iXSxbMSwxLCJbXFxTcGVjb2Z7QX0vR10iXSxbMCwxLCJcXG1hdGhjYWx7WX0iXSxbMSwwLCJcXFNwZWNvZntBfSJdLFswLDAsIlxcU3BlY29me0N9Il0sWzIsMSwiXFxwaSJdLFswLDFdLFszLDBdLFszLDJdLFszLDEsIiIsMix7InN0eWxlIjp7Im5hbWUiOiJjb3JuZXItaW52ZXJzZSJ9fV0sWzQsMl0sWzUsNF0sWzUsM11d
	\[\begin{tikzcd}
		{\Specof{C}} & {\Specof{A}} \\
		{\mathcal{Y}} & {[\Specof{A}/G]} \\
		{\Specof{B}} & {\Specof{A^G}}
		\arrow[from=1-1, to=1-2]
		\arrow[from=1-1, to=2-1]
		\arrow[from=1-2, to=2-2]
		\arrow[from=2-1, to=2-2]
		\arrow[from=2-1, to=3-1]
		\arrow["\ulcorner"{anchor=center, pos=0.125}, draw=none, from=2-1, to=3-2]
		\arrow["\pi", from=2-2, to=3-2]
		\arrow[from=3-1, to=3-2]
	\end{tikzcd}\]
	
	Since the lower square and the outer square is $2$-cartesian, so is the upper.
	Hence, Lemma \ref{lem:pullback-quotient-presentation} gives
	an equivalence
	\[
	\mathcal{Y} \cong [\Specof{C}/G],
	\]
	under which the base change of $\pi$ identifies with the
	morphism
	\[
	[\Specof{C}/G]\longrightarrow\Specof{B}
	\]
	induced by the $G$-invariant ring homomorphism
	$B\to C$, $b\mapsto 1\otimes b$.
	
	Since $B$ is flat over $A^G$,
	Lemma~\ref{lem:invariants-flat-base-change} shows that the
	canonical homomorphism
	\[
	B \xlongrightarrow{\cong} C^G
	\]
	is an isomorphism. Hence the base-changed morphism identifies
	with
	\[
	[\Specof{C}/G]\longrightarrow\Specof{C^G},
	\]
	which is a coarse moduli space by
	Lemma~\ref{lem:affine-quotient-finite-group}.
\end{proof}


\section{Examples}

\begin{example}[Trivial case]
	If $\mathcal{X} = X$ is an algebraic space, then the identity
	\[
	X \to X
	\]
	is a oarse moduli space.
\end{example}


\begin{example}[Quotients by $\mathbb{Z}/2$]\label{ex:Z2-coarse}
		(\cite[][Example 5.2.4.]{alper2026stacks})
		Assume $\operatorname{char}k\neq2$ and let $G=\mathbb{Z}/2=\{\pm1\}$ act on affine space $\mathbb{A}^n=\operatorname{Spec}k[x_1,\dots,x_n]$ by $-1\cdot x_i=-x_i$. In the affine case the coarse moduli space of $[\operatorname{Spec}A/G]$ is $\operatorname{Spec}A^G$, so we compute the invariant ring.
		
		\medskip\noindent\textbf{(a) The case $n=1$.}
		Let $G$ act on $\mathbb{A}^1=\operatorname{Spec}k[x]$ via $-1\cdot x=-x$. Then a polynomial is invariant if and only if only even powers of $x$ occur, so
		\[
		k[x]^G=k[x^2].
		\]
		The quotient map is
		\[
		\mathbb{A}^1=\operatorname{Spec}k[x]\longrightarrow\operatorname{Spec}k[x^2]=\mathbb{A}^1,\qquad t\mapsto t^2,
		\]
		which is a geometric quotient. The induced morphism
		\[
		[\mathbb{A}^1/G]\longrightarrow\operatorname{Spec}k[x^2]
		\]
		is the coarse moduli space of $[\mathbb{A}^1/G]$. The action is free away from the origin, and the origin has stabilizer $G=\mathbb{Z}/2$.
		
		\medskip\noindent\textbf{(b) The case $n=2$.}
		Let $G$ act on $\mathbb{A}^2=\operatorname{Spec}k[x,y]$ via $-1\cdot(x,y)=(-x,-y)$. A monomial $x^ay^b$ is invariant if and only if $a+b$ is even, so
		\[
		k[x,y]^G=k[x^2,xy,y^2].
		\]
		The geometric quotient is
		\[
		\mathbb{A}^2\longrightarrow\mathbb{A}^2/G=\operatorname{Spec}k[x^2,xy,y^2].
		\]
		Setting $A=x^2$, $B=xy$, $C=y^2$ identifies the invariant ring with
		\[
		k[A,B,C]/(AC-B^2),
		\]
		since the three generators satisfy $AC=x^2y^2=B^2$ and this is the only relation. Hence $\mathbb{A}^2/G$ is the quadric cone $\{AC=B^2\}\subset\mathbb{A}^3$. As in (a), the morphism $[\mathbb{A}^2/G]\to\mathbb{A}^2/G$ is the coarse moduli space. The stack $[\mathbb{A}^2/G]$ is a DM stack with the origin being the only stacky point, while its coarse space is singular at the image of the origin, see Figure \ref{fig:Z2-quotient}.
	\end{example}
