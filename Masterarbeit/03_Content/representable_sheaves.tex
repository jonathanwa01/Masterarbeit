\section{Representable presheaves are fppf sheaves}\label{sec:representable-fppf}


We now deduce from fppf descent for quasi-coherent sheaves that every scheme defines a sheaf for the fppf topology, and hence for all topologies of Definition~\ref{def:standard-sites}. The argument has three steps:
\begin{enumerate}
	\item The fully faithful part of Theorem~\ref{thm:fppf-descent-qcoh}, applied to the structure sheaf, shows that $T\mapsto\Gamma(T,\mathcal{O}_T)$ satisfies fppf descent.
	\item Morphisms into an affine scheme $\Specof{A}$ are ring maps out of $A$, and the representable preshef $\Hom_{\Ring}(A,-)$ preserves limits. This gives the case of affine targets without any computation.
	\item For an arbitrary target we descend open subsets along the covering and reduce to (2).
\end{enumerate}

\begin{corollary}\label{cor:structure-sheaf-fppf}
	Let $p:U\to T$ be an fppf covering of schemes. Then
	\[
	\begin{tikzcd}
		\Gamma(T,\mathcal{O}_T) \arrow[r, "p^*"] & \Gamma(U,\mathcal{O}_U) \arrow[r, shift left, "\mathrm{pr}_0^*"] \arrow[r, shift right, "\mathrm{pr}_1^*"'] & \Gamma(U\times_TU,\mathcal{O}_{U\times_TU})
	\end{tikzcd}
	\]
	is an equalizer diagram of rings.
\end{corollary}

\begin{proof}
	By Theorem~\ref{thm:fppf-descent-qcoh}, the functor $\QCoh(T)\to\Desc(U/T)$, $\mathcal{F}\mapsto(p^*\mathcal{F},\mathrm{can}_\mathcal{F})$ (Definition~\ref{con:pullback-to-descent}), is fully faithful. We apply this to $\mathcal{F}=\mathcal{G}=\mathcal{O}_T$.
	
	On the left, evaluation at $1_T\in \Gamma(T, \mathcal{O}_T)$ identifies $\Hom_{\QCoh(T)}(\mathcal{O}_T,\mathcal{O}_T)$ with $\Gamma(T,\mathcal{O}_T)$.
	On the right, $p^*\mathcal{O}_T=\mathcal{O}_U$ canonically, and under the identifications $\mathrm{pr}_0^*\mathcal{O}_U=\mathcal{O}_{U\times_TU}=\mathrm{pr}_1^*\mathcal{O}_U$ the canonical descent datum $\mathrm{can}_{\mathcal{O}_T}$ is the identity. A morphism of descent data $(\mathcal{O}_U,\mathrm{id})\to(\mathcal{O}_U,\mathrm{id})$ is therefore an endomorphism of $\mathcal{O}_U$, i.e.\ a section $s\in\Gamma(U,\mathcal{O}_U)$, such that $\mathrm{pr}_0^*(s)=\mathrm{pr}_1^*(s)$.
	Under these identifications, the map on morphism sets is $p^*:\Gamma(T,\mathcal{O}_T)\to\Gamma(U,\mathcal{O}_U)$.
	
	By full faithfulness, $p^*$ is a bijection onto the equalizer of $\mathrm{pr}_0^*$ and $\mathrm{pr}_1^*$. Since $p^*$ is a ring map and equalizers of rings are computed on underlying sets, the diagram is an equalizer of rings.
\end{proof}


\begin{lemma}\label{lem:representable-affine-target}
	Let $X=\Specof{A}$ be an affine scheme and let $p:U\to T$ be an fppf covering. Then $h_X$ satisfies descent along $p$.
\end{lemma}

\begin{proof}
	By the adjunction $\Gamma \dashv \Spec$ for every scheme $T$, the is a natural isomorphism
	\begin{align*}
			\Hom_{\Sch}(T,\Specof{A}) & \to\Hom_{\Ring}(A,\Gamma(T,\mathcal{O}_T)) \\
				 f &\mapsto f^\sharp,
	\end{align*} see (\cite[][Tag 01I1]{stacks-project}). Hence there is a natural isomorphism of presheaves
	\[
	h_X\cong\Hom_{\Ring}(A,-)\circ\Gamma(-, \mathcal{O}_{(-)}).
	\]
	The representable functor $\Hom_{\Ring}(A,-):\Ring\to\Set$ preserves all limits, in particular equalizers and by the adjunction $\Gamma \dashv \Spec$, $\Gamma$ preserves all limites. Hence, applying $\Hom_{\Ring}(A, - )$ to the equalizer of Corollary \ref{cor:structure-sheaf-fppf} shows that
	\[
	h_X(T)\longrightarrow h_X(U)\rightrightarrows h_X(U\times_TU)
	\]
	is an equalizer, which means that $h_X$ satisfies descent along $p$ by Proposition \ref{prop:cech-descent-equalizer}.
\end{proof}

For a general target, the only additional input is that open subsets descend along fppf coverings.


\begin{theorem}\label{thm:representable-fppf-sheaf}
	(\cite[][Tag 023Q]{stacks-project}, \cite[][Theorem~2.55]{vistoli2007notes})
	Let $X$ be a scheme. Then the representable presheaf $h_X=\Hom_\Sch(-,X)$ satisfies the sheaf condition for every fppf covering. In particular, $h_X$ is a sheaf on $\Sch_\tau$ for $\tau\in\{\mathrm{Zariski}, \text{\etale}, \mathrm{fppf}\}$.
\end{theorem}

\begin{proof}
	Morphisms of schemes glue uniquely along open coverings.
	By Corollary~\ref{cor:sheaf-via-cech}, it therefore suffices
	to prove that, for every fppf covering $p:U\to T$, the diagram
	\[
	\Hom(T,X)\longrightarrow \Hom(U,X)
	\rightrightarrows \Hom(U\times_T U,X)
	\]
	is an equalizer. Write
	$\mathrm{pr}_1,\mathrm{pr}_2:U\times_T U\rightrightarrows U$
	for the projections, and choose an affine open covering
	$X=\bigcup_{i\in I}X_i$.
	
	\emph{Uniqueness.}
	Let $g,g':T\to X$ satisfy $g\circ p=g'\circ p$.
	Since $p$ is surjective, the equality
	\[
	p^{-1}\bigl(g^{-1}(X_i)\bigr)
	=
	p^{-1}\bigl(g'^{-1}(X_i)\bigr)
	\]
	implies
	\[
	T_i:=g^{-1}(X_i)=g'^{-1}(X_i).
	\]
	Hence, the restrictions $g|_{T_i},g'|_{T_i}:T_i\to X_i$ agree
	after pullback along the fppf covering $p^{-1}(T_i)\to T_i$. Since $X_i$ are assumed to be affine, $g|_{T_i}$ and $g'|_{T_i}$ are equal by
	Lemma~\ref{lem:representable-affine-target}.
	As the $T_i$ cover $T$, we obtain $g=g'$.
	
	\emph{Existence.}
	Let $f:U\to X$ be a morphism such that
	\begin{align}\label{eq:compability-condition}
		f\circ\mathrm{pr}_1=f\circ\mathrm{pr}_2.
	\end{align}
	Write $\iota_i:X_i\hookrightarrow X$ for the inclusions.
	Set $U_i:=f^{-1}(X_i)$. Then
	\[
	\mathrm{pr}_1^{-1}(U_i)
	=
	(f\circ\mathrm{pr}_1)^{-1}(X_i)
	=
	(f\circ\mathrm{pr}_2)^{-1}(X_i)
	=
	\mathrm{pr}_2^{-1}(U_i).
	\]
	
	Set $T_i = p(U_i)$. Since $p$ is flat and locally of finite presentation,
	it is open by \cite[][Tag 01UA]{stacks-project}. Thus each $T_i$ is open in $T$.
	We claim that $p^{-1}(T_i)=U_i$.
	The inclusion $U_i\subseteq p^{-1}(T_i)$ is obvious.
	By construction, $U_i\to T_i$ is surjective.
	Its base change
	\[
	q:p^{-1}(T_i)\times_{T_i}U_i
	\longrightarrow p^{-1}(T_i)
	\]
	is therefore surjective
	(\cite[][Tag 01S1]{stacks-project}).
	Under the canonical identification
	\[
	p^{-1}(T_i)\times_{T_i}U_i
	\cong U\times_T U_i,
	\]
	the morphism $q$ is induced by the first projection.
	On $U\times_T U_i$, the compatibility condition gives
	\[
	f\circ\mathrm{pr}_0=f\circ\mathrm{pr}_1.
	\]
	Since the second projection factors through $U_i$,
	the right-hand side factors through $X_i$.
	Consequently, the first projection factors through
	$f^{-1}(X_i)=U_i$.
	Thus $q$ factors through the open immersion
	\[
	U_i\hookrightarrow p^{-1}(T_i).
	\]
	Surjectivity of $q$ implies that this open immersion
	is surjective, hence an isomorphism.
	Therefore $p^{-1}(T_i)=U_i$.
	
	
	
	
	Hence, the following diagram is cartesian and $p_i:U_i\to T_i$ is an fppf covering:

	\iota_i\circ f_i\circ \mathrm{pr}_1 = f_{\mid U_i} \circ \mathrm{pr}_1
	= f_{\mid U_i} \circ \mathrm{pr}_2
	= \iota_i\circ f_i\circ \mathrm{pr}_2.
	\]
	As $\iota_i$ is a monomorphism, this implies
	\[
	f_i\circ \mathrm{pr}_1=f_i\circ \mathrm{pr}_2.
	\]
	Since $X_i$ is affine,
	Lemma~\ref{lem:representable-affine-target} yields a unique
	morphism $g_i:T_i\to X_i$ such that
	\[
	g_i\circ p_i=f_i.

	We now verify that the morphisms
	\[
	\widetilde g_i:=\iota_i\circ g_i:T_i\to X
	\]
	agree on overlaps to glue them to the morphism $g: T\to X$ fitting into diagram \ref{diag:representable-sheaf-zariski-descent} by Zariski descent.
	For $i,j\in I$, let
	\[
	T_{ij}:=T_i\cap T_j,
	\qquad
	U_{ij}:=p^{-1}(T_{ij})=U_i\cap U_j.
	\]
	The morphism $p_{ij}:U_{ij}\to T_{ij}$ induced by $p$
	is again an fppf covering.
	By construction,
	\[
	\begin{aligned}
		\bigl(\widetilde g_i|_{T_{ij}}\bigr)\circ p_{ij}
		&=f|_{U_{ij}},\\
		\bigl(\widetilde g_j|_{T_{ij}}\bigr)\circ p_{ij}
		&=f|_{U_{ij}}.
	\end{aligned}
	\]
	By uniqueness, applied to
	$p_{ij}:U_{ij}\to T_{ij}$, implies
	\[
	\widetilde g_i|_{T_{ij}}
	=
	\widetilde g_j|_{T_{ij}}.
	\]
	Hence the $\widetilde g_i$ glue to a morphism $g:T\to X$ by Zariski descent, since $\{T_i\subset T\}_{i\in I}$ is an open covering.
	
	Finally, on every $U_i$ we have
	\[
	(g\circ p)|_{U_i}
	=
	\widetilde g_i\circ p_i
	=
	\iota_i\circ f_i
	=
	f|_{U_i}.
	\]
	Since the $U_i$ cover $U$, it follows that $g\circ p=f$.
\end{proof}


\begin{corollary}[Relative version]\label{cor:representable-relative}
	Let $S$ be a scheme,\\
	$\tau\in\{\mathrm{Zariski}, \text{\etale}, \mathrm{fppf}\}$, and $X$ a scheme over $S$. Then $h_{X/S}=\Hom_S(-,X)$ is a sheaf on $(\SchOver{S})_\tau$.
\end{corollary}

\begin{proof}
	Let $\{U_i\to T\}$ be a $\tau$-covering in $\SchOver{S}$ and let $f_i:U_i\to X$ be $S$-morphisms that agree on all $U_i\times_TU_j$. By Theorem~\ref{thm:representable-fppf-sheaf} they glue uniquely to a morphism $f:T\to X$. Composing with the structure morphism $X\to S$, the two morphisms $T\to S$ given by $X\to S$ composed with $f$ and by the structure morphism of $T$ agree on every $U_i$, so they are equal by injectivity in Theorem~\ref{thm:representable-fppf-sheaf} applied to $h_S$. Hence $f$ is an $S$-morphism.
\end{proof}

\begin{remark}
	Theorem~\ref{thm:representable-fppf-sheaf} says that the sites $\Sch_\tau$ and $(\SchOver{S})_\tau$ are subcanonical. By the Yoneda lemma, $X\mapsto h_X$ is a fully faithful functor $\SchOver{S}\to\Sh{(\SchOver{S})_{\mathrm{fppf}}}$. This is the starting point of the chapter on algebraic spaces, where algebraic spaces are defined as fppf sheaves that are \etale\ locally representable.
\end{remark}
