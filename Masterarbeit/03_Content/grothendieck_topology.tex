\chapter{Grothendieck Topologies}\label{chap:grothendieck-topologies}

In this chapter, we introduce Grothendieck topologies on categories as a generalization of classical topologies on spaces. Given a topological space $X$, its open subsets form a category $\mathrm{Open}(X)$, whose objects are the open sets $U\subset X$ and whose morphisms are inclusions $V\hookrightarrow U$. Sheaves on $X$ can be viewed as contravariant functors on this category that satisfy a gluing condition with respect to open coverings. In this way, the usual notion of ``covering an open subset $U$ by smaller opens $\{U_i\subset U\}$'' is encoded purely in terms of the category $\mathrm{Open}(X)$.

Grothendieck's idea is to abstract this picture: instead of restricting to open subsets and inclusion maps, one chooses a category $\CatC$ and specifies, for each object $U\in \CatC$, which families of morphisms $\{U_i \to U\}$ should be considered as ``coverings''. These covering families are required to satisfy axioms, which mimic those for open covers, and the resulting structure is called a Grothendieck topology on $\CatC$. In this way one can treat, for example, \etale\ or fppf coverings of schemes within the same framework as open coverings.

Conceptually, this follows the Yoneda philosophy: an object can be studied through the morphisms into it. Sheaves are then the presheaves which satisfy a descent condition with respect to the covering families. We formulate this descent condition via the \v{C}ech nerve of a covering. This viewpoint comes from simplicial homotopy theory: a covering $U\to X$ gives a simplicial object $\check{C}(U/X)_\bullet$ augmented over $X$, a presheaf $F$ turns it into a cosimplicial object $F(\check{C}(U/X)_\bullet)$, and the sheaf condition says that $F(X)$ is the totalization of this cosimplicial object. For presheaves of sets the totalization is an ordinary limit, which we compute in Lemma~\ref{lem:tot-cosimplicial-set}. For presheaves of groupoids it is the $2$-categorical totalization studied in Chapter~\ref{chap:descent-theory}.

\section{Sites}
%==========================================================

We work with the formulation of Grothendieck topologies in terms of covering families, which is the one used in \cite{stacks-project} and \cite{vistoli2007notes}.

\begin{definition}[Site]\label{def:site}
	(\cite[][Definition~2.24]{vistoli2007notes}, \cite[][Tag 00VH]{stacks-project})
	Let $\CatC$ be a category. A \emph{Grothendieck topology} on $\CatC$ assigns to each object $X\in\CatC$ a collection of families of morphisms $\{U_i\to X\}_{i\in I}$, called \emph{coverings} of $X$, such that:
	\begin{enumerate}
		\item If $V\to X$ is an isomorphism, then $\{V\to X\}$ is a covering.
		\item (Base change) If $\{U_i\to X\}_{i\in I}$ is a covering and $Y\to X$ is any morphism, then the fiber products $U_i\times_X Y$ exist and $\{U_i\times_X Y\to Y\}_{i\in I}$ is a covering of $Y$.
		\item (Composition) If $\{U_i\to X\}_{i\in I}$ is a covering and for each $i$ the family $\{V_{ij}\to U_i\}_{j\in J_i}$ is a covering, then $\{V_{ij}\to U_i\to X\}_{i\in I,\,j\in J_i}$ is a covering of $X$.
	\end{enumerate}
	A category together with a Grothendieck topology is called a \emph{site}.
\end{definition}

\begin{remark}[Sieves]\label{rem:sieves}
	In \cite{MacLaneMoerdijk1992}, a Grothendieck topology is defined in terms of sieves. A \emph{sieve} on $X$ is a subfunctor $S\subset h_X=\Hom_\CatC(-,X)$ (\cite[][p.~38]{MacLaneMoerdijk1992}). A Grothendieck topology in this sense assigns to every $X$ a collection $J(X)$ of sieves on $X$, called \emph{covering sieves}, such that (\cite[][Sec.~III.2, Def.~1]{MacLaneMoerdijk1992}):
	\begin{enumerate}
		\item the maximal sieve $h_X$ belongs to $J(X)$;
		\item if $S\in J(X)$ and $f:Y\to X$ is a morphism, then $f^*S=\{g:Z\to Y\mid f\circ g\in S\}$ belongs to $J(Y)$;
		\item if $S\in J(X)$ and $R$ is any sieve on $X$ such that $f^*R\in J(Y)$ for all $f:Y\to X$ in $S$, then $R\in J(X)$.
	\end{enumerate}
	A Grothendieck topology in the sense of Definition~\ref{def:site} is called a \emph{basis} for a Grothendieck topology in \cite[][Sec.~III.2, Def.~2]{MacLaneMoerdijk1992} and a \emph{pretopology} in SGA~4. It generates a Grothendieck topology in the sense of sieves: a sieve is covering if and only if it contains the sieve generated by some covering family. The two notions have the same sheaves (\cite[][Remark~2.25 and Proposition~2.42]{vistoli2007notes}). Since all our arguments use covering families, we only work with Definition~\ref{def:site}.
\end{remark}

\begin{remark}[Set-theoretic issues]\label{rem:set-theory-sites}
	The category $\Sch$ of all schemes is not small. To obtain honest sites one replaces $\Sch$ by a suitably large but small full subcategory, see \cite[][Tag 021A]{stacks-project} for the \etale\ case. None of the arguments in this thesis depend on this choice, so we suppress it from the notation.
\end{remark}

\begin{definition}[Standard topologies on schemes]\label{def:standard-sites}
	(\cite[][Tags 020N, 0215, 021M]{stacks-project})
	Let $\tau \in \{\mathrm{Zariski}, \text{\etale}, \mathrm{smooth}, \mathrm{fppf}\}$. A family of morphisms of schemes $\{f_i:U_i \to X\}_{i\in I}$ is a \emph{$\tau$-covering} if
	\begin{enumerate}
		\item each $f_i$ is an open immersion (Zariski), \etale\ (\etale), smooth (smooth), or flat and locally of finite presentation (fppf), respectively, and
		\item the family is jointly surjective, i.e.\ $X=\bigcup_{i\in I} f_i(U_i)$.
	\end{enumerate}
	We write $\Sch_\tau$ for the resulting site, and $(\SchOver{S})_\tau$ for the induced site on schemes over a fixed scheme $S$.
\end{definition}

\begin{definition}[fpqc covering]\label{def:fpqc-covering}
	(\cite[][Tag 022B]{stacks-project})
	A family of morphisms of schemes $\{f_i:U_i\to X\}_{i\in I}$ is an \emph{fpqc covering} if each $f_i$ is flat and for every affine open $V\subset X$ there are finitely many indices $i_1,\dots,i_n\in I$ and affine opens $W_k\subset U_{i_k}$ such that $V=\bigcup_{k=1}^n f_{i_k}(W_k)$.
	In particular, an fpqc covering is jointly surjective.
\end{definition}

\begin{proposition}[Hierarchy of topologies]\label{prop:gross-topology-chain}
	Each of the four classes of coverings in Definition~\ref{def:standard-sites}, and the class of fpqc coverings, satisfies the axioms of Definition~\ref{def:site}. Moreover,
	\[
	\mathrm{Zariski} \subset \text{\etale} \subset \mathrm{smooth} \subset \mathrm{fppf} \subset \mathrm{fpqc}
	\]
	in the sense that every Zariski covering is an \etale\ covering, every \etale\ covering is a smooth covering, and so on.
\end{proposition}

\begin{proof}
	For the axioms see \cite[][Chapter 020K]{stacks-project}; for fpqc coverings this is \cite[][Tag 022D]{stacks-project}. The first three inclusions follow from Lemma~\ref{lem:etale-permanence}(3) and Lemma~\ref{lem:etale-consequences}, since smooth morphisms are flat and locally of finite presentation (\cite[][Tag 021N]{stacks-project}). The last inclusion is \cite[][Tag 022C]{stacks-project}.
\end{proof}

\begin{remark}[The fpqc topology]\label{rem:fpqc-not-a-site}
	The fpqc coverings form a Grothendieck topology in the sense of Definition~\ref{def:site}, but this topology is set-theoretically too large: for a nonzero ring $R$, there is no set of fpqc coverings of $\Specof{R}$ such that every fpqc covering is refined by one of them (\cite[][Tag 0BBK]{stacks-project}). As a consequence, the usual construction of sheafification breaks down; the Stacks Project refers to an example of Waterhouse of a presheaf without fpqc sheafification (\cite[][Tag 022A]{stacks-project}). For this reason we never sheafify in the fpqc topology. We only use the sheaf \emph{condition} for fpqc coverings (Definition~\ref{def:sheaf-on-cite}), which makes sense for every presheaf on $\Sch$.
\end{remark}

\begin{lemma}[Refinement by affine schemes]\label{prop:affine-refinement-standard-topologies}
	Let $\tau\in\{\mathrm{Zariski}, \text{\etale}, \mathrm{smooth}, \mathrm{fppf}\}$ and let $\{f_i:U_i\to X\}_{i\in I}$ be a $\tau$-covering. Choose affine open covers $U_i=\bigcup_{j\in J_i}V_{ij}$. Then $\{V_{ij}\to X\}_{i\in I, j\in J_i}$ is a $\tau$-covering by affine schemes. If $X$ is quasi-compact, finitely many of the $V_{ij}$ suffice.
\end{lemma}

\begin{proof}
	Open immersions \etale, smooth, and flat and locally of finite presentation, and each of these classes is stable under composition, so every $V_{ij}\to U_i\to X$ is of type $\tau$. The family is jointly surjective because the $V_{ij}$ cover $U_i$. Every flat morphism locally of finite presentation is open (\cite[][Tag 01UA]{stacks-project}), so the images $f_i(V_{ij})$ form an open cover of $X$. If $X$ is quasi-compact, finitely many of them cover $X$.
\end{proof}

\begin{remark}
	A presheaf satisfying the sheaf condition for a refinement need not formally satisfy it for the original covering. We therefore never use Lemma~\ref{prop:affine-refinement-standard-topologies} to ``replace'' a covering in the verification of a sheaf condition, only to produce coverings with good properties.
\end{remark}

\section{Simplicial objects and the \v{C}ech nerve}
%==========================================================

\begin{definition}[Simplicial and cosimplicial objects]\label{def:simplicial-objects}
	Let $\Delta$ denote the simplex category, whose objects are the finite ordered sets $[n]=\{0<1<\dots<n\}$ for $n\geq 0$ and whose morphisms are non-decreasing maps. For $0\le i\le n$ we write $\delta^i:[n-1]\to[n]$ for the injection omitting $i$ and $\sigma^i:[n+1]\to[n]$ for the surjection repeating $i$.
	A \emph{simplicial object} in a category $\CatC$ is a functor $X_\bullet:\Delta^{\op}\to\CatC$, and a \emph{cosimplicial object} is a functor $X^\bullet:\Delta\to\CatC$. We write $d_i:=X(\delta^i)$, $s_i:=X(\sigma^i)$ for the face and degeneracy maps of a simplicial object, and $d^i:=X(\delta^i)$, $s^i:=X(\sigma^i)$ for the coface and codegeneracy maps of a cosimplicial object.
\end{definition}

\begin{definition}[\v{C}ech nerve]\label{def:cech-nerve}
	Let $\CatC$ be a category with fiber products and let $p:U\to X$ be a morphism. The \emph{\v{C}ech nerve} of $p$ is the simplicial object $\check{C}(U/X)_\bullet$ in $\CatC_{/X}$ with
	\[
	\check{C}(U/X)_n = U^{\times_X(n+1)} = \underbrace{U\times_X\cdots\times_X U}_{n+1\text{ factors}},
	\]
	and, for $\theta:[m]\to[n]$ in $\Delta$, with $\theta^*:\check{C}(U/X)_n\to\check{C}(U/X)_m$ the unique morphism over $X$ whose $k$-th component is the projection $\mathrm{pr}_{\theta(k)}$. In particular, $d_i$ forgets the $i$-th factor and $s_i$ repeats the $i$-th factor. Together with $p$ it is an augmented simplicial object $\check{C}(U/X)_\bullet\to X$.
	We draw it as
	\[
	\begin{tikzcd}
		\cdots\ U\times_X U\times_X U
		\arrow[r, shift left=2] \arrow[r] \arrow[r, shift right=2]
		& U\times_X U
		\arrow[r, shift left=1] \arrow[r, shift right=1]
		& U \arrow[r, "p"] & X .
	\end{tikzcd}
	\]
\end{definition}

\begin{remark}[Homotopical meaning]\label{rem:cech-homotopy}
	The \v{C}ech nerve is the $0$-coskeleton of $U$ in $\CatC_{/X}$: a morphism of simplicial objects $Y_\bullet\to\check{C}(U/X)_\bullet$ over $X$ is the same as a morphism $Y_0\to U$ over $X$. If $\CatC=\Set$ and $p$ is surjective, then $\check{C}(U/X)_\bullet$ is the nerve of the equivalence relation $U\times_X U\rightrightarrows U$, and every fiber of $\check{C}(U/X)_\bullet\to X$ is the nerve of a contractible groupoid (the indiscrete groupoid on a nonempty set). Hence $\check{C}(U/X)_\bullet\to X$ is a weak equivalence of simplicial sets, where $X$ is regarded as a constant simplicial set.
	More generally, if $p$ admits a section $s$, the augmented \v{C}ech nerve has an \emph{extra degeneracy} induced by $s$, which gives a contraction (for the dual statement about cosimplicial objects see \cite[][Tag 019Z]{stacks-project}). Faithfully flat descent can be read as the algebraic shadow of this fact: after a faithfully flat base change along $p$ itself, $p$ acquires a section (the diagonal), and the corresponding Amitsur complex becomes contractible (\cite[][Tags 023L, 023M]{stacks-project}).
\end{remark}

\begin{lemma}[\v{C}ech nerve of a disjoint union]\label{lem:cech-nerve-coproduct}
	Let $\{U_i\to X\}_{i\in I}$ be a family of morphisms of schemes and $U:=\coprod_{i\in I}U_i$. Then for every $n\ge 0$ there is a canonical isomorphism
	\[
	\check{C}(U/X)_n\cong\coprod_{(i_0,\dots,i_n)\in I^{n+1}}U_{i_0}\times_X\cdots\times_X U_{i_n},
	\]
	under which the face and degeneracy maps are given summand-wise by the projections and diagonals.
\end{lemma}

\begin{proof}
	It suffices to show that $(\coprod_i U_i)\times_X Y\cong\coprod_i(U_i\times_X Y)$ for every $X$-scheme $Y$; the claim then follows by induction on $n$. The $U_i$ form a cover of $U$ by pairwise disjoint open subschemes. For the projection $\mathrm{pr}_1:U\times_XY\to U$, the open subscheme $\mathrm{pr}_1^{-1}(U_i)$ is canonically $U_i\times_XY$, and these open subschemes are pairwise disjoint and cover $U\times_XY$. This gives the isomorphism.
\end{proof}

\begin{lemma}[Totalization of a cosimplicial set]\label{lem:tot-cosimplicial-set}
	Let $X^\bullet:\Delta\to\Set$ be a cosimplicial set. Then the map
	\[
	\lim_{[n]\in\Delta}X^n\longrightarrow\Eq\left(\begin{tikzcd}X^0\arrow[r, shift left, "d^0"]\arrow[r, shift right, "d^1"'] & X^1\end{tikzcd}\right),\qquad (x_n)_{n\ge 0}\longmapsto x_0,
	\]
	is a bijection. In other words, the limit of $X^\bullet$ only depends on its restriction to $\Delta_{\le 1}$.
\end{lemma}

\begin{proof}
	An element of the limit is a family $(x_n)_n$, $x_n\in X^n$, with $X(\theta)(x_m)=x_n$ for all $\theta:[m]\to[n]$. For $0\le k\le n$ let $v_k:[0]\to[n]$ be the map with image $k$. Then $x_n=X(v_0)(x_0)$, so the family is determined by $x_0$, and the map is injective. Moreover $x_0$ lies in the equalizer, because $d^0(x_0)=x_1=d^1(x_0)$.
	
	Conversely, let $x_0\in X^0$ with $d^0x_0=d^1x_0$. For $0\le k\le n$, let $e:[1]\to[n]$ be the map with $e(0)=0$, $e(1)=k$. Then $v_0=e\circ\delta^1$ and $v_k=e\circ\delta^0$, hence
	\[
	X(v_k)(x_0)=X(e)\,d^0(x_0)=X(e)\,d^1(x_0)=X(v_0)(x_0).
	\]
	So $X(v)(x_0)$ does not depend on the choice of $v:[0]\to[n]$. Put $x_n:=X(v_0)(x_0)$. For $\theta:[m]\to[n]$ we get $X(\theta)(x_m)=X(\theta\circ v_0)(x_0)=x_n$, since $\theta\circ v_0:[0]\to[n]$. Thus $(x_n)_n$ lies in the limit and maps to $x_0$.
\end{proof}

\begin{remark}
	Lemma~\ref{lem:tot-cosimplicial-set} is the $1$-categorical reason why the sheaf condition only involves $U$ and $U\times_X U$: the higher stages of the \v{C}ech nerve impose no further conditions on sets. For a cosimplicial groupoid, the $2$-categorical totalization depends on $\Delta_{\le 2}$, which produces the cocycle condition on $U\times_X U\times_X U$ (Chapter~\ref{chap:descent-theory}); for higher groupoids the whole of $\Delta$ is needed.
\end{remark}

\section{Sheaves and \v{C}ech descent}
%==========================================================

\begin{definition}[Sheaf on a site]\label{def:sheaf-on-cite}
	(\cite[][Tag 00VM]{stacks-project})
	Let $(\CatC,\tau)$ be a site and $F:\CatC^{\op}\to\Set$ a presheaf. Let $\mathcal{U}=\{f_i:U_i\to X\}_{i\in I}$ be a family of morphisms such that the fiber products $U_i\times_X U_j$ exist. We say that $F$ \emph{satisfies the sheaf condition for $\mathcal{U}$} if
	\begin{equation}\label{eq:sheaf-condition-equalizer}
		\begin{tikzcd}
			{F(X)} & {\prod_iF(U_i)} & {\prod_{i,j}F(U_i\times_X U_j)}
			\arrow["{(f_i^*)_i}", from=1-1, to=1-2]
			\arrow["{\mathrm{pr}_0^*}", shift left, from=1-2, to=1-3]
			\arrow["{\mathrm{pr}_1^*}"', shift right, from=1-2, to=1-3]
		\end{tikzcd}
	\end{equation}
	is an equalizer diagram, where $\mathrm{pr}_0:U_i\times_XU_j\to U_i$ and $\mathrm{pr}_1:U_i\times_XU_j\to U_j$ are the projections. We say $F$ is \emph{separated for $\mathcal{U}$} if the first map is injective. $F$ is a \emph{sheaf} if it satisfies the sheaf condition for every covering of the site. We write $\Sh{\CatC,\tau}\subset\PshC$ for the full subcategory of sheaves.
\end{definition}

\begin{definition}[\v{C}ech descent]\label{def:cech-descent}
	Let $\CatC$ be a category with fiber products, $F:\CatC^{\op}\to\Set$ a presheaf and $p:U\to X$ a morphism. Applying $F$ to the augmented \v{C}ech nerve gives an augmented cosimplicial set $F(X)\to F(\check{C}(U/X)_\bullet)$. We say that $F$ \emph{satisfies descent along $p$} if the induced map
	\[
	F(X)\longrightarrow\lim_{[n]\in\Delta}F\bigl(\check{C}(U/X)_n\bigr)
	\]
	is a bijection.
\end{definition}

\begin{proposition}\label{prop:cech-descent-equalizer}
	In the situation of Definition~\ref{def:cech-descent}, $F$ satisfies descent along $p$ if and only if $F$ satisfies the sheaf condition for the single-morphism family $\{p:U\to X\}$, i.e.\ if and only if
	\[
	F(X)\xrightarrow{\ p^*\ }F(U)\ \substack{\xrightarrow{\ \mathrm{pr}_0^*\ }\\[-0.2em]\xrightarrow[\ \mathrm{pr}_1^*\ ]{}}\ F(U\times_X U)
	\]
	is an equalizer.
\end{proposition}

\begin{proof}
	The coface maps $d^0,d^1:F(U)\to F(U\times_XU)$ are $F(d_0)=\mathrm{pr}_1^*$ and $F(d_1)=\mathrm{pr}_0^*$, so the claim follows from Lemma~\ref{lem:tot-cosimplicial-set}. The comparison maps are compatible because both are induced by $p^*$.
\end{proof}

To pass from covering families to single morphisms we need that $F$ converts disjoint unions into products. This holds for every Zariski sheaf.

\begin{lemma}[Zariski sheaves and disjoint unions]\label{lem:sheaf-preserves-coproducts}
	Let $F$ be a presheaf on $\Sch$ satisfying the sheaf condition for all Zariski coverings. Then $F(\varnothing)$ is a point, and for every family of schemes $\{U_i\}_{i\in I}$ the restriction maps induce a bijection
	\[
	F\Big(\coprod_{i\in I}U_i\Big)\xrightarrow{\ \sim\ }\prod_{i\in I}F(U_i).
	\]
\end{lemma}

\begin{proof}
	The empty family is a Zariski covering of $\varnothing$, and the sheaf condition for it says that $F(\varnothing)$ is the equalizer of two maps between empty products, i.e.\ a point. Next, $\{U_j\hookrightarrow\coprod_iU_i\}_{j\in I}$ is a Zariski covering with $U_j\times_{\coprod_iU_i}U_k=U_j\cap U_k$, which is $U_j$ for $j=k$ and $\varnothing$ otherwise. On the factors with $j\ne k$ both maps in~\eqref{eq:sheaf-condition-equalizer} land in the point $F(\varnothing)$, and on the factors with $j=k$ both are the identity of $F(U_j)$. Hence the equalizer is all of $\prod_jF(U_j)$, and the sheaf condition gives the claim.
\end{proof}

\begin{proposition}[Sheaf condition as \v{C}ech descent]\label{lem:sheaf-single-cover}
	Let $F$ be a presheaf on $\Sch$ satisfying the sheaf condition for all Zariski coverings. Let $\mathcal{U}=\{U_i\to X\}_{i\in I}$ be a family of morphisms of schemes and set $U:=\coprod_iU_i$ with the induced morphism $p:U\to X$. Then $F$ satisfies the sheaf condition for $\mathcal{U}$ if and only if $F$ satisfies descent along $p$.
\end{proposition}

\begin{proof}
	By Lemma~\ref{lem:cech-nerve-coproduct}, $U\times_XU\cong\coprod_{i,j}U_i\times_XU_j$, and the projections $U\times_XU\rightrightarrows U$ restrict on the summand $U_i\times_XU_j$ to the projections onto $U_i$ and $U_j$. Applying Lemma~\ref{lem:sheaf-preserves-coproducts} to $U$ and to $U\times_XU$ gives an isomorphism between the diagram of Proposition~\ref{prop:cech-descent-equalizer} and the diagram~\eqref{eq:sheaf-condition-equalizer}, which is the identity on $F(X)$. Being an equalizer is invariant under isomorphism of diagrams, so Proposition~\ref{prop:cech-descent-equalizer} gives the claim.
\end{proof}

\begin{lemma}\label{lem:coverings-vs-single-morphism}
	Let $\tau\in\{\text{\etale}, \mathrm{smooth}, \mathrm{fppf}, \mathrm{fpqc}\}$. A family $\{U_i\to X\}_{i\in I}$ is a $\tau$-covering if and only if $\{\coprod_iU_i\to X\}$ is a $\tau$-covering.
\end{lemma}

\begin{proof}
	For $\tau\ne\mathrm{fpqc}$, being \etale, smooth, resp.\ flat and locally of finite presentation is local on the source (\cite[][Tag 02GJ]{stacks-project} and its analogues), and joint surjectivity is the same condition for both. For fpqc, flatness is local on the source. An affine open $W\subset U_{i}$ is also an affine open of $\coprod_iU_i$, so the finiteness condition of Definition~\ref{def:fpqc-covering} for the family implies the one for $\coprod_iU_i\to X$. Conversely, an affine open $W\subset\coprod_iU_i$ is quasi-compact, hence meets only finitely many $U_i$, and $W\cap U_i$ is open and closed in $W$, hence affine. This gives the converse.
\end{proof}

\begin{corollary}[Sheaves via \v{C}ech descent]\label{cor:sheaf-via-cech}
	Let $\tau\in\{\text{\etale}, \mathrm{smooth}, \mathrm{fppf}, \mathrm{fpqc}\}$ and let $F$ be a presheaf on $\Sch$. Then $F$ satisfies the sheaf condition for all $\tau$-coverings if and only if
	\begin{enumerate}
		\item $F$ satisfies the sheaf condition for all Zariski coverings, and
		\item $F$ satisfies descent along every morphism $p:U\to X$ such that $\{p\}$ is a $\tau$-covering.
	\end{enumerate}
\end{corollary}

\begin{proof}
	Zariski coverings are $\tau$-coverings by Proposition~\ref{prop:gross-topology-chain}, and by Proposition~\ref{prop:cech-descent-equalizer} condition (2) is the sheaf condition for the $\tau$-covering $\{p\}$, so both conditions are necessary. Conversely, given a $\tau$-covering $\{U_i\to X\}$, the morphism $\coprod_iU_i\to X$ forms a $\tau$-covering by Lemma~\ref{lem:coverings-vs-single-morphism}, and Proposition~\ref{lem:sheaf-single-cover} applies.
\end{proof}

\begin{remark}
	Corollary~\ref{cor:sheaf-via-cech} is the precise form of the slogan $F(X)\cong\Tot F(\check{C}(U/X)_\bullet)$ from the introduction to Chapter~\ref{chap:descent-theory}: up to the Zariski-local condition (1), being a sheaf means that $F$ sends augmented \v{C}ech nerves of single coverings to limit diagrams. Working with a single morphism $U\to X$ instead of a family removes the indices $i,j,k$ from all subsequent constructions, which becomes essential for presheaves valued in groupoids.
\end{remark}

\begin{lemma}[Limits of sheaves]\label{cor:limit-sheaf}
	Let $(\CatC,\tau)$ be a site. Limits in $\PshC$ are computed objectwise, and a limit of sheaves computed in $\PshC$ is again a sheaf. In particular $\Sh{\CatC,\tau}$ is complete and the inclusion $\Sh{\CatC,\tau}\hookrightarrow\PshC$ preserves limits.
\end{lemma}

\begin{proof}
	The first statement is standard, see \cite[][Sec.~V.3]{MacLaneCFWM}. Let $F=\lim_kF_k$ be a limit of sheaves and $\{U_i\to X\}$ a covering. The diagram~\eqref{eq:sheaf-condition-equalizer} for $F$ is the limit over $k$ of the corresponding diagrams for the $F_k$, which are equalizers. Since limits commute with limits (\cite[][Sec.~IX.2]{MacLaneCFWM}), it is an equalizer as well.
\end{proof}

\begin{lemma}[Sheafification]\label{lem:sheafification}
	(\cite[][p.~134]{MacLaneMoerdijk1992})
	Let $(\CatC,\tau)$ be a small site. Then the inclusion $\iota:\Sh{\CatC,\tau}\hookrightarrow\PshC$ admits a left adjoint $(-)^\sharp$, called \emph{sheafification}.
\end{lemma}

\begin{remark}[Representable presheaves]\label{rem:representable-forward}
	A site is \emph{subcanonical} if every representable presheaf $h_X=\Hom_\CatC(-,X)$ is a sheaf. The most important example for us is that all sites $\Sch_\tau$ of Definition~\ref{def:standard-sites} are subcanonical. In fact $h_X$ satisfies the sheaf condition even for all fpqc coverings. We prove this in Theorem~\ref{thm:representable-fpqc-sheaf} as a consequence of fpqc descent for quasi-coherent sheaves (Theorem~\ref{thm:fpqc-descent-qcoh}). By the Yoneda lemma, the functor $X\mapsto h_X$ is then a fully faithful embedding of $\Sch$ into $\Sh{\Sch_\tau}$, and we will identify a scheme with the sheaf it represents.
\end{remark}
