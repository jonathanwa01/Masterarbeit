\chapter{Grothendieck Topology}

In this chapter, we introduce Grothendieck topologies on categories as a generalization of classical topologies on spaces. Given a topological space 
$X$, its open subsets form a category $Open(X)$, whose objects are the open sets $U\subset X$ and whose morphisms are inclusions $V\hookrightarrow U$. Sheaves on $X$ can be viewed equivalently as certain contravariant functors on this category that satisfy a gluing condition with respect to open coverings. In this way, the usual notion of “covering an open subset $U$ by smaller opens $\{U_i\subset U\}$" is encoded purely in terms of the category $Open(X)$.

Grothendieck’s idea is to abstract this picture: Instead of restricting to open subsets and inclusion maps, one chooses a category $\CatC$ and specifies, for each object $U\in \CatC$, which families of morphisms  $\{U_i \to U\}$ should be considered  as "coverings”. These covering families are required to satisfy axioms modeled on the behavior of open covers in classical topology, and the resulting structure is called a Grothendieck topology on $\CatC$.

By this approach, one can treat, for example, coverings of schemes (such as \etale\ or fppf covers) and coverings of rings (via their associated affine schemes) within the same framework.

Conceptually, this follows the general Yoneda philosophy: A space (or more generally, an object) can be studied and reconstructed from its functor points, i.e. the morphisms into it. Grothendieck topologies implement this idea by replacing open subsets of a space with families of morphisms in an arbitrary category that behave like covers, and then defining sheaves as functors on $\CatC$ that satisfy a suitable descent condition with respect to these covering families.

\begin{definition}[Sieve]
	(\cite[][p.~38]{MacLaneMoerdijk1992})
	Let $X\in \CatC$ be an object of a small category. A sieve on $X$ is a subfunctor of the representable presheaf $\Hom_\CatC(-, X)$. We will write $h_X$ for the Yoneda embedding $\Hom_\CatC(-, X)$.
\end{definition}

\begin{definition}[Grothendieck Topology]
	(\cite[(Definition 1), p. ~110]{MacLaneMoerdijk1992})
	A Grothendieck topology $J$ on a small categrory $\CatC$ is a function that assigns to each object $X\in C$ a collection of sieves $J(X)$ called covering sieves, such that the following axioms are satisfied:
	\begin{enumerate}
		\item For each object $X\in \CatC$, $\Hom_\CatC(-, X)$ is a sieve.
		\item For each morphism $f:X\to Y$ in $\CatC$ and each sieve $S\in J(Y)$ the pullback $f^*(S)\in J(X)$ is a sieve on $X$.
		\item For each covering sieve $S\in J(Y)$ and each sieve $R$ on $Y$, such that $f^*(R)\in J(X)$ for all $f:X\to Y\in S$, then $R\in J(Y)$
	\end{enumerate}
\end{definition}

\begin{remark}
	For a sieve $S$ on $Y\in\CatC$ and any $f:X\to Y$ in $\CatC$ we have $f^*S = \{g:Z\to X\mid f\circ g\in S\}\subset \Hom_\CatC(-, X)$
\end{remark}

\begin{definition}[Site]
	(\cite[][p. ~110]{MacLaneMoerdijk1992})
	For a small category $\CatC$ and a Grothendieck topology $J$ we will call the pair $(\CatC, J)$  a site on $\CatC$.
\end{definition}

\begin{definition}[Matching family]
	(\cite[][p.~121]{MacLaneMoerdijk1992})
	Let $(\CatC, J)$ be a site, $P:\CatCop\to \Set$ a presheaf on $\CatC$ and $S\in J(C)$ a covering sieve. A matching familiy for $S$ of elements of $P$ is natural transformation $\eta:S\Rightarrow P$.
	
	This assigns to each $f:D\to C\in S(D)$ an element $x_f\in P(D)$, such that $P(g)(x_f) = x_{f\circ g}$ for all $g:E\to D\in \CatC$. Notice that $f\circ g\in S(E)$, since $S$ is a sieve.
	
	An amalgation of the matching family $\eta$ is an element $x\in P(C)$, such that $P(f)(x) = x_f$ for all $f\in S$
\end{definition}

\begin{lemma}\label{lem:sheaf-preserves-coproducts}
	Let $F$ be a sheaf on a cite $(\CatC,J)$. Then
	\[
	F\big(\coprod_i U_i\big) \cong \prod_i F(U_i).
	\]
\end{lemma}

\begin{lemma}[Sheafication]\label{lem:sheafification}
	(\cite[p. 134]{MacLaneMoerdijk1992})
	Let $(\CatC, J)$ be a site, then the fully faithful inclusion $\iota: \Sh{\CatC,J} \hookrightarrow \PshC$ admits a left adjoint $\adj{a}{\iota}$.
	We write $\eta^2: F \to F^\sharp$ for the unit evaluated at a presheaf $F$.
\end{lemma}

\begin{corollary}\label{cor:limit-sheaf}
	By \cite[]{MacLaneCFWM}, the category $\Sh{\CatC,J}$ is complete. In particular the limit of sheaves is again a sheaf and can be calculated in the  category $\PshC$ of presheaves.
\end{corollary}
