\chapter{Descent Theory}

Descent theory provides a unified framework for reconstructing global objects from local data along coverings. In algebraic geometry, this theory appears at several levels of generality, forming a hierarchy that can be organized by truncation level in higher category theory. At the most algebraic level, faithfully flat descent for commutative rings is governed by the exactness of the Amitsur complex. Descent for modules, quasi-coherent sheaves, and affine schemes reduces to this algebraic statement. Algebraic spaces satisfy descent as sheaves of sets, while stacks satisfy descent as sheaves of groupoids, which amounts to truncating the Čech descent diagram at the level of triple overlaps. More generally, $\infty$-stacks satisfy descent with respect to the entire Čech nerve. Thus, descent theory in algebraic geometry forms a hierarchy ranging from faithfully flat descent for rings to $\infty$-categorical descent, and the theory of stacks can be understood both as a higher-categorical extension of classical faithfully flat descent and as a 2-truncated form of $\infty$-categorical descent.

The guiding principle for the entire chapter should be:

Descent for a geometric object $F$ means that for a covering 
$U\to X$, the natural map $F(X) \to \Totof{F(U_\bullet)}$ is an equivalence, where $U_\bullet$ is the Čech nerve of the covering.



\begin{definition}[Čech nerve]
	Let $f : U \to X$ be a morphism in a category with fiber products.
	The \emph{Čech nerve} of $f$ is the simplicial object
	\[
	U_\bullet : \Delta^{op} \to \mathcal{C}
	\]
	given by
	\[
	U_n = \underbrace{U \times_X \cdots \times_X U}_{n+1 \text{ times}}.
	\]
	The face maps are given by the projections and the degeneracy maps by diagonals.
\end{definition}

\begin{definition}[Totalization]
	Let $\mathcal{C}$ be a category with limits and let
	\[
	X^\bullet : \Delta \to \mathcal{C}
	\]
	be a cosimplicial object. The \emph{totalization} of $X^\bullet$
	is defined as the limit of the cosimplicial diagram
	\[
	\operatorname{Tot}(X^\bullet)
	:=
	\lim_{\Delta} X^\bullet.
	\]
	If $\mathcal{C}$ is a homotopical category, the totalization is
	defined as the homotopy limit
	\[
	\Totof{X^\bullet}
	:=
	\operatorname{holim}_{\Delta} X^\bullet.
	\]
\end{definition}

\begin{remark}
	In the applications relevant for descent theory, the category $\mathcal{C}$
	will typically be the category $\Cat$ of (small) categories, or a closely
	related $(2,1)$-category such as $\Grpd$. In this case, the totalization
	$\Tot(X^\bullet)$ is again a category, obtained as the limit in $\Cat$,
	and can be described concretely in terms of compatible objects and morphisms
	in the diagram $X^\bullet$. In particular, this viewpoint will allow us to
	interpret totalizations as categories of descent data.
\end{remark}

\begin{definition}[Truncated totalization]
	Let $X^\bullet : \Delta \to \mathcal{C}$ be a cosimplicial object.
	For $n \ge 0$, define the $n$-th totalization by
	\[
	\Tot_n(X^\bullet)
	:=
	\lim_{\Delta_{\le n}} X^\bullet,
	\]
	where $\Delta_{\le n}$ is the full subcategory of $\Delta$
	on objects $[0],\dots,[n]$.
\end{definition}

\begin{remark}
	Totalization can be viewed as the simplicial limit of a cosimplicial
	object. In descent theory, given a covering $U \to X$ with Čech nerve
	$U_\bullet$ and a presheaf $F$, the cosimplicial object
	$F(U_\bullet)$ is obtained by evaluating $F$ on the Čech nerve.
	Descent for $F$ is equivalent to the statement that
	\[
	F(X) \cong \Tot(F(U_\bullet)).
	\]
	In this work we restrict to $1$-stacks, i.e.\ stacks taking values in the
	$(2,1)$-category $\Grpd$. Since groupoids correspond to $1$-truncated
	homotopy types, all higher homotopy groups $\pi_n$ vanish for $n \ge 2$.
	Consequently, the homotopy limit over the Čech nerve is already determined
	by the truncated totalization over $\Delta_{\le 2}$, see \cite[][Propoition 2.8]{khan2022stacks}.
\end{remark}

\begin{definition}[Descent datum]
	Let $F$ be a presheaf and let $U \to X$ be a covering with Cech nerve
	$U_\bullet$. A descent datum for $F$ along $U \to X$ is an object of
	the totalization
	\[
	\Tot\big(F(U_\bullet)\big).
	\]
	If $F$ takes values in $\Grpd$, this is equivalently an object of the
	truncated totalization $\Tot_2(F(U_\bullet))$, hence consists of:
	\begin{enumerate}
		\item an object $\xi \in F(U)$,
		\item an isomorphism between the two pullbacks of $\xi$ to
		$U\times_X U$,
		\item a cocycle condition on $U\times_X U\times_X U$.
	\end{enumerate}
\end{definition}

\begin{proposition}[Sheaf condition as Cech descent]
	Let $F$ be a sheaf of sets on a site and let $U \to X$ be a covering
	with Cech nerve $U_\bullet$. Then
	\[
	F(X)
	\cong
	\operatorname{Tot}
	\big(
	F(U_\bullet)
	\big).
	\]
	Since sets are $0$-truncated, the totalization reduces to the equalizer
	\[
	F(X)
	\to
	F(U)
	\rightrightarrows
	F(U \times_X U).
	\]
\end{proposition}

The Amitsur complex computes the totalization of the Čech nerve of
$\Spec(B) \to \Spec(A)$, and faithfully flat descent for rings states
that this totalization recovers the ring $A$. Thus descent theory in
algebraic geometry ultimately reduces to exactness of the Amitsur complex.

\begin{theorem}[Amitsur complex]
	Let $A \to B$ be a faithfully flat morphism of rings.
	Then the augmented Amitsur complex
	\[
	0 \to A
	\longrightarrow B
	\longrightarrow B \otimes_A B
	\longrightarrow B \otimes_A B \otimes_A B
	\longrightarrow \cdots
	\]
	with differentials given by the alternating sum of the coface maps
	is exact.
\end{theorem}

\begin{proof}
	The cosimplicial ring $B^\bullet$ associated to the morphism
	$A \to B$ is the Amitsur complex of the extension.
	If $A \to B$ is faithfully flat, the augmented Amitsur complex
	\[
	0 \to A \to B \to B \otimes_A B \to B \otimes_A B \otimes_A B \to \cdots
	\]
	is exact. Exactness of this complex implies that $A$ is the limit
	of the cosimplicial diagram $B^\bullet$. Hence
	\[
	A \cong \operatorname{Tot}(B^\bullet).
	\]
\end{proof}

\begin{remark}
	The cosimplicial ring $B^\bullet$ is the Amitsur complex associated to
	$A \to B$. Faithfully flat descent for rings is equivalent to exactness of this
	complex.
\end{remark}



\begin{theorem}[Faithfully flat descent for modules]\label{thm:faithfully-flat-descent-modules}
	(\cite[Tag 023F, 03OA]{stacks-project})
	Let $A \to B$ be a faithfully flat morphism of rings and let $B^\bullet$
	be the Čech nerve. Then the natural functor
	\[
	\Mod_A
	\longrightarrow
	\Totof{\Mod(B^\bullet)}
	\]
	is an equivalence of categories.
\end{theorem}

\begin{proof}
	We divide the proof into two steps.
	
	\medskip
	\textbf{Step 1: Fully faithfulness.}
	Let $M$ be an $A$-module. Consider the associated cosimplicial
	$B$-module
	\[
	M \otimes_A B^\bullet.
	\]
	The Amitsur complex of $M$ is the complex
	\[
		M \to M \otimes_A B \to M \otimes_A B \otimes_A B \to \cdots
	\]
	obtained from this cosimplicial object.
	
	Since $A \to B$ is faithfully flat, the Amitsur complex is exact in positive
	degrees and has $H^0 \cong M$.
	
	It follows that
	\[
	M \cong \Totof{M \otimes_A B^\bullet}.
	\]
	Thus $M$ can be recovered from its pullback to $B^\bullet$, which shows that
	the functor
	\[
	\Mod_A \to \Totof{\Mod(B^\bullet)}
	\]
	is fully faithful.
	
	\medskip
	
	\noindent	
	\textbf{Step 2: Essential surjectivity.}
	Let $(N,\varphi)$ be an object of $\Totof{\Mod(B^\bullet)}$, i.e.\ a
	$B$-module $N$ together with descent data, that is, an isomorphism
	\[
	\varphi : N\otimes_A B \xrightarrow{\sim} B\otimes_A N
	\]
	satisfying the cocycle condition
	\[
	d^0(N) \cong_\varphi d^1(N).
	\]
	This suffices, since descent data for modules is governed by objects,
	isomorphisms, and a cocycle condition, i.e.\ by levels $0,1,2$ of the Čech
	nerve. Equivalently, the cosimplicial diagram $\Mod(B^\bullet)$ takes values
	in the $2$-category $\Cat$, so its totalization is already determined by the
	truncation $\Tot_2$, and no higher coherence data is required.
	Define the $A$-module
	\[
	M := \{ n \in N \mid 1 \otimes n = \varphi(n \otimes 1) \}.
	\]
	Equivalently, $M$ is the equalizer
	\[
	\begin{tikzcd}
		M \arrow[r,"\psi"] & N \arrow[r,shift left,"n\mapsto n\otimes 1"] \arrow[r,shift right,"\;n\mapsto \varphi(n\otimes 1)"'] & N\otimes_A B.
	\end{tikzcd}
	\]
	
	We claim that the natural map
	\[
	B \otimes_A M \longrightarrow N, \qquad b\otimes m \mapsto b\cdot \psi(m),
	\]
	is an isomorphism.
	
	\medskip
	
	Consider the two cosimplicial modules
	\[
	M_\bullet := M \otimes_A B^\bullet,
	\qquad
	N_\bullet := \big(N \to N\otimes_A B \to N\otimes_A B\otimes_A B \to \cdots\big),
	\]
	where $M_\bullet$ is the Amitsur complex of $M$, (\cite[Tag 03OA]{stacks-project}), and $N_\bullet$ is the
	twisted Amitsur complex associated to $(N,\varphi)$ (\cite[Tag 023F]{stacks-project}), with coface maps
	given as follows:
	\[
	\begin{aligned}
		\delta^0(n) &= 1\otimes n, &
		\delta^1(n) &= \varphi(n\otimes 1), \\
		\delta^0(n\otimes b) &= 1\otimes n\otimes b, &
		\delta^1(n\otimes b) &= n\otimes 1\otimes b, &
		\delta^2(n\otimes b) &= (\varphi\otimes \mathrm{id})(n\otimes b\otimes 1),
	\end{aligned}
	\]
	and similarly in higher degrees. The cocycle condition on $\varphi$
	ensures that these maps satisfy the cosimplicial identities.
	
	By construction of $M$ as the equalizer, we have
	\[
	M \cong H^0(N_\bullet).
	\]
	
	Tensoring $N_\bullet$ with $B$, we obtain a cosimplicial $B$-module
	\[
	B \otimes_A N_\bullet.
	\]
	Using the descent isomorphism $\varphi$, one checks that the twisting
	disappears after base change, and there is a canonical identification
	\[
	B \otimes_A N_\bullet \;\cong\; B \otimes_A (M \otimes_A B^\bullet).
	\]
	In particular, $B \otimes_A N_\bullet$ identifies with the Amitsur complex
	of the $B$-module $B\otimes_A M$.
	
	Since $A \to B$ is faithfully flat, the Amitsur complex of $M$ is exact,
	hence $M_\bullet$ is exact. Flatness implies that tensoring with $B$
	preserves exactness, so $B \otimes_A M_\bullet$ is exact. Therefore
	\[
	B \otimes_A N_\bullet
	\]
	is exact.
	
	Again by faithful flattness, $N_\bullet$ itself is exact.
	
	Taking $H^0$, we obtain
	\[
	B \otimes_A M \;\cong\; H^0(B \otimes_A N_\bullet)
	\;\cong\; H^0(N_\bullet)
	\;\cong\; N,
	\]
	which proves that the map $B \otimes_A M \to N$ is an isomorphism.
	
	\medskip
	
	Thus every descent datum is effective, and the functor
	\[
	\Mod_A \longrightarrow \Totof{\Mod(B^\bullet)}
	\]
	is essentially surjective.
\end{proof}


\begin{proposition}[Descent for affine schemes]
	Let $U \to X$ be a faithfully flat morphism of affine schemes and let
	$U_\bullet$ be its Čech nerve. Then
	\[
	X \simeq \operatorname{colim}(U_\bullet)
	\]
	in the category of schemes.
	Equivalently, for every scheme $T$ we have
	\[
	\Hom(T, X)
	\simeq
	\Tot
	\big(
	\Hom(T, U_\bullet)
	\big).
	\]
\end{proposition}

\begin{theorem}[Faithfully flat descent for quasi-coherent sheaves]
	(\cite[Tag 023T]{stacks-project})
	Let $f : U \to X$ be an fpqc covering of schemes and let $U_\bullet$
	be the Čech nerve.
	Then the pullback functor induces an equivalence
	\[
	\QCoh(X)
	\cong
	\operatorname{Tot}
	\big(
	\QCoh(U_\bullet)
	\big).
	\]
\end{theorem}

\begin{proof}
	We first reduce to the affine case and then glue the local statements.
	
	Choose an affine open basis
	\[
	X = \bigcup_{j \in J} V_j,
	\qquad
	V_j = \Spec A_j,
	\]
	which is closed under intersection.
	Then each intersection $V_j\cap V_{j'}$ is again affine.
	For each $j$, set
	\[
	U_j := U \times_X V_j.
	\]
	Since fpqc morphisms are stable under base change, each morphism
	\[
	U_j \to V_j
	\]
	is again an fpqc covering, and its Cech nerve is given by
	\[
	(U_j)_\bullet = U_\bullet \times_X V_j.
	\]

			\QCoh(V_j) \longrightarrow \Tot\big(\QCoh((U_j)_\bullet)\big)

	\]
	be a descent datum on $U$. Restricting to $U_j$ yields for each $j$ an object
	\[
	(\mathcal F_{U_j},\alpha_j)\in \Tot\big(\QCoh((U_j)_\bullet)\big).
	\]
	Assuming the affine case, there exists a quasi-coherent sheaf
	\[
	\mathcal F_j\in \QCoh(V_j)

	whose pullback along $U_j\to V_j$ is identified with

	whose pullback to $U_{jj'}$ is the identity on $\mathcal{F}_U|_{U_{jj'}}$. By uniqueness, these isomorphisms satisfy

	glue to a quasi-coherent sheaf
	\[
	\mathcal F\in \QCoh(X).

	By construction, $f^*\mathcal{F}$ with its canonical descent datum is identified with $(\mathcal{F}_U, \alpha)$.
	This proves essential surjectivity globally.
	
	Now for fully faithfulness let $\mathcal{F}, \mathcal{G}\in \QCoh(X)$. We show that there is a bijection
	\[
	\Hom_{\QCohX}(\mathcal{F}, \mathcal{G}) \to \Hom_{\Totof{\QCoh(U_\bullet)}}(f^*\mathcal{F}, f^*\mathcal{G}).
	\]	
	
	Injectivity: 
	
	Let $\varphi, \psi: \mathcal{F}\to \mathcal{G}$ have the same pullback to $U$, then for every $V_j$ their restrictions
	\[
	\varphi|_{V_j}, \psi|_{V_j}: \mathcal{F}|_{V_j} \to \mathcal{G}|_{V_j}
	\]
	have the same pullback to $U_j$. By the affine case for $V_j$, we get $\varphi|_{V_j} = \psi|_{V_j}$. Since they agree on an open cover, $\varphi = \psi$. 
	
	Surjectivity:
	
	Let $\theta:f^*\mathcal{F} \to f^*\mathcal{G}$ be a morphism in $\Totof{\QCoh(U_\bullet)}$. Restricting to $U_j$ yields a morphism in $\theta_{j}:f^*\mathcal{F}|_{U_j}\to f^*\mathcal{G}|_{U_j}$ in $\Totof{\QCoh((U_j)_\bullet)}$. By the affine case this corresponds to a morphism $\mathcal{F}|_{U_j} \to \mathcal{G}|_{U_j}$. Now for $j, j'$, the morphism
	$\theta_{j}|_{U_{jj'}}$ and $\theta_{j'}|_{U_{jj'}}$ restrict to the same pullback $\theta_{jj'}$ in 
	\[
	\Totof{\QCoh((U_{jj'})_\bullet)},
	\]
	which corresponds to a morphism $\mathcal{F}|_{U_{jj'}} \to \mathcal{G}|_{U_{jj'}}$, again by the affine version. 
	By the sheaf property, these morphisms agree on overlaps and therefore glue uniquely to a morphism
	\[
	\mathcal F \to \mathcal G.
	\]
	By construction, this morphism pulls back to $\theta$ in $\Totof{\QCoh(U_\bullet)}$, which proves surjectivity.
	
	Hence, it remains to prove the affine case. Thus assume
	\[
	X=\Spec A
	\]
	is affine. Since $U\to X$ is fpqc, we may choose a finite affine open covering
	\[
	U=\bigcup_{i\in I} U_i,
	\qquad
	U_i=\Spec B_i.
	\]
	Then
	\[
	\coprod_{i\in I} U_i \to X
	\]
	is again an fpqc covering, and its Cech nerve refines the Cech nerve of
	$U\to X$.
	Since descent for quasi-coherent sheaves is local on the source, replacing $U$ by the fpqc covering $\coprod_{i\in I} U_i \to X$ does not change the category of descent data (\cite[Tag 02L4]{stacks-project}). Hence, it suffices to prove the statement for the affine fpqc cover
	\[
	\coprod_{i\in I}\Spec B_i \to \Spec A.
	\]
	For an affine scheme, we have equivalence of categories, see \cite[Tag 01IB]{stacks-project}:
	\[
	\QCoh(\Spec A)\cong \Mod_A
	\]
	and
	\[
	\QCoh(\Spec B_i)\cong \Mod_{B_i}
	\]
	Under these identifications, the Cech nerve of the covering
	\[
	\coprod_i \Spec B_i \to \Spec A
	\]
	corresponds to the cosimplicial ring obtained from the fpqc algebra
	\[
	A \to \prod_i B_i.
	\]
	Since a finite product of faithfully flat $A$-algebras is again faithfully
	flat, descent for quasi-coherent sheaves over this affine fpqc cover reduces
	to faithfully flat descent for modules (Thm \ref{thm:faithfully-flat-descent-modules}):
	\[
	\Mod_A \cong \Tot\big(\Mod(B^\bullet)\big),
	\]
	where $B=\prod_i B_i$ and $B^\bullet$ is the associated Amitsur-Cech nerve.
	
	Equivalently, if one prefers to keep the covering indexed by the family
	$\{B_i\}$, then descent data for quasi-coherent sheaves on the affine fpqc
	cover $\{\Spec B_i\to \Spec A\}$ is exactly descent data for $A$-modules with
	respect to the corresponding Cech diagram, and the equivalence follows from
	faithfully flat descent for modules.
	This proves the affine case, and therefore the theorem.
\end{proof}



\begin{theorem}[Descent for affine schemes via relative spectrum]
	Let $X$ be a scheme and let $U \to X$ be an fpqc covering with Cech nerve $U_\bullet$.
	Then the functor
	\[
	\Aff/X \;\longrightarrow\; \Tot\big((\Aff/U_\bullet)\big)
	\]
	is an equivalence, where $\Aff/X$ denotes the category of affine schemes over $X$.
	
	Equivalently, for quasi-coherent $\mathcal{O}_X$-algebras one has
	\[
	\QCohAlg(X)
	\cong
	\Tot\big(\QCohAlg(U_\bullet)\big),
	\]
	and this equivalence is compatible with the relative spectrum construction
	\[
	\underline{\Spec}_X : \QCohAlg(X)^{op} \;\longrightarrow\; \Aff/X.
	\]
\end{theorem}

\begin{proof}
	We first treat quasi-coherent algebras and then pass to affine schemes via
	relative $\Spec$.
	
	Since quasi-coherent $\mathcal{O}_X$-modules satisfy fpqc descent, the same
	is true for quasi-coherent $\mathcal{O}_X$-algebras: the algebra structure
	(consisting of multiplication and unit maps) is given by morphisms in
	$\QCoh(X)$ and is preserved under limits. Hence
	\[
	\QCohAlg(X)
	\cong
	\Tot\big(\QCohAlg(U_\bullet)\big).
	\]
	
	Now consider the relative spectrum construction
	\[
	\underline{\Spec}_X : \QCohAlg(X)^{op} \to \Aff/X.
	\]
	This functor is functorial and compatible with base change. Moreover, it
	induces an equivalence between quasi-coherent algebras on $X$ and affine
	schemes over $X$.
	
	The cosimplicial diagram
	$\QCohAlg(U_\bullet)$ yields the cosimplicial diagram of affine schemes
	$\Aff/U_\bullet$. Since $\underline{\Spec}$ is compatible with base change,
	it preserves the descent data.
	
	Therefore the equivalence for quasi-coherent algebras induces an equivalence
	\[
	\Aff/X \cong \Tot\big(\Aff/U_\bullet\big).
	\]
\end{proof}

\begin{remark}
	Descent for affine schemes (and more generally affine morphisms) with
	respect to fpqc coverings has been prooven in the Stacks Project,
	see \cite[Tag 0244]{stacks-project}. The reduction to quasi-coherent
	algebras and the compatibility with relative $\Spec$ are standard and
	follow from fpqc descent for quasi-coherent sheaves,
	see \cite[Tag 023T]{stacks-project}.
\end{remark}

\begin{corollary}[Affine constructions are fpqc-local on the base]
	Let $X$ be a scheme and let $U \to X$ be an fpqc covering with Čech nerve
	$U_\bullet$. Then any affine morphism over $X$ may be constructed fpqc-locally
	on $U$ together with descent data, and descends uniquely to an affine morphism
	over $X$.
	
	More precisely, the functor
	\[
	\Aff/X \longrightarrow \Tot\big((\Aff/U_\bullet)\big)
	\]
	is an equivalence. Equivalently, for every cosimplicial system of affine
	$U_n$-schemes
	\[
	Y_\bullet \to U_\bullet
	\]
	equipped with descent data, there exists a unique affine $X$-scheme
	\[
	Y \to X
	\]
	whose pullback to $U_\bullet$ recovers $Y_\bullet$.
\end{corollary}

\begin{remark}
	This corollary is used repeatedly in quotient constructions: one first
	constructs the desired object locally on an fpqc cover, often after reducing
	to affine charts, and then glues the local affine pieces by fpqc descent.
	In particular, auxiliary affine morphisms arising in the proof of the
	Keel--Mori theorem may be constructed locally and descended globally.
\end{remark}
