\chapter{Keel Mori Theorem}
The proof of the Keel-Mori theorem follows the Stacks Project. We proceed by first constructing moduli spaces on a special local class of algebraic stacks, called well-nigh affine stacks, and then gluing these local moduli spaces along
\etale\ overlaps.

\section{Local constructions}

From now on we will be working with Deligne-Momford stacks follwing jarod's notes.


\begin{lemma}[Local Structure Theorem of Deligne-Mumford Stacks]\label{lem:local-structure-theorem-dm-stacks}
	(\cite[][Theorem 4.2.11]{alper2026stacks})
	Let $\mathcal{X}$ be a separated DM stack and let $x:\Specof{k} \to  \mathcal{X}$ be a point with geometric stabilizer $G_x$. Then there exists an affine \etale\ morphism 
	\[
	f: [\Specof{A}/ G_x] \to \mathcal{X}
	\]
	sending $w\in |[\Specof{A}/ G_x]|$ to $x$, such that f induces an isomphism 
	\[
	G_w \xrightarrow{\simeq} G_x.
	\]
\end{lemma}


\begin{lemma}\label{lem:coarse-moduli-etale-local}
	Let $\pi : \mathcal{X} \to X$ be a morphism to an algebraic space. Suppose
	that there is an étale covering $\{f_i : X_i \to X\}$ such that
	\[
	\pi_i : \mathcal{X} \times_X X_i \longrightarrow X_i
	\]
	is a coarse moduli space for each $i$. Then $\pi : \mathcal{X} \to X$ is a
	coarse moduli space.
\end{lemma}

\begin{proof}
	We verify the two axioms of Definition~\ref{def:coarse-moduli-space}.
	
	\textbf{Geometric points}
	Let $k$ be an algebraically closed field and let
	$x : \Spec(k) \to X$ be a geometric point. Since $\{f_i : X_i \to X\}$
	is an étale covering, there exists an  $i$ and a lift
	$\tilde{x} : \Spec(k) \to X_i$ with $f_i \circ \tilde{x} = x$. Indeed,
	algebraically closed fields admit no nontrivial étale extensions, so étale
	covers are surjective on geometric points.
	
	Since $\pi_i$ is a coarse moduli space, the induced map
	\[
	|(\mathcal{X} \times_X X_i)(k)| \longrightarrow X_i(k)
	\]
	is a bijection onto isomorphism classes. The fiber of $\pi$ over $x$
	is canonically identified with the fiber of $\pi_i$ over $\tilde{x}$,
	via the base-change isomorphism
	\[
	\mathcal{X} \times_X \Spec(k)
	\;\cong\;
	(\mathcal{X} \times_X X_i) \times_{X_i} \Spec(k).
	\]
	Hence the bijection for $\pi_i$ implies the corresponding bijection for
	$\pi$.
	
	\medskip
	\textbf{Universality}
	Let $Y$ be an algebraic space and $f : \mathcal{X} \to Y$ a morphism.
	For each $i$, consider the composite
	\[
	\mathcal{X} \times_X X_i \longrightarrow \mathcal{X}
	\xrightarrow{\;f\;} Y.
	\]
	Since $\pi_i$ is a coarse moduli space, there exists a unique morphism
	$g_i : X_i \to Y$ such that the following commutes:

	uniqueness in the universal property of $\pi_{ij}$, we conclude
	\[
	g_i|_{X_{ij}} = g_j|_{X_{ij}}.

	cover $\{X_i \to X\}$. Since $Y$ is an algebraic space, it is a
	sheaf in the étale topology, so the $g_i$ glue to a unique morphism
	$g : X \to Y$.

	along each $f_i$, and morphisms to algebraic spaces can be checked étale
	locally. Uniqueness of $g$ follows from the sheaf property of $Y$: any
	two morphisms $X \to Y$ agreeing on the cover $\{X_i \to X\}$ must
	coincide. Hence Axiom~(1) holds.
\end{proof}

\begin{theorem}[Keel-Mori Theorem]
	Let $\mathcal{X}$ be a Deligne-Mumford stack separated and of finite type over a field. Then there exists a coarse moduli space
	\[
	\pi: \mathcal{X}\to X.
	\]
\end{theorem}

\begin{remark}
	By remark \ref{rem:dm-stack-finite-inertia}, the classical finite inertia condition can be dropped.
\end{remark}

	\medskip
	\textbf{Descending the local maps to the overlaps}
	
	We first construct the global map $\pi : \mathcal{X} \to X$, then verify it
	is a coarse moduli space.
	
	For each pair $(i,j)$, consider the fiber product
	\[
	\mathcal{X}_{ij} := \mathcal{X}_i \times_{\mathcal{X}} \mathcal{X}_j.
	\]
	Since $f_i$ and $f_j$ are representable and étale, the two projections
	\[
	p_i : \mathcal{X}_{ij} \to \mathcal{X}_i,
	\qquad
	p_j : \mathcal{X}_{ij} \to \mathcal{X}_j
	\]
	are also representable and étale. Moreover, for every geometric point
	$\bar{x} \to \mathcal{X}_{ij}$, the induced map on stabilizer groups
	\[
	\Aut_{\mathcal{X}_{ij}}(\bar{x})
	\xrightarrow{\;\sim\;}
	\Aut_{\mathcal{X}_i}(p_i(\bar{x}))
	\]
	is an isomorphism, since $f_i$ induces isomorphisms on all geometric
	stabilizer groups by hypothesis, and $\mathcal{X}_{ij} \to \mathcal{X}_i$
	is a base change of $f_j$ along $f_i$.
	
	In particular, $\mathcal{X}_{ij}$ is a Deligne--Mumford stack with finite
	inertia. Since $\mathcal{X}_i = [U_i/G_i]$ is a quotient of an affine by
	a finite group, the base change
	\[
	\mathcal{X}_{ij} \cong
	[U_i \times_{\mathcal{X}} \mathcal{X}_j \;/\; G_i]
	\]
	is again a quotient of an affine scheme by a finite group. In particular,
	$\mathcal{X}_{ij}$ admits a coarse moduli space
	\[
	\pi_{ij} : \mathcal{X}_{ij} \longrightarrow X_{ij},
	\]
	and the morphisms $\pi_i \circ p_i : \mathcal{X}_{ij} \to X_i$ and
	$\pi_j \circ p_j : \mathcal{X}_{ij} \to X_j$ both factor uniquely through
	$\pi_{ij}$, by the universal property of the coarse moduli space of $\mathcal{X}_{ij}$:
	\[
	\phi_{ij} : X_{ij} \to X_i,
	\qquad
	\phi_{ji} : X_{ij} \to X_j.
	\]
	We claim that $\phi_{ij}$ and $\phi_{ji}$ are both étale. Indeed,
	$\pi_i \circ p_i$ is étale (as a composition of étale morphisms), and
	$\pi_i$ is the coarse moduli space of $\mathcal{X}_i = [U_i/G_i]$. By
	Lemma~\ref{lem:descending-etale-to-quotient}, any étale morphism from
	$\mathcal{X}_{ij}$ to the algebraic space $X_i$ that is an isomorphism
	on geometric stabilizers descends to an étale morphism $X_{ij} \to X_i$.
	The same argument applies to $\phi_{ji}$.
	
	\medskip
	\textbf{Forming the global algebraic space via the étale equivalence relation.}
	
	Define
	\[
	U := \coprod_i X_i.
	\]
	This is a scheme (a disjoint union of affine schemes). Define
	\[
	R := \coprod_{i,j} X_{ij},
	\]
	with source and target maps
	\[
	s : R \to U, \quad s|_{X_{ij}} = \phi_{ij},
	\qquad
	t : R \to U, \quad t|_{X_{ij}} = \phi_{ji}.
	\]
	Both $s$ and $t$ are étale morphisms of schemes, since each component
	$\phi_{ij}$ and $\phi_{ji}$ is étale by Step 3a.
	
	We verify that $(s,t) : R \rightrightarrows U$ is an étale equivalence
	relation in the sense of Lemma~\ref{lem:etale-eq-quotient-of-affines}.
	The key point is that the map
	\[
	j = (s,t) : R \to U \times U
	\]
	is a monomorphism: two sections in $X_{ij}(T)$ with the same image
	in $(U \times U)(T) = X_i(T) \times X_j(T)$ must be equal, because
	the coarse moduli space maps $\pi_i, \pi_j$ separate geometric points
	(condition (2) of Definition~\ref{def:coarse-moduli-space}).
	Reflexivity, symmetry, and transitivity of the relation follow from
	the stack structure of $\mathcal{X}$: the diagonal
	$\mathcal{X}_i \to \mathcal{X}_{ii}$ gives the identity sections,
	the swap $\mathcal{X}_{ij} \cong \mathcal{X}_{ji}$ gives symmetry, and
	the composition $\mathcal{X}_{ij} \times_{\mathcal{X}_j} \mathcal{X}_{jk}
	\cong \mathcal{X}_{ik}$ gives transitivity. All of these pass to the
	coarse moduli spaces $X_{ij}$ by the universal property.
	
	By Lemma~\ref{lem:etale-eq-quotient-of-affines}, the sheaf quotient
	\[
	X := (U / R)^\sharp
	\]
	is an algebraic space, and the canonical morphism
	\[
	p : U = \coprod_i X_i \longrightarrow X
	\]
	is surjective and \etale. In particular

	\medskip
	\textbf{Glue the local coarse moduli spaces via \etale\ descent.}
	
	We make use of Theorem~\ref{thm:fpqc-descent-affine-schemes} and explicitly explain how it enters.
	The morphism $p : U \to X$ is an étale surjection, hence in particular an
	fpqc cover. The fiber product $U \times_X U$ satisfies
	\[
	U \times_X U \;\cong\; R \;=\; \coprod_{i,j} X_{ij},
	\]
	since $p$ coequalizes $s$ and $t$. Hence, two points of $U$ have the same image in $X$ if and only if they are related by $R$. Under this identification,
	the two projections $\pi_1, \pi_2 : U \times_X U \to U$ correspond to $s$
	and $t$ respectively.
	
	For each $i$, the affine scheme $X_i$ comes with its inclusion
	$X_i \hookrightarrow U$, giving an object $(X_i \hookrightarrow U) \in
	\AffOver{U}$. The morphisms $\phi_{ij} : X_{ij} \to X_i$ induced by the universal property of the coarse moduli spaces of $\mathcal{X}_{ij}$ from earlier assemble into a gluing isomorphism
	\[
	\alpha : \pi_1^*U \xrightarrow{\;\sim\;} \pi_2^*U
	\]
	in $\AffOver{U \times_X U}$, whose $(i,j)$-component is induced by
	$\phi_{ij}$ and $\phi_{ji}$. The normalization condition $\Delta^*\alpha =
	id$ holds by reflexivity of $R$ (i.e.\ $\phi_{ii} = id_{X_i}$), and the
	cocycle condition
	\[
	\pi_{23}^*(\alpha) \circ \pi_{12}^*(\alpha) = \pi_{13}^*(\alpha)
	\]
	in $\AffOver{U \times_X U \times_X U}$ follows from transitivity of $R$
	established above.
	
	Thus $(U, \alpha)$ is an object of $\Desc_p(\AffOver{(-)})$, i.e.\ a
	descent datum for the pseudofunctor $\AffOver{(-)}$ along the fpqc cover
	$p : U \to X$. By Theorem~\ref{thm:fpqc-descent-affine-schemes}, this
	descent datum is effective, i.e. there exists a unique affine scheme over $X$
	whose pullback along $p$ recovers $(U, \alpha)$. This descended object is
	precisely $X$ itself, confirming that the local affine pieces $X_i$ are
	the pullbacks of $X$ along $p$, and that $X$ is the global scheme
	reconstructed from the local data by fpqc descent.
	
	\medskip
	\textbf{Constructing the morphism $\pi : \mathcal{X} \to X$.}
	
	For each $i$, the composition
	\[
	\mathcal{X}_i \xrightarrow{\;\pi_i\;} X_i \hookrightarrow U
	\xrightarrow{\;p\;} X
	\]
	gives a morphism $\mathcal{X}_i \to X$. On overlaps, the two composites
	$\mathcal{X}_{ij} \rightrightarrows X$ coming from $i$ and $j$ agree, since both factor through $\pi_{ij} : \mathcal{X}_{ij} \to X_{ij}$,
	and the two maps $X_{ij} \rightrightarrows X$ are exactly
	$p \circ \phi_{ij}$ and $p \circ \phi_{ji}$, which are equal in $X$
	because $(X_{ij}, \phi_{ij}, \phi_{ji})$ is precisely one component of
	the equivalence relation $R$ and $p$ coequalizes $s$ and $t$.
	
	Since $\{f_i : \mathcal{X}_i \to \mathcal{X}\}$ is an étale cover of
	$\mathcal{X}$ and the morphisms $\mathcal{X}_i \to X$ are compatible
	on overlaps, they descend to a unique morphism
	\[
	\pi : \mathcal{X} \longrightarrow X,
	\]
	by the sheaf property of $X$ as an algebraic space over the big étale
	site.
	
	\medskip
	\textbf{Verifying the coarse moduli space axioms.}
	
	\emph{(i) Geometric points.}
	Let $\kappa$ be an algebraically closed field and $\bar{x} : \Spec\kappa
	\to \mathcal{X}$ a geometric point. Choose $i$ such that $\bar{x}$ lifts
	to $\bar{x}_i : \Spec\kappa \to \mathcal{X}_i$ (possible since the
	$\mathcal{X}_i$ cover $\mathcal{X}$). Then
	\[
	\pi(\bar{x}) = p(\pi_i(\bar{x}_i)).
	\]
	Since $\pi_i$ is a coarse moduli space, $\pi_i$ identifies
	$|\mathcal{X}_i(\kappa)|$ with $X_i(\kappa)$.
	Two geometric points $\bar{x}, \bar{y}$ of $\mathcal{X}$ satisfy
	$\pi(\bar{x}) = \pi(\bar{y})$ in $X(\kappa)$ if and only if their images
	in $U$ are identified by $R$, i.e.\ $(\pi_i(\bar{x}_i), \pi_j(\bar{y}_j))
	\in X_{ij}(\kappa)$ for some $i,j$. By construction of $X_{ij}$ as the
	coarse moduli space of $\mathcal{X}_{ij} = \mathcal{X}_i
	\times_{\mathcal{X}} \mathcal{X}_j$, this happens if and only if $\bar{x}$
	and $\bar{y}$ are isomorphic in $\mathcal{X}(\kappa)$.
	Hence $|\mathcal{X}(\kappa)| \to X(\kappa)$ is a bijection.
	
	\emph{(ii) Universality.}
	Let $Y$ be an algebraic space and $h : \mathcal{X} \to Y$ a morphism.
	For each $i$, the restriction $h \circ f_i : \mathcal{X}_i \to Y$ factors
	uniquely through $\pi_i$ by the universal property of the coarse moduli
	space $X_i$, yielding a unique
	morphism $g_i : X_i \to Y$.
	On each overlap $X_{ij}$, the two morphisms
	$g_i \circ \phi_{ij}$ and $g_j \circ \phi_{ji}$ are both equal to the
	unique factorization of $h \circ f_i \circ p_i = h \circ f_j \circ p_j
	: \mathcal{X}_{ij} \to Y$ through $X_{ij}$.
	Hence the collection $(g_i)$ defines a morphism of the equivalence
	relation $(R \rightrightarrows U) \to Y$, and by
	the universal property of the quotient $X = (U/R)^\sharp$, they glue
	to a unique morphism
	\[
	g : X \longrightarrow Y.
	\]
	By construction, $g \circ \pi = h$.
	Uniqueness of $g$: if $g' : X \to Y$ also satisfies $g' \circ \pi = h$,
	then $g' \circ p \circ \pi_i = g \circ p \circ \pi_i$ for all $i$. Since
	$\pi_i$ is an epimorphism, $g' \circ p|_{X_i} =
	g \circ p|_{X_i}$, i.e.\ $g' = g$ after pullback along the étale
	surjection $p : U \to X$. Since $p$ is an epimorphism of sheaves,
	$g' = g$.
	
	\medskip
	
	This completes the construction of the coarse moduli space
	$\pi : \mathcal{X} \to X$.
\end{proof}
