\chapter{Keel Mori Theorem}
The proof of the Keel-Mori theorem follows the Stacks Project. We proceed by first constructing moduli spaces on a special local class of algebraic stacks, called well-nigh affine stacks, and then gluing these local moduli spaces along
\etale\ overlaps.

\section{Local constructions}

From now on we will be working with Deligne-Momford stacks follwing jarod's notes.


\begin{lemma}[Local Structure Theorem of Deligne-Mumford Stacks]\label{lem:local-structure-theorem-dm-stacks}
	(\cite[][Theorem 4.2.11]{alper2026stacks})
	Let $\mathcal{X}$ be a separated DM stack and let $x:\Specof{k} \to  \mathcal{X}$ be a point with geometric stabilizer $G_x$. Then there exists an affine \etale\ morphism 
	\[
	f: [\Specof{A}/ G_x] \to \mathcal{X}
	\]
	sending $w\in |[\Specof{A}/ G_x]|$ to $x$, such that f induces an isomphism 
	\[
	G_w \xrightarrow{\simeq} G_x.
	\]
\end{lemma}



\begin{lemma}[Affine quotient by a finite group]\label{lem:affine-quotient-finite-group}
	Let $G$ be a finite group acting on an affine scheme $\Spec(A)$. Then the canonical morphism
	\[
	\pi : \Spec(A) \longrightarrow \Spec(A^G)
	\]
	induces a morphism of stacks
	\[
	[\Spec(A)/G] \longrightarrow \Spec(A^G),
	\]
	and this morphism is a coarse moduli space.
\end{lemma}

\begin{proof}
	We check the two defining properties of a coarse moduli space.
	
	\emph{(i) Geometric points.}
	The induced map on geometric points identifies the set of $G$-orbits in
	$\Spec(A)(\bar{k})$ with the set of points of $\Spec(A^G)(\bar{k})$.
	The morphism $\Spec(A) \to \Spec(A^G)$ is finite and surjective, since
	$A$ is integral over $A^G$ for a finite group action. In particular, two
	geometric points have the same image in $\Spec(A^G)$ if and only if they lie in the same $G$-orbit.
	
	\emph{(2) Universality.}
	Let $Y$ be an algebraic space and let
	\[
	f : [\Spec(A)/G] \longrightarrow Y
	\]
	be a morphism. Since $f$ is $G$-invariant, its restriction to $\Spec(A)$
	corresponds locally on an étale cover of $Y$ to a $G$-invariant morphism of
	affine schemes. Equivalently, on coordinate rings it is given by a
	$G$-invariant homomorphism, hence factors uniquely through $A^G$.
	Since $Y$ is a sheaf on the \etale\ site, these local factorizations glue
	to a unique morphism $\Spec(A^G)\to Y$ making the diagram commute.
	
	Thus $\pi : [\Spec(A)/G]\to \Spec(A^G)$ satisfies the universal property of a
	coarse moduli space.
\end{proof}

\begin{lemma}\label{lem:coarse-moduli-etale-local}
	(\cite[Lemma 5.2.10]{alper2026stacks})
	Let $\pi : \mathcal{X} \to X$ be a morphism to an algebraic space. Suppose
	that there is an étale covering $\{f_i : X_i \to X\}$ such that
	\[
	\pi_i : \mathcal{X} \times_X X_i \longrightarrow X_i
	\]
	is a coarse moduli space for each $i$. Then $\pi : \mathcal{X} \to X$ is a
	coarse moduli space.
\end{lemma}

\begin{proof}
	We verify the two axioms of Definition~\ref{def:coarse-moduli-space}.
	Let $\mathcal{X}_i := \mathcal{X} \times_X X_i$.
	
	\textbf{Geometric points}
	Let $k$ be an algebraically closed field and let
	$x : \Spec(k) \to X$ be a geometric point. Since $\{f_i : X_i \to X\}$
	is an étale covering, there exists an  $i$ and a lift
	$\tilde{x} : \Spec(k) \to X_i$ with $f_i \circ \tilde{x} = x$, by lemma \ref{lem:geometric-points-etale-cover}.
	
	Since $\pi_i$ is a coarse moduli space, the induced map
	\[
	|(\mathcal{X}_i)(k)| \longrightarrow X_i(k)
	\]
	is a bijection onto isomorphism classes. The fiber of $\pi$ over $x$
	is canonically identified with the fiber of $\pi_i$ over $\tilde{x}$,
	via the base-change isomorphism
	\[
	\mathcal{X} \times_X \Spec(k)
	\;\cong\;
	(\mathcal{X} \times_X X_i) \times_{X_i} \Spec(k) = \mathcal{X}_i\times_{X_i}\Specof{k}.
	\]
	Hence the bijection for $\pi_i$ implies the corresponding bijection for
	$\pi$.
	
	\medskip

	\xrightarrow{\;f\;} Y.
	\]

	$g_i : X_i \to Y$ such that the following commutes:

\section{Keel Mori Theorem}

\begin{theorem}[Keel-Mori Theorem]
	Let $\mathcal{X}$ be a Deligne-Mumford stack separated and of finite type over a field. Then there exists a coarse moduli space
	\[
	\pi: \mathcal{X}\to X.
	\]
\end{theorem}

\begin{remark}
	By remark \ref{rem:dm-stack-finite-inertia}, the classical finite inertia condition can be dropped.
\end{remark}

\begin{proof}
	\medskip
	\textbf{Construct covering with coarse moduli space}
	
	Let $x:\Specof{k} \to \mathcal{X}$ be a geometric point. By the local structure theorem of DM stacks (\ref{lem:local-structure-theorem-dm-stacks}), we can find an \etale\ neighborhood of $x$ denoted by:
	\[
	f: (\mathcal{W},w) \to (\mathcal{X}, x),
	\]
	where $\mathcal{W} = [\Specof{A}/G_x]$, such that $f$ is affine, \etale, and induces an isomorphism between the stabilizer groups
	\[
	G_w \xrightarrow{\simeq} G_x.
	\]
	We now shrink $\mathcal{W}$ to a smaller \etale\ neighbourhood, such that $f$ induces isomorphisms on all stabilizer groups.
	
	Let $\mathcal{U}\subset \mathcal{W}$ be the locus consisting of points $z\in |\mathcal{W}|$, such that $f$ induces isomorphisms $G_z \xrightarrow{\simeq} G_{f(z)}$.
	
	We claim, that $\mathcal{U}$ is open. 
	
	\begin{proof}[Proof of the claim]
		
	By a diagram chase, the following is 2-cartesian:
	
	% % https://q.uiver.app/#q=WzAsNCxbMCwwLCJcXG1hdGhjYWx7SX1fXFxtYXRoY2Fse1d9Il0sWzAsMSwiXFxtYXRoY2Fse1d9Il0sWzEsMCwiXFxtYXRoY2Fse0l9X1xcbWF0aGNhbHtYfVxcdGltZXNfXFxtYXRoY2Fse1h9XFxtYXRoY2Fse1d9Il0sWzEsMSwiXFxtYXRoY2Fse1d9XFx0aW1lc19cXG1hdGhjYWx7WH0gXFxtYXRoY2Fse1d9Il0sWzAsMiwiXFxQaGkiXSxbMCwxXSxbMSwzXSxbMiwzXSxbMCwzLCIiLDEseyJzdHlsZSI6eyJuYW1lIjoiY29ybmVyIn19XV0=
	\begin{equation}\label{diag:inertia-pullback-open-substack}
		\begin{tikzcd}
			{\mathcal{I}_\mathcal{W}} & {\mathcal{I}_\mathcal{X}\times_\mathcal{X}\mathcal{W}} \\
			{\mathcal{W}} & {\mathcal{W}\times_\mathcal{X} \mathcal{W}}
			\arrow["\Phi", from=1-1, to=1-2]
			\arrow[from=1-1, to=2-1]
			\arrow["\lrcorner"{anchor=center, pos=0.125}, draw=none, from=1-1, to=2-2]
			\arrow[from=1-2, to=2-2]
			\arrow[from=2-1, to=2-2]
		\end{tikzcd}
	\end{equation}
	
	Now by Lemma \ref{lem:local-structure-theorem-dm-stacks} the morphism $\mathcal{W} \to \mathcal{X}$ is affine \etale, in particular representable and separated, since $\mathcal{X}$ is separated. 
	Consider the diagonal $\Delta_{\mathcal{W}/\mathcal{X}}:\mathcal{W} \to \mathcal{W}\times_\mathcal{X}\mathcal{W}$.
	Since $\mathcal{W}\to \mathcal{X}$ is separated, $\Delta_{\mathcal{W}/\mathcal{X}}$ is closed. Since $\mathcal{W}\to \mathcal{X}$ is representable and \etale, base change along a scheme $T$ induces an \etale\ morphism of algebraic spaces $T\times_{\mathcal{X}} \mathcal{W} \to T$, which is unramified. Hence, by \cite[][Tag 05W1]{stacks-project}, $\Delta_{\mathcal{W}/\mathcal{X}}$ is an open immersion. 
	Hence, $\Delta_{\mathcal{W}/\mathcal{X}}$ is a closed and open immersion and by diagram \ref{diag:inertia-pullback-open-substack}, so is $\Phi$.
	
	Let
	$$
	Z:=
	\left|\mathcal{I}_{\mathcal{X}}\times_{\mathcal{X}}\mathcal{W}\right|
	\setminus
	\left|\Phi(\mathcal{I}_{\mathcal{W}})\right|.
	$$
	
	Since \(\Phi\) is an open and closed immersion, \(Z\) is a closed subset of \(\left|\mathcal{I}_{\mathcal{X}}\times_{\mathcal{X}}\mathcal{W}\right|\).
	
	For every geometric point \(z:\operatorname{Spec}\bar{k}\to\mathcal{W}\), the geometric fiber of
	
	$$
	\operatorname{pr}_2:
	\mathcal{I}_{\mathcal{X}}\times_{\mathcal{X}}\mathcal{W}
	\longrightarrow\mathcal{W}
	$$
	
	over \(z\) identifies with \(\operatorname{Aut}_{\mathcal{X}}(f(z))\), since 
	
	\[
		\Spec{\bar{k}}\times_\mathcal{W}(\mathcal{I}_\mathcal{X} \times_\mathcal{X} \mathcal{W}) \cong \Spec{\bar{k}} \times_\mathcal{X}\mathcal{I}_\mathcal{X} \cong \Aut_\mathcal{X}(f(x)),
	\]
	where $\Spec{\bar{k}} \to \mathcal{X}$ is given by the composite $\Spec{\bar{k}}\to \mathcal{W} \xrightarrow{f} \mathcal{X}$.
	
	
	Under this identification, the image of \(\Phi\) in the fiber is precisely the image of the induced homomorphism
	
	$$
	\Phi_z:
	\operatorname{Aut}_{\mathcal{W}}(z)
	\longrightarrow
	\operatorname{Aut}_{\mathcal{X}}(f(z)).
	$$
	
	Consequently,
	
	$$
	Z\times_{\mathcal{W}}\operatorname{Spec}\bar{k}
	\neq\varnothing
	\quad\Longleftrightarrow\quad
	\Phi_z\text{ is not surjective}.
	$$
	
	Since \(\Phi_z\) is injective, it is an isomorphism if and only if it is surjective. Hence
	
	$$
	|\mathcal{U}|
	=
	|\mathcal{W}|\setminus\operatorname{pr}_2(Z).
	$$
	
	Finally, the morphism \(\operatorname{pr}_2\) is finite, being the base change of the finite morphism \(\mathcal{I}_{\mathcal{X}}\to\mathcal{X}\). In particular, it is a closed map. Since \(Z\) is closed, \(\operatorname{pr}_2(Z)\) is closed in \(|\mathcal{W}|\). Therefore, \(|\mathcal{U}|\) is open in \(|\mathcal{W}|\), as desired.
	
	
	\end{proof}
	Now by the affince case (Lemma \ref{lem:affine-quotient-finite-group}), $\mathcal{W}$ admits a coarse moduli space
	\[
	\pi_\mathcal{W}: \mathcal{W} \to W = \Specof{A^{G_x}}.
	\]
	
	Since $\mathcal{U}$ is open and $\pi_\mathcal{W}$ is a homeomorphism, $\pi_\mathcal{W}(\mathcal{U})$ is open in $\Specof{A^{G_x}}$.

	Now consider $\pi_\mathcal{W}^{-1}(X')$ as an open substack of $\mathcal{W}$. Open substacks of $\mathcal{W} = [\Specof{A}/G_x]$ correspond precisely to $G_x$-stable open subschemes $V\subset \Specof{A}$.
