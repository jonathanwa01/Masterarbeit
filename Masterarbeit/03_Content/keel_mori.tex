\chapter{Keel Mori Theorem}
The proof of the Keel-Mori theorem follows the Stacks Project. We proceed by first constructing moduli spaces on a special local class of algebraic stacks, called well-nigh affine stacks, and then gluing these local moduli spaces along
\etale\ overlaps.

\section{Local constructions}

From now on we will be working with Deligne-Momford stacks follwing jarod's notes.


\begin{lemma}[Local Structure Theorem of Deligne-Mumford Stacks]\label{lem:local-structure-theorem-dm-stacks}
	(\cite[][Theorem 4.2.11]{alper2026stacks})
	Let $\mathcal{X}$ be a separated DM stack and let $x:\Specof{k} \to  \mathcal{X}$ be a point with geometric stabilizer $G_x$. Then there exists an affine \etale\ morphism 
	\[
	f: [\Specof{A}/ G_x] \to \mathcal{X}
	\]
	sending $w\in |[\Specof{A}/ G_x]|$ to $x$, such that f induces an isomphism 
	\[
	G_w \xrightarrow{\simeq} G_x.
	\]
\end{lemma}


\begin{lemma}\label{lem:coarse-moduli-etale-local}
	Let $\pi : \mathcal{X} \to X$ be a morphism to an algebraic space. Suppose
	that there is an étale covering $\{f_i : X_i \to X\}$ such that
	\[
	\pi_i : \mathcal{X} \times_X X_i \longrightarrow X_i
	\]
	is a coarse moduli space for each $i$. Then $\pi : \mathcal{X} \to X$ is a
	coarse moduli space.
\end{lemma}

\begin{proof}
	We verify the two axioms of Definition~\ref{def:coarse-moduli-space}.
	
	\textbf{Geometric points}
	Let $k$ be an algebraically closed field and let
	$x : \Spec(k) \to X$ be a geometric point. Since $\{f_i : X_i \to X\}$
	is an étale covering, there exists an  $i$ and a lift
	$\tilde{x} : \Spec(k) \to X_i$ with $f_i \circ \tilde{x} = x$. Indeed,
	algebraically closed fields admit no nontrivial étale extensions, so étale
	covers are surjective on geometric points.
	
	Since $\pi_i$ is a coarse moduli space, the induced map
	\[
	|(\mathcal{X} \times_X X_i)(k)| \longrightarrow X_i(k)
	\]
	is a bijection onto isomorphism classes. The fiber of $\pi$ over $x$
	is canonically identified with the fiber of $\pi_i$ over $\tilde{x}$,
	via the base-change isomorphism
	\[
	\mathcal{X} \times_X \Spec(k)
	\;\cong\;
	(\mathcal{X} \times_X X_i) \times_{X_i} \Spec(k).
	\]
	Hence the bijection for $\pi_i$ implies the corresponding bijection for
	$\pi$.
	
	\medskip
	\textbf{Universality}
	Let $Y$ be an algebraic space and $f : \mathcal{X} \to Y$ a morphism.
	For each $i$, consider the composite
	\[
	\mathcal{X} \times_X X_i \longrightarrow \mathcal{X}
	\xrightarrow{\;f\;} Y.
	\]
	Since $\pi_i$ is a coarse moduli space, there exists a unique morphism
	$g_i : X_i \to Y$ such that the following commutes:

	factor the morphism
	\[
	\mathcal{X} \times_X X_{ij} \longrightarrow Y
	\]
	through the coarse moduli space $\pi_{ij}$ of $\mathcal{X} \times_X X_{ij}$, which exists since $\{X_{ij} \to X_i\}$ refines the given cover and
	$\mathcal{X} \times_X X_{ij}$ inherits a coarse moduli space. By
	uniqueness in the universal property of $\pi_{ij}$, we conclude
	\[
	g_i|_{X_{ij}} = g_j|_{X_{ij}}.
	\]
	
	The collection $\{g_i : X_i \to Y\}$ is compatible on the étale
	cover $\{X_i \to X\}$. Since $Y$ is an algebraic space, it is a
	sheaf in the étale topology, so the $g_i$ glue to a unique morphism
	$g : X \to Y$.
	

	The identity $f = g \circ \pi$ holds because it holds after pulling back
	along each $f_i$, and morphisms to algebraic spaces can be checked étale
	locally. Uniqueness of $g$ follows from the sheaf property of $Y$: any
	two morphisms $X \to Y$ agreeing on the cover $\{X_i \to X\}$ must
	coincide. Hence Axiom~(1) holds.
\end{proof}




\begin{lemma}[Affine quotient by a finite group]\label{lem:affine-quotient-finite-group}
	Let $G$ be a finite group acting on an affine scheme $\Spec(A)$. Then the canonical morphism
	\[
	\pi : \Spec(A) \longrightarrow \Spec(A^G)
	\]
	induces a morphism of stacks
	\[
	[\Spec(A)/G] \longrightarrow \Spec(A^G),
	\]
	and this morphism is a coarse moduli space.
\end{lemma}

\begin{proof}
	We check the two defining properties of a coarse moduli space.
	
	\emph{(i) Geometric points.}
	The induced map on geometric points identifies the set of $G$-orbits in
	$\Spec(A)(\bar{k})$ with the set of points of $\Spec(A^G)(\bar{k})$.
	The morphism $\Spec(A) \to \Spec(A^G)$ is finite and surjective, since
	$A$ is integral over $A^G$ for a finite group action. In particular, two
	geometric points have the same image in $\Spec(A^G)$ if and only if they lie in the same $G$-orbit.
	
	\emph{(2) Universality.}
	Let $Y$ be an algebraic space and let
	\[
	f : [\Spec(A)/G] \longrightarrow Y
	\]
	be a morphism. Since $f$ is $G$-invariant, its restriction to $\Spec(A)$
	corresponds locally on an étale cover of $Y$ to a $G$-invariant morphism of
	affine schemes. Equivalently, on coordinate rings it is given by a
	$G$-invariant homomorphism, hence factors uniquely through $A^G$.
	By descent for morphisms to algebraic spaces, these local factorizations glue
	to a unique morphism $\Spec(A^G)\to Y$ making the diagram commute.
	
	Thus $\pi : [\Spec(A)/G]\to \Spec(A^G)$ satisfies the universal property of a
	coarse moduli space.
\end{proof}

\begin{theorem}[Keel-Mori Theorem]
	Let $\mathcal{X}$ be a Deligne-Mumford stack separated and of finite type over a field. Then there exists a coarse moduli space
	\[
	\pi: \mathcal{X}\to X.
	\]
\end{theorem}

\begin{remark}
	By remark \ref{rem:dm-stack-finite-inertia}, the classical finite inertia condition can be dropped.
\end{remark}

\begin{proof}
	\medskip
	\textbf{Construct covering with coarse moduli space}
	
	Let $x:\Specof{k} \to \mathcal{X}$ be a geometric point. By the local structure theorem of DM stacks (\ref{lem:local-structure-theorem-dm-stacks}), we can find an \etale\ neighborhood of $x$ denoted by:
	\[
	f: (\mathcal{W},w) \to (\mathcal{X}, x),
	\]
	where $\mathcal{W} = [\Specof{A}/G_x]$, such that $f$ induces an isomorphism between the stabilizer groups
	\[
	G_w \xrightarrow{\simeq} G_x.
	\]
	We now shrink $\mathcal{W}$ to a smaller \etale\ neighbourhood, such that $f$ induces isomorphisms on all stabilizer groups.
	
	Let $\mathcal{U}\subset \mathcal{W}$ be the locus consisting of points $z\in |\mathcal{W}|$, such that $f$ induces isomorphisms $G_z \xrightarrow{\simeq} G_x$.
	
	We claim, that $\mathcal{U}$ is open. 
	
	\begin{proof}[Proof of the claim]
		
	By a diagram chase, the following is 2-cartesian:

	In particular, for every geometric point $z:\Spec{\bar{k}} \to \mathcal{W}$, the induced map on stabilizer groups
	\[
	\Phi_{z} \colon \mathrm{Aut}_W(z) \longrightarrow \mathrm{Aut}_X(z)
	\]
	is injective, and its image is an open and closed subgroup of $\mathrm{Aut}_X(z)$.
	\[
	\operatorname{pr}_2:\mathcal{I}_\mathcal{X} \times_\mathcal{X} \mathcal{W}\to \mathcal{W}
	\]
	is finite as a pullback of the finite morphism $\mathcal{I}_\mathcal{X} \to \mathcal{X}$. In particular, $\operatorname{pr}_2$ is closed. 
	
	Consider the closed subset 
	\[
	Z:= |\mathcal{I}_\mathcal{X}\times_\mathcal{X}\mathcal{W}| \setminus |\mathcal{I}_\mathcal{W}|.
	\]
	
	This is the locus of points of the inertia of $\mathcal{X}$ over $\mathcal{W}$ that do not come from inertia of $\mathcal{W}$. Equivalently, for a geometric point $z:\Spec{\bar{k}} \to \mathcal{W}$, the fiber of $z$ over $Z$ is nonempty if and only if the map
	\[
	\Phi_{z} \colon \mathrm{Aut}_\mathcal{W}(z) \to \mathrm{Aut}_\mathcal{W}(z)
	\]
	fails to be surjective. Since $\Phi_{z}$ is injective and a closed and open immersion, the failure of $\Phi_{z}$ being an isomorphism is equivalent to $\Phi_{z}$ failing to be surjective.
	Thus the image $\operatorname{pr}_2(Z) \subset |\mathcal{W}|$ is exactly the locus of points $z \in |W|$ at which $\Phi_z$ fails to be surjective and hence, $|\mathcal{W}|\setminus \operatorname{pr}_2(Z)$ is precisely the locus, where $\Phi_z$ is an isomorphism, which is identified with $\mathcal{U}$.
	
	Since $\operatorname{pr}_2$ is closed, $\operatorname{pr}_2(Z) \subset |\mathcal{W}|$ is closed.
	and therefore, $\mathcal{U}$, as being identified with $\operatorname{pr}_2(Z) \subset |\mathcal{W}|$ in $|\mathcal{W}|$, is open.

	\pi_\mathcal{W}: \mathcal{W} \to W = \Specof{A^{G_x}}.

	which induces isomorphism on all geometric stabilizer groups, since $\mathcal{X}' \subset \mathcal{U}$.
	This yields a cover $\{\mathcal{X}_i \xhookrightarrow{f_i} \mathcal{X}\}_i$ by open substacks of the form $[U_i/G_x]$ with affine $U_i = \Specof{A_i}$, such that $f_i$ induces isomorphism on all geometric stabilizer groups.
	
	
	\medskip
	\textbf{Descending the local maps to the overlaps}
	
	We first construct the global map $\pi : \mathcal{X} \to X$, then verify it
	is a coarse moduli space.
	
	For each pair $(i,j)$, consider the fiber product
	\[
	\mathcal{X}_{ij} := \mathcal{X}_i \times_{\mathcal{X}} \mathcal{X}_j.
	\]
	Since $f_i$ and $f_j$ are representable and \etale, the two projections
	\[
	p_i : \mathcal{X}_{ij} \to \mathcal{X}_i,
	\qquad
	p_j : \mathcal{X}_{ij} \to \mathcal{X}_j
	\]
	are also representable and étale. Moreover, for every geometric point
	$\bar{x} \to \mathcal{X}_{ij}$, the induced map on stabilizer groups
	\[
	\Aut_{\mathcal{X}_{ij}}(\bar{x})
	\xrightarrow{\;\sim\;}
	\Aut_{\mathcal{X}_i}(p_i(\bar{x}))
	\]
	is an isomorphism, since $f_i$ induces isomorphisms on all geometric
	stabilizer groups by hypothesis, and $\mathcal{X}_{ij} \to \mathcal{X}_i$
	is a base change of $f_j$ along $f_i$.
	
	In particular, $\mathcal{X}_{ij}$ is a DM stack with finite
	inertia. Since $\mathcal{X}_i = [U_i/G_i]$ is a quotient of an affine by
	a finite group, the base change
	\[
	\mathcal{X}_{ij} \cong
	[U_i \times_{\mathcal{X}} \mathcal{X}_j \;/\; G_i]
	\]
	is again a quotient of an affine scheme by a finite group. In particular,
	$\mathcal{X}_{ij}$ admits a coarse moduli space
	\[
	\pi_{ij} : \mathcal{X}_{ij} \longrightarrow X_{ij},
	\]
	and the morphisms $\pi_i \circ p_i : \mathcal{X}_{ij} \to X_i$ and
	$\pi_j \circ p_j : \mathcal{X}_{ij} \to X_j$ both factor uniquely through
	$\pi_{ij}$, by the universal property of the coarse moduli space of $\mathcal{X}_{ij}$:
	\[
	\phi_{ij} : X_{ij} \to X_i,
	\qquad
	\phi_{ji} : X_{ij} \to X_j.
	\]
	We claim that $\phi_{ij}$ and $\phi_{ji}$ are both \etale. Indeed,
	$\pi_i \circ p_i$ is étale (as a composition of étale morphisms), and
	$\pi_i$ is the coarse moduli space of $\mathcal{X}_i = [U_i/G_i]$. By
	Lemma~\ref{lem:descending-etale-to-quotient}, any \etale\ morphism from
	$\mathcal{X}_{ij}$ to the algebraic space $X_i$ that is an isomorphism
	on geometric stabilizers descends to an étale morphism $X_{ij} \to X_i$.
	The same argument applies to $\phi_{ji}$.
	
	\medskip
	\textbf{Forming the global algebraic space via the étale equivalence relation.}
	
	Define
	\[
	U := \coprod_i X_i,
	\qquad
	R := \coprod_{i,j} X_{ij}
	\]
	with source and target maps \(s,t:R\rightrightarrows U\). Those are \etale, and by construction they form an \etale\ equivalence relation. Hence, by Lemma~\ref{lem:etale-eq-quotient-of-affines}, the quotient sheaf
	\[
	X := (U/R)^\sharp
	\]
	is an algebraic space, and the canonical map \(p:U\to X\) is étale and surjective.
	
	\medskip
	\textbf{Glue the local coarse moduli spaces via \etale\ descent.}
	
	The overlap maps \(X_{ij} \rightrightarrows X_i, X_j\) form fpqc descent data
	for the affine schemes \(X_i\). Hence, by Theorem~\ref{thm:fpqc-descent-affine-schemes},
	these local pieces glue to a global algebraic space \(X\).
	
	\medskip
	\textbf{Constructing the morphism $\pi : \mathcal{X} \to X$.}
	
	The local morphisms $\mathcal{X}_i \to X$ are compatible on overlaps, since
	the identifications on $X_{ij}$ are precisely the ones used in the quotient
	construction of $X$. Hence they glue to a unique morphism
	\[
	\pi : \mathcal{X} \longrightarrow X.
	\]
	
	\medskip
	\textbf{Verifying the coarse moduli space axioms.}
	
	\medskip
	\textbf{Verifying the coarse moduli space axioms.}
	
	The coarse moduli property is \etale\ local on the target. Therefore, by
	Lemma~\ref{lem:coarse-moduli-etale-local}, it suffices to check the pullbacks
	along the étale cover \(\{X_i \to X\}\). Since each
	\[
	\pi_i : \mathcal{X} \times_X X_i \to X_i
	\]
	is a coarse moduli space by construction, it follows that
	\[
	\pi : \mathcal{X} \to X
	\]
	is a coarse moduli space.
	
	\medskip
	
	This completes the construction of the coarse moduli space
	$\pi : \mathcal{X} \to X$.
\end{proof}
