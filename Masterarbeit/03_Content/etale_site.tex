\chapter{\Etale\ Site}


\begin{definition}[\Etale\ covering]
	(\cite[][(Definition 20.21), p.~101]{GoertzWedhornAG2})
	An \etale\ covering of a scheme $X$ is a family of \etale\ morphisms $(g_i: U_i\to X)_{i\in I}$, such that $X = \bigcup_{i\in I} g_i(U_i)$. Let $\mathcal{U} =(U_i\to X)_{i\in I}$ and $\mathcal{V} = (V_j\to X)_{j\in J}$ be \etale\ coverings, than we call $\mathcal{V}$ a refinement of $\mathcal{U}$, if there exists a map $\alpha: J\to I$, such that for each $j\in J$ there exists a morphism $V_j\to U_{\alpha(j)}$, such that the following commute:
	% https://q.uiver.app/#q=WzAsMyxbMCwwLCJWX2oiXSxbMiwwLCJVX3tcXGFscGhhX2p9Il0sWzEsMSwiWCJdLFswLDJdLFsxLDJdLFswLDEsIlxcdmFycGhpX2oiXV0=
	\[\begin{tikzcd}
		{V_j} && {U_{\alpha(j)}} \\
		& X
		\arrow[from=1-1, to=1-3]
		\arrow[from=1-1, to=2-2]
		\arrow[from=1-3, to=2-2]
	\end{tikzcd}\]
\end{definition}

\begin{definition}[\Etale\ Site]
	We write $\Sch_{et}$ for the \etale\ site $(\Sch, J)$, where $J(X)$ contains all \etale\ coverings of $X$.
\end{definition}

\begin{proposition}\label{prop:affine-etale-refinement}
	(\cite[][p.~101]{GoertzWedhornAG2})
	Let $X$ be a scheme. Then every \etale\ covering $\mathcal{U} =(g_i: U_i\to X)_{i\in I}$ of $X$ admits a refinement $\mathcal{V} = (V_j\to X)_{j\in J}$ of affine schemes. If $X$ quasi-compact, the refinement can be chosen to be finite.
\end{proposition}

\begin{proof}
	For each $(U_i)_{i\in I}$ chose an open affine cover $U_i = \bigcup_{j\in J_i} V_{ij}$. Now consider the composite $V_{ij} \hookrightarrow U_i\xrightarrow{g_i} X$ . Each $V_{ij}\hookrightarrow U_i$ is an open immersion, hence \etale. Let $J = \coprod_{i\in I}J_i = \{(i, j) \mid i\in I, j\in J_i\}$.  This yields an \etale\ cover $\mathcal{V} = (V_{ij} \to X)_{(i, j)\in J}$, since the composite of two \etale\ morohisms is \etale\ and the image covers $X$ by construction.
	$\mathcal{V}$ is a refinement of $\mathcal{U}$ via $\alpha: J\to I, (i, j) \mapsto i$.
	If $X$ is quasi-compact, there exists a finite subset $J'\subseteq J$ such that the opens $g_{\alpha(j)}(V_j)$ cover $X$.
\end{proof}

\begin{lemma}\label{lem:faithfully_flat_implies_injective}
	(\cite[][Ch.~3, Ex.~16]{AtiyahMacdonaldICA})
	Let $f:R\to S$ be faithfully flat ring, then $f$ is injective.
\end{lemma}

\begin{lemma}[Representable presheaves of affine schemes]\label{lem:repr-presheafs-of-affines}
	Let $A$ be an commutative ring and let $\{\alpha_i: R\to R_i\}_{i\in I}$ be a morphism of \etale\ ring maps, such that the induced morphism of schemes 
	\[
	\coprod_{i\in I}\Specof{R_i} \to \Specof{R}
	\]
	is surjective, i.e. an \etale\ covering.\\
	Then the sequence 
	% https://q.uiver.app/#q=WzAsMyxbMCwwLCJcXEhvbShBLCBSKSJdLFsxLDAsIlxccHJvZF9pXFxIb20oQSwgUl9pKSJdLFsyLDAsIlxccHJvZF97aSxqfVxcSG9tKEEsIFJfaVxcb3RpbWVzX1JSX2opIl0sWzAsMV0sWzEsMiwiXFxiZXRhIiwyLHsib2Zmc2V0IjoyfV0sWzEsMiwiXFxhbHBoYSIsMCx7Im9mZnNldCI6LTJ9XV0=
	\[\begin{tikzcd}
		{\Hom(A, R)} & {\prod_i\Hom(A, R_i)} & {\prod_{i,j}\Hom(A, R_i\otimes_RR_j)}
		\arrow[from=1-1, to=1-2]
		\arrow[shift right=2, from=1-2, to=1-3]
		\arrow[shift left=2, from=1-2, to=1-3]
	\end{tikzcd}\]
	is exact, where the two parallel morphisms are induced by $R_i\to R_i\otimes_RR_j,r\mapsto r\otimes_R 1$ and $R_j\to R_i\otimes R_j, r\mapsto 1\otimes_R r$.
\end{lemma}

\begin{proof}
	Since $\Specof{R}$ is quasi compact, by \ref{prop:affine-etale-refinement} we can refine the cover to a finite cover. So without loss of generality we assume $I$ to be finite.
	

	Assume we are given $\varphi_i:A\to R_i$, such that 
	\begin{align}\label{eq:affine_compatibility}
			\varphi_i\otimes_R 1 = 1\otimes_R \varphi_j: A\to R_i\otimes_R R_j
	\end{align}
	for all $i,j$, i.e. the following commutes.
	% https://q.uiver.app/#q=WzAsNSxbMCwxLCJBIl0sWzEsMV0sWzEsMCwiUl9pIl0sWzEsMiwiUl9qIl0sWzIsMSwiUl9pXFxvdGltZXNfUiBSX2oiXSxbMCwyLCJcXHZhcnBoaV9pIl0sWzAsMywiXFx2YXJwaGlfaiIsMl0sWzMsNCwiMVxcb3RpbWVzIGlkIiwyXSxbMiw0LCJpZFxcb3RpbWVzX1IgMSJdXQ==
	\[\begin{tikzcd}
		& {R_i} \\
		A & {} & {R_i\otimes_R R_j} \\
		& {R_j}
		\arrow["{id\otimes_R 1}", from=1-2, to=2-3]
		\arrow["{\varphi_i}", from=2-1, to=1-2]
		\arrow["{\varphi_j}"', from=2-1, to=3-2]
		\arrow["{1\otimes id}"', from=3-2, to=2-3]
	\end{tikzcd}\]
	
	We will construct a morphism unique $\varphi: A\to R$, such that $\alpha_i\circ \varphi = \varphi_i$, i.e. the following commutes:
	% https://q.uiver.app/#q=WzAsMyxbMCwwLCJBIl0sWzEsMCwiUiJdLFsxLDEsIlJfaSJdLFswLDEsIlxcdmFycGhpIl0sWzEsMiwiXFxhbHBoYV9pIl0sWzAsMiwiXFx2YXJwaGlfaSIsMl1d
	\begin{equation}\label{diagram:lift_varphi}
		\begin{tikzcd}
			A & R \\
			& {R_i}
			\arrow["\varphi", from=1-1, to=1-2]
			\arrow["{\varphi_i}"', from=1-1, to=2-2]
			\arrow["{\alpha_i}", from=1-2, to=2-2]
		\end{tikzcd}
	\end{equation}
	
	The $\varphi_i:A\to R_i$ induce a morphism $\Phi:A \to S$ given by $\Phi(a) = (\varphi_i(a))_i$. Let $S:= \prod_i R_i$ and $\alpha: R\to S$ be the canonical ring map induced by the $\alpha_i$, i.e. $\alpha(r) = (\alpha_i(r))_i$. By \cite[][Tag 00HQ]{stacks-project}, a flat ring map $R \to S$ is faithfully flat if and only if $\Spec(S) \to \Spec(R)$ is surjective. Since the $R_i$ cover $R$, and $S$ is free, hence flat, the induced map $R\to S$ is faithfully flat.
	
	
	Let $\delta_1, \delta_2:S\to S\otimes_R S$ be given by $\delta_1 = id\otimes_R 1, \delta_2 = 1\otimes_R  id$. Since, the cover is finite, $S$ is finitely generated. Hence we obtain a natural isomorphism
	\[
	S\otimes_R S \cong \big(\prod_i R_i \big)\otimes_R \big(\prod_j R_j\big) \cong \prod_i \big( R_i\otimes_R \prod_jR_j) \cong \prod_{i,j} \big(R_i\otimes_R R_j\big)
	\]
	The morphisms $\delta_1, \delta_2$ induce $\tilde{\delta_1}, \tilde{\delta_2}: S\to \prod_{i,j} R_i\otimes_R R_j$ defined by $\tilde{\delta_1}((r_i)_i) = (r_i\otimes_R 1)_{i,j}$ and $\tilde{\delta_2} = (1\otimes_R r_j)_{i,j}$.
	Now precomposing with $\Phi$ yields 
	\begin{align*}
			\tilde{\delta_1}(\Phi(a)) = (\varphi_i(a)\otimes_R 1)_{i,j} \\
			\tilde{\delta_2}(\Phi(a)) = (1\otimes_R \varphi_j(a))_{i,j}
	\end{align*}
	Now the compatibility assumption \ref{eq:affine_compatibility} yields 
	\begin{align}\label{eq:delta_phi}
		\delta_1 \circ \Phi = \delta_2 \circ \Phi
	\end{align}

	Consider the following diagram:
	% https://q.uiver.app/#q=WzAsNCxbMCwwLCIwIl0sWzEsMCwiUiJdLFsyLDAsIlMiXSxbMywwLCJTXFxvdGltZXNfUiBTIl0sWzAsMV0sWzEsMiwiXFxhbHBoYSJdLFsyLDMsIlxcZGVsdGFfMS1cXGRlbHRhXzIiXV0=
	\begin{equation}\label{diagram:exactness_R_S}
		\begin{tikzcd}
			0 & R & S & {S\otimes_R S}
			\arrow[from=1-1, to=1-2]
			\arrow["\alpha", from=1-2, to=1-3]
			\arrow["{\delta_1-\delta_2}", from=1-3, to=1-4]
		\end{tikzcd}
	\end{equation}


	Claim: This is exact, i.e. $R \cong ker(\delta_1-\delta_2) \cong Eq(\delta_1, \delta_2)$.
	
	By lemma \ref{lem:faithfully_flat_implies_injective} $\alpha$ is injective.
	Let $r\in R$, then $\delta_1(\alpha(r)) = \alpha(r) \otimes_R 1 = 1\otimes_R  \alpha(r) = \delta_2 (\alpha(r))$, so $im(\alpha) \subset Eq(\delta_1, \delta_2)$.
	
	Now assume $s\in ker(\delta_1 - \delta_2)$, i.e. $s\otimes_R 1 = 1\otimes_R s$. Let $\bar{s}$ be the image in $\pi: S\to S/\alpha(R)$. Tensoring $\pi$ with $S$ yields $\pi\otimes_R id: S\otimes_R S\to (S/\alpha(R))\otimes_R S$. Next apply this to  $s\otimes_R 1 = 1\otimes_R s$ yields
	\[
	\bar{s}\otimes 1 = (\pi\otimes_R id)(s\otimes_R 1) = (\pi\otimes_R id)(1\otimes_R s) = \pi(1)\otimes_R s = 0\otimes s = 0.
	\]
	
	So $\bar{s} \otimes_R 1 = 0$ in $(S/\alpha(R))\otimes_R S$. Now since $\alpha$ is faithfully flat, 
	\[
	S/\alpha(R) \to (S/\alpha(R))\otimes_R S
	\]is injective. Hence $\bar{s} = 0$, i.e. $s \in \alpha(R)$, which proves the claim.
	
	Now by \ref{eq:delta_phi} and by exactness of \ref{diagram:exactness_R_S}, the universal property of the kernal $R$ induces a unique morphism $\varphi:A\to R$, such that the following commutes:
	% https://q.uiver.app/#q=WzAsNSxbMSwxLCJSIl0sWzIsMSwiUyJdLFsyLDAsIkEiXSxbMCwxLCIwIl0sWzMsMSwiU1xcb3RpbWVzX1IgUyJdLFsyLDEsIlxcUGhpIl0sWzAsMSwiXFxhbHBoYSJdLFszLDBdLFsxLDQsIlxcZGVsdGFfMSAtXFxkZWx0YV8yIiwyXSxbMiw0XSxbMiwwLCJcXHZhcnBoaSIsMix7InN0eWxlIjp7ImJvZHkiOnsibmFtZSI6ImRhc2hlZCJ9fX1dXQ==
	\[\begin{tikzcd}
		&& A \\
		0 & R & S & {S\otimes_R S}
		\arrow["\varphi"', dashed, from=1-3, to=2-2]
		\arrow["\Phi", from=1-3, to=2-3]
		\arrow[from=1-3, to=2-4]
		\arrow[from=2-1, to=2-2]
		\arrow["\alpha", from=2-2, to=2-3]
		\arrow["{\delta_1 -\delta_2}"', from=2-3, to=2-4]
	\end{tikzcd}\]
	
	Hence, $(\alpha_i(\varphi(a))_i = \alpha(\varphi(a)) = \Phi(a) = (\varphi_i(a))_i$, which proves commutativity of \ref{diagram:lift_varphi}.

\end{proof}

This allows us to restrict to finite affine \etale\ covers. Such reductions are standard in
algebraic geometry: many statements can then be checked on affines and translated into
commutative algebra.

\begin{theorem}[Representable presheaves of schemes]
	Let $X$ be a scheme. Consider the representable presheaf 
	\[h_X: (\SchOver{Et})^\op \to \Set\]
	on the \etale\ site of \Sch. Then $h_X$ is a sheaf in the \etale\ topology.
\end{theorem}

	Now we assume $X$ to be any scheme.
	Let $\{W_i \hookrightarrow X\}_{i\in I}$ be an open affine cover of the scheme $X$, and set
	\[
	W_{ij} := W_i \cap W_j \qquad (i,j\in I).
	\]
	
	Then via Zariksi gluing we have a natural isomorphim of schemes
	\[
	X \cong \Coeq\big(\coprod_{i, j} W_{ij}  \;\rightrightarrows\; \coprod_i W_i\big).
	\]
	
	Then  for every scheme $S$, by continuity of $\Hom$ there is a natural isomorphism
	\[
	h_X(S)=\Hom_\Sch(S,X)
	\;\cong\;
	\Eq\!\left(
	\prod_{i} \Hom_\Sch(S,W_i)
	\;\rightrightarrows\;
	\prod_{i,j} \Hom_\Sch(S,W_{ij})
	\right),
	\]
	where the two arrows are induced by restricting a morphism $S\to W_i$ to
	$S\to W_{ij}$ along the open immersions $W_{ij}\hookrightarrow W_i$ and
	$W_{ij}\hookrightarrow W_j$.
	
	\medskip
	
	Via Yoneda, this is equivalent to an isomorphism of presheaves in $\Psh{\Sch}$,
	\[
	h_X \;\cong\;
	\Eq\!\left(
	\prod_{i} h_{W_i}
	\;\rightrightarrows\;
	\prod_{i,j} h_{W_{ij}}
	\right).
	\]
	In particular, $h_X$ is a limit of the diagram of the $h_{W_i}$
	and $h_{W_{ij}}$, hence also
	computed sectionwise for presheaves. 
	
	By the frist part of the proof, $h_{W_i}$ and all $h_{W_{ij}}$ are sheaves.
	By corrolary \ref{cor:limit-sheaf}, the category $\Sh{\Sch_{et}}$ is closed under limits, hence $h_X$ is a sheaf too.
